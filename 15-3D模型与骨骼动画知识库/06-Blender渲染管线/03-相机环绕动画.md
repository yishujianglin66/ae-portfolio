# 相机环绕动画

## 问题场景

本项目使用相机360°环绕代替骨骼动画（因为宽袍模型蒙皮质量不够）。需要创建平滑的相机环绕路径，保持角色始终在画面中心。

## 核心原理

### 圆形环绕轨道

```python
import bpy
import math
from mathutils import Vector

def create_camera_orbit(target_pos, radius=8.0, height=2.0, 
                        frame_start=1, frame_end=60, loops=1):
    """
    创建相机环绕动画
    target_pos: 注视目标（角色中心）
    radius: 环绕半径
    height: 相机高度（相对target）
    loops: 环绕圈数
    """
    scene = bpy.context.scene
    
    # 创建相机
    cam_data = bpy.data.cameras.new("Camera")
    cam_obj = bpy.data.objects.new("Camera", cam_data)
    bpy.context.collection.objects.link(cam_obj)
    scene.camera = cam_obj
    
    # 相机设置
    cam_data.lens = 50  # 焦距
    cam_data.clip_end = 100
    
    # 创建空物体作为注视目标
    target = bpy.data.objects.new("CamTarget", None)
    bpy.context.collection.objects.link(target)
    target.location = target_pos
    
    # Track To约束（始终注视目标）
    track = cam_obj.constraints.new(type='TRACK_TO')
    track.target = target
    track.track_axis = 'TRACK_NEGATIVE_Z'
    track.up_axis = 'UP_Y'
    
    # 关键帧动画
    total_frames = frame_end - frame_start
    for frame in range(frame_start, frame_end + 1):
        t = (frame - frame_start) / total_frames  # 0~1
        angle = t * math.pi * 2 * loops  # 角度
        
        # 圆形轨道位置
        x = target_pos[0] + radius * math.cos(angle)
        y = target_pos[1] + radius * math.sin(angle)
        z = target_pos[2] + height
        
        cam_obj.location = (x, y, z)
        cam_obj.keyframe_insert(data_path="location", frame=frame)
    
    print(f"[Camera] Orbit: radius={radius}, height={height}, {loops} loop(s)")
    return cam_obj, target
```

        x = target_pos[0] + radius_x * math.cos(angle)
        y = target_pos[1] + radius_y * math.sin(angle)

### 相机微摆（增加动感）

```python
def add_camera_sway(cam_obj, frame_start, frame_end, amplitude=0.1):
    """添加相机微摆（呼吸感）"""
    import math
    
    for frame in range(frame_start, frame_end + 1):
        t = (frame - frame_start) / (frame_end - frame_start)
        
        # 正弦微摆
        sway_x = math.sin(t * math.pi * 4) * amplitude
        sway_z = math.cos(t * math.pi * 6) * amplitude * 0.5
        
        # 叠加到现有关键帧
        current_loc = cam_obj.location.copy()
        cam_obj.location = (
            current_loc.x + sway_x,
            current_loc.y,
            current_loc.z + sway_z
        )
        cam_obj.keyframe_insert(data_path="location", frame=frame)
```
