# 渲染性能优化

## 问题场景

Blender background模式渲染60帧1080p动画时，渲染时间可能从几分钟到几小时不等。需要针对三渲二场景优化渲染速度，同时保持画面质量。

## 核心原理

### EEVEE性能优化

```python
import bpy

def optimize_eevee_render():
    """EEVEE渲染性能优化"""
    scene = bpy.context.scene
    
    # 1. 降低采样数（EEVEE默认64，三渲二可降到16）
    scene.eevee.taa_render_samples = 16  # 渲染采样
    scene.eevee.taa_samples = 8          # 视口预览采样
    
    # 2. 关闭不需要的效果
    scene.eevee.use_ssr = False           # 屏幕空间反射
    scene.eevee.use_ssr_refraction = False
    scene.eevee.use_gtao = False          # 环境光遮蔽
    scene.eevee.use_bloom = False         # 泛光
    
    # 3. 阴影优化
    scene.eevee.shadow_cube_size = '512'   # 立方体阴影贴图
    scene.eevee.shadow_cascade_size = '1024'  # 级联阴影
    scene.eevee.use_shadow_high_bitdepth = False
    
    # 4. 体积光关闭（三渲二不需要）
    scene.eevee.volumetric_samples = 0
```

### Cycles性能优化

```python
def optimize_cycles_render():
    """Cycles渲染性能优化"""
    scene = bpy.context.scene
    
    # 1. 设备选择
    prefs = bpy.context.preferences.addons['cycles'].preferences
    prefs.compute_device_type = 'CUDA'  # 或 'OPTIX'（更快）
    prefs.get_devices()
    for device in prefs.devices:
        device.use = True  # 启用所有GPU
    
    scene.cycles.device = 'GPU'
    
    # 2. 采样优化
    scene.cycles.samples = 64  # 主采样
    scene.cycles.use_denoising = True  # 降噪（大幅减少所需采样）
    
    # 3. 光路反弹限制
    scene.cycles.max_bounces = 4        # 总反弹
    scene.cycles.diffuse_bounces = 2    # 漫反射
    scene.cycles.glossy_bounces = 2     # 高光
    scene.cycles.transmission_bounces = 2  # 透射
    scene.cycles.transparent_max_bounces = 4  # 透明
    
    # 4. Tile大小（GPU渲染用大tile）
    scene.cycles.tile_size = 256  # GPU推荐256-512
    
    # 5. 持久化数据（多帧渲染时保持GPU内存）
    scene.render.use_persistent_data = True
```

### 通用优化

```python
def general_render_optimization():
    """通用渲染优化"""
    scene = bpy.context.scene
    
    # 1. 分辨率缩放（测试阶段用50%）
    scene.render.resolution_percentage = 100  # 正式渲染100%
    # scene.render.resolution_percentage = 50  # 测试用50%
    
    # 2. 简化模式（视口预览）
    # bpy.context.scene.render.use_simplify = True
    # bpy.context.scene.render.simplify_subdivision = 1
    
    # 3. 关闭不需要的Modifier
    for obj in bpy.data.objects:
        if obj.type == 'MESH':
            for mod in obj.modifiers:
                if mod.type == 'SUBSURF':
                    mod.render_levels = 1  # 降低渲染细分
                elif mod.type == 'LINE_ART':
                    # LineArt计算量大，测试时可关闭
                    pass
    
    # 4. 纹理压缩
    for img in bpy.data.images:
        if img.source == 'FILE':
            # 大纹理缩放（4K→2K）
            if img.size[0] > 2048:
                img.scale(2048, 2048)
    
    # 5. 多线程
    scene.render.threads_mode = 'AUTO'  # 自动线程数
```

### 渲染时间估算

```python
import time

def benchmark_render(frame_count=5):
    """渲染基准测试"""
    scene = bpy.context.scene
    
    # 渲染前N帧估算总时间
    start = time.time()
    
    original_end = scene.frame_end
    scene.frame_end = scene.frame_start + frame_count - 1
    
    bpy.ops.render.render(animation=True)
    
    elapsed = time.time() - start
    per_frame = elapsed / frame_count
    
    # 恢复
    scene.frame_end = original_end
    
    total_estimate = per_frame * (original_end - scene.frame_start + 1)
    print(f"[Benchmark] {per_frame:.1f}s/frame, "
          f"total estimate: {total_estimate/60:.1f}min")
    
    return per_frame, total_estimate
```

### 内存管理

```python
def manage_memory():
    """渲染内存管理"""
    import gc
    
    # 1. 清理未使用的数据块
    for block_collection in [
        bpy.data.meshes, bpy.data.materials,
        bpy.data.textures, bpy.data.images
    ]:
        for block in block_collection:
            if block.users == 0:
                block_collection.remove(block)
    
    # 2. Python垃圾回收
    gc.collect()
    
    # 3. 渲染前释放视口缓存
    # （background模式无此问题）
```

## 常见陷阱

### 陷阱1：GPU渲染比CPU慢
```python
# 原因：场景太小，GPU启动开销 > 渲染收益
# 解决：小场景用CPU，大场景用GPU
# 判断标准：面数 > 100万时用GPU
```

### 陷阱2：persistent_data导致内存溢出
```python
# 多帧渲染时GPU内存持续增长
# 解决：每100帧清理一次
if scene.frame_current % 100 == 0:
    scene.render.use_persistent_data = False
    bpy.ops.render.render(animation=False)
    scene.render.use_persistent_data = True
```

### 陷阱3：LineArt修改器拖慢渲染

```python
# LineArt每帧都要重新计算边缘

```
