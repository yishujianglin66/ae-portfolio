# 渲染错误诊断

## 问题场景

Blender background模式渲染可能遇到各种错误：GPU内存不足、材质节点错误、帧输出失败、渲染黑屏等。由于background模式无GUI，错误诊断依赖日志分析。

## 核心原理

### 常见渲染错误分类

```python
# 错误类型1：GPU内存不足
# 日志特征：
# "CUDA error: out of memory"
# "OptiX error: CUDA out of memory"
# 原因：场景面数/纹理超出GPU显存
# 解决：降低纹理分辨率、减少tile_size、用CPU渲染

# 错误类型2：材质节点错误
# 日志特征：
# "Error: Node type ... not available"
# "Shader node ... has no input ..."
# 原因：API版本差异、节点名称变更
# 解决：检查Blender版本兼容性

# 错误类型3：输出路径错误
# 日志特征：
# "Error: Cannot open file ... for writing"
# "Save Error: failed to write frame"
# 原因：路径不存在、权限不足、磁盘空间不足
# 解决：确保目录存在、检查权限

# 错误类型4：渲染黑屏
# 日志特征：无错误，但输出全黑
# 原因：相机未对准物体、灯光未添加、材质全黑
# 解决：检查相机朝向、灯光存在性
```

### 渲染前检查脚本

```python
import bpy
import os

def pre_render_check():
    """渲染前完整性检查"""
    scene = bpy.context.scene
    errors = []
    warnings = []
    
    # 1. 相机检查
    if not scene.camera:
        errors.append("No active camera set!")
    else:
        cam = scene.camera
        # 检查相机是否对准场景中心
        cam_dir = cam.matrix_world.to_quaternion() @ Vector((0, 0, -1))
        print(f"[Check] Camera direction: {cam_dir}")
    
    # 2. 灯光检查
    lights = [obj for obj in bpy.data.objects if obj.type == 'LIGHT']
    if not lights:
        warnings.append("No lights in scene - render may be dark")
    
    # 3. 输出路径检查
    output_dir = os.path.dirname(scene.render.filepath)
    if output_dir and not os.path.exists(output_dir):
        errors.append(f"Output directory not found: {output_dir}")
        os.makedirs(output_dir, exist_ok=True)
    
    # 4. 材质检查
    for obj in bpy.data.objects:
        if obj.type == 'MESH':
            if not obj.data.materials:
                warnings.append(f"Object '{obj.name}' has no material")
            for mat in obj.data.materials:
                if mat and mat.use_nodes:
                    # 检查是否有输出节点连接
                    output_node = None
                    for node in mat.node_tree.nodes:
                        if node.type == 'OUTPUT_MATERIAL':
                            output_node = node
                    if not output_node:
                        warnings.append(
                            f"Material '{mat.name}' has no output node")
    
    # 5. 渲染设置检查
    if scene.render.resolution_x < 100:
        warnings.append("Resolution very low (< 100px)")
    if scene.frame_end < scene.frame_start:
        errors.append("frame_end < frame_start!")
    
    # 输出报告
    print(f"\n{'='*50}")
    print(f"[Pre-Render Check]")
    print(f"  Errors: {len(errors)}")
    for e in errors:
        print(f"    [ERROR] {e}")
    print(f"  Warnings: {len(warnings)}")
    for w in warnings:
        print(f"    [WARN] {w}")
    print(f"{'='*50}\n")
    
    return len(errors) == 0
```

### 渲染异常捕获

```python
def safe_render(scene, frame_start, frame_end):
    """安全渲染（带异常捕获）"""
    import traceback
    
    scene.frame_start = frame_start
    scene.frame_end = frame_end
    
    try:
        # 渲染动画
        bpy.ops.render.render(animation=True)
        print(f"[Render] Success: frames {frame_start}-{frame_end}")
        return True
        
    except RuntimeError as e:
        error_msg = str(e)
        print(f"[Render ERROR] {error_msg}")
        
        # 分类处理
        if "out of memory" in error_msg.lower():
            print("[Recovery] Trying CPU render...")
            scene.cycles.device = 'CPU'
            try:
                bpy.ops.render.render(animation=True)
                return True
            except:
                pass
        
        elif "Cannot open" in error_msg:
            print("[Recovery] Check output path permissions")
        
        traceback.print_exc()
        return False
        
    except Exception as e:
        print(f"[Render FATAL] {type(e).__name__}: {e}")
        traceback.print_exc()
        return False
```

### 渲染黑屏诊断

```python
def diagnose_black_render():
    """诊断渲染黑屏"""
    from mathutils import Vector
    
    scene = bpy.context.scene
    print("[Diagnose] Checking for black render causes...")
    
    # 1. 相机位置和朝向
    cam = scene.camera
    if cam:
        cam_pos = cam.matrix_world.translation
        cam_dir = cam.matrix_world.to_quaternion() @ Vector((0, 0, -1))
        print(f"  Camera pos: {cam_pos}")
        print(f"  Camera dir: {cam_dir}")
        
        # 检查相机前方是否有物体
        # 射线检测
        depsgraph = bpy.context.evaluated_depsgraph_get()
        result, loc, normal, idx, obj, mat = scene.ray_cast(
            depsgraph, cam_pos, cam_dir)
        if result:
            print(f"  Camera hits: {obj.name} at {loc}")
        else:
            print("  [!] Camera ray hits nothing - object may be off-screen")
    
    # 2. 灯光能量
    for obj in bpy.data.objects:
        if obj.type == 'LIGHT':
            energy = obj.data.energy
            print(f"  Light '{obj.name}': energy={energy}")
            if energy < 1:
                print(f"    [!] Very low energy!")
    
    # 3. World背景
    if scene.world and scene.world.use_nodes:
        bg = scene.world.node_tree.nodes.get('Background')
        if bg:
            strength = bg.inputs['Strength'].default_value
            print(f"  World background strength: {strength}")
    
    # 4. 物体可见性
    for obj in bpy.data.objects:
        if obj.type == 'MESH':
            visible = not obj.hide_render
            print(f"  Mesh '{obj.name}': hide_render={obj.hide_render}")
```

### 日志级别控制

```python
import logging

def setup_render_logging(log_file="render.log"):
    """配置渲染日志"""
    logger = logging.getLogger("blender_render")
    logger.setLevel(logging.DEBUG)
    
    # 文件Handler
    fh = logging.FileHandler(log_file, encoding='utf-8')
    fh.setLevel(logging.DEBUG)
    
    # 控制台Handler
    ch = logging.StreamHandler()
    ch.setLevel(logging.INFO)
    
    formatter = logging.Formatter(
        '%(asctime)s [%(levelname)s] %(message)s')
    fh.setFormatter(formatter)
    ch.setFormatter(formatter)
    
    logger.addHandler(fh)
    logger.addHandler(ch)
    
    return logger
```

## 常见陷阱

### 陷阱1：Background模式无错误弹窗
```python
# background模式所有错误只输出到stdout/stderr
# 必须重定向输出到文件：
# blender --background --python script.py > log.txt 2>&1
```

### 陷阱2：渲染成功但文件为空
```python
# 检查磁盘空间
import shutil
total, used, free = shutil.disk_usage(output_dir)
if free < 100 * 1024 * 1024:  # < 100MB
    print("[ERROR] Disk space too low!")
```

### 陷阱3：EEVEE Next渲染崩溃
```python
# Blender 5.x EEVEE Next在background模式可能不稳定
# 回退方案：使用Cycles或旧版EEVEE
# 检查：bpy.app.version >= (4, 2, 0)
```

- Blender 5.1.0 Alpha: 建议Cycles作为fallback
