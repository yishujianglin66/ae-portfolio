# 渲染管线完整工作流

## 问题场景

将前述所有渲染步骤串联为完整的自动化管线：从场景准备到最终视频输出。需要处理各步骤的依赖关系、错误恢复、进度报告。

## 核心原理

### 管线架构

```
┌─────────────────────────────────────────────────────────┐
│                  渲染管线完整流程                         │
├─────────────────────────────────────────────────────────┤
│                                                         │
│  1. 场景准备                                            │
│     ├── FBX导入                                         │
│     ├── 骨骼创建                                        │
│     ├── 权重分配                                        │
│     └── 武器归位                                        │
│                                                         │
│  2. 渲染设置                                            │
│     ├── 相机环绕动画                                    │
│     ├── 灯光设置                                        │
│     ├── 赛璐璐Shader                                   │
│     ├── LineArt描边                                    │
│     └── 输出配置                                        │
│                                                         │
│  3. 渲染执行                                            │
│     ├── 预检查                                          │
│     ├── 逐帧渲染                                        │
│     └── 进度报告                                        │
│                                                         │
│  4. 后处理                                              │
│     ├── 帧质量验证                                      │
│     ├── FFmpeg合成                                      │
│     └── 输出到成品库                                    │
│                                                         │
└─────────────────────────────────────────────────────────┘
```

### 完整管线脚本

```python
import bpy
import os
import sys
import time
import subprocess
import traceback

class CelRenderPipeline:
    """三渲二渲染管线"""
    
    def __init__(self, fbx_path, output_dir, config=None):
        self.fbx_path = fbx_path
        self.output_dir = output_dir
        self.frames_dir = os.path.join(output_dir, "frames")
        self.config = config or self.default_config()
        self.logger = self._setup_logger()
        
    def default_config(self):
        return {
            "resolution": (1920, 1080),
            "fps": 30,
            "frame_count": 60,
            "camera_distance": 5.0,
            "camera_height": 1.5,
            "samples": 32,
            "lineart_thickness": 2.0,
        }
    
    def _setup_logger(self):
        import logging
        logger = logging.getLogger("pipeline")
        logger.setLevel(logging.INFO)
        handler = logging.StreamHandler()
        handler.setFormatter(logging.Formatter(
            '[%(asctime)s] %(message)s', '%H:%M:%S'))
        logger.addHandler(handler)
        return logger
    
    def run(self):
        """执行完整管线"""
        start_time = time.time()
        self.logger.info("=" * 60)
        self.logger.info("Cel Render Pipeline Start")
        self.logger.info(f"  FBX: {self.fbx_path}")
        self.logger.info(f"  Output: {self.output_dir}")
        self.logger.info("=" * 60)
        
        try:
            # Phase 1: 场景准备
            self._phase1_scene_setup()
            
            # Phase 2: 渲染设置
            self._phase2_render_setup()
            
            # Phase 3: 渲染执行
            self._phase3_render()
            
            # Phase 4: 后处理
            self._phase4_post_process()
            
            elapsed = time.time() - start_time
            self.logger.info(f"Pipeline complete in {elapsed:.1f}s")
            return True
            
        except Exception as e:
            self.logger.error(f"Pipeline failed: {e}")
            traceback.print_exc()
            return False
    
    def _phase1_scene_setup(self):
        """Phase 1: 场景准备"""
        self.logger.info("[Phase 1] Scene Setup")
        
        # 清空场景
        bpy.ops.object.select_all(action='SELECT')
        bpy.ops.object.delete()
        
        # 导入FBX
        self.logger.info("  Importing FBX...")
        bpy.ops.import_scene.fbx(
            filepath=self.fbx_path,
            use_anim=False,
            ignore_leaf_bones=True,
        )
        
        # 获取导入的对象
        meshes = [o for o in bpy.data.objects if o.type == 'MESH']
        self.logger.info(f"  Imported {len(meshes)} mesh objects")
        
        # 创建骨骼（如果需要）
        # ... 调用骨骼创建模块
        
        # 权重分配
        # ... 调用权重模块
        
        # 武器归位
        # ... 调用武器归位模块
    
    def _phase2_render_setup(self):
        """Phase 2: 渲染设置"""
        self.logger.info("[Phase 2] Render Setup")
        scene = bpy.context.scene
        
        # 分辨率
        scene.render.resolution_x = self.config["resolution"][0]
        scene.render.resolution_y = self.config["resolution"][1]
        scene.render.resolution_percentage = 100
        
        # 帧范围
        scene.frame_start = 1
        scene.frame_end = self.config["frame_count"]
        scene.render.fps = self.config["fps"]
        
        # EEVEE设置
        scene.render.engine = 'BLENDER_EEVEE_NEXT'
        scene.eevee.taa_render_samples = self.config["samples"]
        
        # 透明背景
        scene.render.film_transparent = True
        
        # 输出路径
        os.makedirs(self.frames_dir, exist_ok=True)
        scene.render.image_settings.file_format = 'PNG'
        scene.render.image_settings.color_mode = 'RGBA'
        scene.render.filepath = os.path.join(self.frames_dir, "frame_")
        
        # 相机设置
        self._setup_camera()
        
        # 灯光设置
        self._setup_lighting()
        
        # 材质（赛璐璐）
        self._setup_cel_materials()
        
        self.logger.info("  Render setup complete")
    
    def _setup_camera(self):
        """设置环绕相机"""
        import math
        
        scene = bpy.context.scene
        dist = self.config["camera_distance"]
        height = self.config["camera_height"]
        
        # 创建相机
        cam_data = bpy.data.cameras.new("RenderCam")
        cam_obj = bpy.data.objects.new("RenderCam", cam_data)
        bpy.context.collection.objects.link(cam_obj)
        scene.camera = cam_obj
        
        # 环绕动画
        for frame in range(1, self.config["frame_count"] + 1):
            t = (frame - 1) / self.config["frame_count"]
            angle = t * 2 * math.pi
            
            x = dist * math.cos(angle)
            y = dist * math.sin(angle)
            
            cam_obj.location = (x, y, height)
            
            # 看向中心
            direction = Vector((0, 0, height * 0.5)) - cam_obj.location
            rot_quat = direction.to_track_quat('-Z', 'Y')
            cam_obj.rotation_euler = rot_quat.to_euler()
            
            cam_obj.keyframe_insert("location", frame=frame)
            cam_obj.keyframe_insert("rotation_euler", frame=frame)
    
    def _setup_lighting(self):
        """设置灯光"""
        light = bpy.data.lights.new("MainLight", 'AREA')
        light.energy = 300
        light.size = 10.0
        light_obj = bpy.data.objects.new("MainLight", light)
        bpy.context.collection.objects.link(light_obj)
        light_obj.location = (3, -5, 8)
    
    def _setup_cel_materials(self):
        """设置赛璐璐材质"""
        for obj in bpy.data.objects:
            if obj.type == 'MESH':
                for mat_slot in obj.material_slots:
                    mat = mat_slot.material
                    if mat and mat.use_nodes:
                        # 转换为Toon BSDF
                        self._convert_to_toon(mat)
    
    def _convert_to_toon(self, mat):
        """将材质转换为Toon"""
        tree = mat.node_tree
        # 找到Principled BSDF
        principled = None
        for node in tree.nodes:
            if node.type == 'BSDF_PRINCIPLED':
                principled = node
                break
        
        if principled:
            # 获取Base Color
            base_color = principled.inputs['Base Color'].default_value[:]
            
            # 创建Toon BSDF
            toon = tree.nodes.new('ShaderNodeBsdfToon')
            toon.inputs['Base Color'].default_value = base_color
            toon.location = principled.location
            
            # 连接到输出
            output = None
            for node in tree.nodes:
                if node.type == 'OUTPUT_MATERIAL':
                    output = node
            
            if output:
                tree.links.new(toon.outputs['BSDF'],
                              output.inputs['Surface'])
    
    def _phase3_render(self):
        """Phase 3: 渲染执行"""
        self.logger.info("[Phase 3] Rendering")
        
        scene = bpy.context.scene
        total = scene.frame_end - scene.frame_start + 1
        
        # 预检查
        if not self._pre_render_check():
            raise RuntimeError("Pre-render check failed")
        
        # 渲染
        render_start = time.time()
        bpy.ops.render.render(animation=True)
        render_time = time.time() - render_start
        
        self.logger.info(
            f"  Rendered {total} frames in {render_time:.1f}s "
            f"({render_time/total:.1f}s/frame)")
    
    def _pre_render_check(self):
        """渲染前检查"""
        scene = bpy.context.scene
        
        if not scene.camera:
            self.logger.error("  No camera!")
            return False
        
        output_dir = os.path.dirname(scene.render.filepath)
        if not os.path.exists(output_dir):
            os.makedirs(output_dir, exist_ok=True)
        
        return True
    
    def _phase4_post_process(self):
        """Phase 4: 后处理"""
        self.logger.info("[Phase 4] Post-processing")
        
        # 验证帧
        import glob
        frames = glob.glob(os.path.join(self.frames_dir, "frame_*.png"))
        self.logger.info(f"  {len(frames)} frames rendered")
        
        if len(frames) == 0:
            raise RuntimeError("No frames rendered!")
        
        # FFmpeg合成
        output_video = os.path.join(self.output_dir, "output.mp4")
        self._ffmpeg_composite(output_video)
        
        self.logger.info(f"  Output: {output_video}")
    
    def _ffmpeg_composite(self, output_path):
        """FFmpeg合成视频"""
        frame_pattern = os.path.join(self.frames_dir, "frame_%04d.png")
        
        cmd = [
            "ffmpeg", "-y",
            "-framerate", str(self.config["fps"]),
            "-i", frame_pattern,
            "-c:v", "libx264",
            "-pix_fmt", "yuv420p",
            "-crf", "18",
            output_path
        ]
        
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0:
            raise RuntimeError(f"FFmpeg failed: {result.stderr}")


# 入口
if __name__ == "__main__":
    # 从命令行参数获取
    argv = sys.argv
    if "--" in argv:
        args = argv[argv.index("--") + 1:]
    else:
        args = []
    
    fbx_path = args[0] if len(args) > 0 else "model.fbx"
    output_dir = args[1] if len(args) > 1 else "./output"
    
    pipeline = CelRenderPipeline(fbx_path, output_dir)
    success = pipeline.run()
    sys.exit(0 if success else 1)
```

### 进度报告

```python
import bpy

class RenderProgressHandler(bpy.types.RenderStats):
    """渲染进度回调"""
    pass

def setup_progress_callback():
    """设置进度回调（通过depsgraph更新）"""
    scene = bpy.context.scene
    
    # 方法1：通过handler
    @bpy.app.handlers.persistent
    def frame_handler(scene):
        frame = scene.frame_current
        total = scene.frame_end - scene.frame_start + 1
        progress = (frame - scene.frame_start) / total * 100
        print(f"[Progress] Frame {frame}/{scene.frame_end} ({progress:.0f}%)")
    
    bpy.app.handlers.frame_change_post.append(frame_handler)
```

### 错误恢复策略

```python
def render_with_recovery(scene, max_retries=3):
    """带重试的渲染"""
    for attempt in range(max_retries):
        try:
            bpy.ops.render.render(animation=True)
            return True
        except RuntimeError as e:
            print(f"[Retry {attempt+1}/{max_retries}] {e}")
            
            # 根据错误类型调整策略
            if "memory" in str(e).lower():
                # 降低质量重试
                scene.eevee.taa_render_samples //= 2
                print(f"  Reduced samples to {scene.eevee.taa_render_samples}")
            else:
                # 其他错误直接失败
                return False
    
    return False
```
