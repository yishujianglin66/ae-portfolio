# 渲染质量检查清单

## 问题场景

渲染完成后需要验证输出质量：帧完整性、画面内容、色彩正确性、描边效果等。自动化检查可以在FFmpeg合成前拦截问题帧。

## 核心原理

### 帧完整性检查

```python
import os
import glob
from PIL import Image
import numpy as np

def check_frame_integrity(frames_dir, expected_count, resolution=(1920, 1080)):
    """帧完整性检查"""
    frames = sorted(glob.glob(os.path.join(frames_dir, "frame_*.png")))
    issues = []
    
    # 1. 数量检查
    if len(frames) != expected_count:
        issues.append(f"Frame count: {len(frames)}/{expected_count}")
    
    # 2. 逐帧检查
    for i, fpath in enumerate(frames):
        try:
            img = Image.open(fpath)
            
            # 分辨率检查
            if img.size != resolution:
                issues.append(
                    f"Frame {i+1}: resolution {img.size} != {resolution}")
            
            # 空帧检查（全黑/全透明）
            arr = np.array(img)
            if arr.max() == 0:
                issues.append(f"Frame {i+1}: completely black/transparent")
            
            # 文件大小异常
            fsize = os.path.getsize(fpath)
            if fsize < 5000:  # < 5KB可能是空帧
                issues.append(f"Frame {i+1}: suspiciously small ({fsize}B)")
                
        except Exception as e:
            issues.append(f"Frame {i+1}: cannot open ({e})")
    
    return issues
```

### 画面内容验证

```python
def check_frame_content(frame_path, min_coverage=0.05):
    """
    检查帧中是否有角色内容
    min_coverage: 最小非透明像素占比
    """
    img = Image.open(frame_path).convert('RGBA')
    arr = np.array(img)
    
    # Alpha通道分析
    alpha = arr[:, :, 3]
    non_transparent = np.count_nonzero(alpha > 10)
    total_pixels = alpha.size
    coverage = non_transparent / total_pixels
    
    if coverage < min_coverage:
        return False, f"Coverage too low: {coverage:.2%}"
    
    # 颜色分布检查（避免单色帧）
    rgb = arr[:, :, :3]
    visible_mask = alpha > 10
    if visible_mask.any():
        visible_rgb = rgb[visible_mask]
        color_std = visible_rgb.std()
        if color_std < 5:  # 几乎单色
            return False, f"Near-monochrome frame (std={color_std:.1f})"
    
    return True, f"Coverage: {coverage:.2%}"
```

### 描边效果验证

```python
def check_lineart_quality(frame_path):
    """验证描边是否生效"""
    img = Image.open(frame_path).convert('RGB')
    arr = np.array(img).astype(float)
    
    # 边缘检测（Sobel算子简化版）
    gray = arr.mean(axis=2)
    
    # 水平梯度
    gx = np.abs(np.diff(gray, axis=1))
    # 垂直梯度
    gy = np.abs(np.diff(gray, axis=0))
    
    # 强边缘像素占比
    edge_threshold = 50  # 梯度阈值
    strong_edges_x = np.count_nonzero(gx > edge_threshold)
    strong_edges_y = np.count_nonzero(gy > edge_threshold)
    
    total = gray.size
    edge_ratio = (strong_edges_x + strong_edges_y) / (total * 2)
    
    # 三渲二描边通常占1-5%像素
    if edge_ratio < 0.005:
        return False, "No lineart detected (edge ratio too low)"
    elif edge_ratio > 0.15:
        return False, "Too many edges (lineart may be too dense)"
    
    return True, f"Edge ratio: {edge_ratio:.3f}"
```

### 帧间一致性检查

```python
def check_temporal_consistency(frames_dir, max_diff=0.3):
    """
    检查相邻帧之间的变化是否平滑
    max_diff: 允许的最大帧间差异（0-1）
    """
    frames = sorted(glob.glob(os.path.join(frames_dir, "frame_*.png")))
    issues = []
    
    prev_arr = None
    for i, fpath in enumerate(frames):
        img = Image.open(fpath).convert('RGB')
        arr = np.array(img).astype(float) / 255.0
        
        if prev_arr is not None:
            # 计算帧间差异
            diff = np.abs(arr - prev_arr).mean()
            
            if diff > max_diff:
                issues.append(
                    f"Frame {i}→{i+1}: large jump (diff={diff:.3f})")
        
        prev_arr = arr
    
    return issues
```

### 综合质量报告

```python
def generate_quality_report(frames_dir, output_report):
    """生成渲染质量报告"""
    frames = sorted(glob.glob(os.path.join(frames_dir, "frame_*.png")))
    
    report = {
        "total_frames": len(frames),
        "issues": [],
        "stats": {
            "avg_coverage": 0,
            "avg_edge_ratio": 0,
            "min_file_size": float('inf'),
            "max_file_size": 0,
        }
    }
    
    coverages = []
    for i, fpath in enumerate(frames):
        # 内容检查
        ok, msg = check_frame_content(fpath)
        if not ok:
            report["issues"].append(f"Frame {i+1}: {msg}")
        else:
            # 提取coverage数值
            cov = float(msg.split(": ")[1].strip('%')) / 100
            coverages.append(cov)
        
        # 文件大小
        fsize = os.path.getsize(fpath)
        report["stats"]["min_file_size"] = min(
            report["stats"]["min_file_size"], fsize)
        report["stats"]["max_file_size"] = max(
            report["stats"]["max_file_size"], fsize)
    
    if coverages:
        report["stats"]["avg_coverage"] = sum(coverages) / len(coverages)
    
    # 写入报告
    import json
    with open(output_report, 'w', encoding='utf-8') as f:
        json.dump(report, f, indent=2, ensure_ascii=False)
    
    print(f"[Quality] Report: {output_report}")
    print(f"  Frames: {report['total_frames']}")
    print(f"  Issues: {len(report['issues'])}")
    
    return report
```

## 常见陷阱

### 陷阱1：首帧/末帧异常
```python
# 相机动画的首末帧可能有跳变
# 解决：单独检查第1帧和最后1帧
first_ok, _ = check_frame_content(frames[0])
last_ok, _ = check_frame_content(frames[-1])
```

### 陷阱2：Alpha通道丢失
```python
# 某些渲染设置不输出Alpha
# 检查：scene.render.image_settings.color_mode == 'RGBA'
# 且：scene.render.film_transparent = True
```
