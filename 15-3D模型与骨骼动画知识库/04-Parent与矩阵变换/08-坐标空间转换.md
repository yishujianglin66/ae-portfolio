# 坐标空间转换（世界/局部/骨骼）

## 问题场景

在不同坐标空间之间转换点是3D编程的基础操作。常见需求：将世界坐标点转为物体局部坐标、将骨骼局部坐标转为世界坐标、计算两点在不同空间下的距离。

## 核心原理

### 空间转换公式

```python
from mathutils import Vector, Matrix

# 世界 → 物体局部
def world_to_local(obj, world_point):
    return obj.matrix_world.inverted() @ world_point

# 物体局部 → 世界
def local_to_world(obj, local_point):
    return obj.matrix_world @ local_point

# 世界 → 骨骼局部
def world_to_bone(armature_obj, bone_name, world_point):
    pb = armature_obj.pose.bones[bone_name]
    bone_world = armature_obj.matrix_world @ pb.matrix
    return bone_world.inverted() @ world_point

# 骨骼局部 → 世界
def bone_to_world(armature_obj, bone_name, bone_point):
    pb = armature_obj.pose.bones[bone_name]
    bone_world = armature_obj.matrix_world @ pb.matrix
    return bone_world @ bone_point

# 物体A局部 → 物体B局部（通过世界中转）
def local_a_to_local_b(obj_a, obj_b, point_in_a):
    world_point = obj_a.matrix_world @ point_in_a
    return obj_b.matrix_world.inverted() @ world_point
```

### 方向/向量转换（不含平移）

```python
# 转换方向向量时，只用旋转部分（3x3矩阵）
def transform_direction(obj, local_dir):
    """将局部方向转为世界方向（不受平移影响）"""
    return (obj.matrix_world.to_3x3() @ local_dir).normalized()

def transform_normal(obj, local_normal):
    """转换法线（需要用逆转置矩阵）"""
    # 非均匀缩放时，法线不能用普通旋转矩阵
    normal_matrix = obj.matrix_world.inverted().transposed().to_3x3()
    return (normal_matrix @ local_normal).normalized()
```

### 本项目的坐标转换应用

```python
# 扇子握点对齐（cel_shading.py L1725-1735）

# 1. 扇柄局部握点 → 世界坐标
grip_local = Vector((grip_x, grip_y, grip_z))
grip_world = _fan_handle.matrix_world @ grip_local

# 2. 手部世界坐标（从网格顶点检测得到）
hand_world = Vector((-0.6, 0.7, -2.72))

# 3. 计算世界空间偏移
shift = hand_world - grip_world

# 4. 应用偏移到扇柄（世界空间平移）
_fan_handle.location += shift
```

### 获取物体特定局部点的世界坐标

```python
def get_local_point_world(obj, local_point):
    """获取物体局部空间中某点的世界坐标"""
    bpy.context.view_layer.update()
    return obj.matrix_world @ Vector(local_point)

def get_origin_world(obj):
    """获取物体原点的世界坐标"""
    bpy.context.view_layer.update()
    return obj.matrix_world.translation.copy()

def get_bound_box_center_world(obj):
    """获取物体包围盒中心的世界坐标"""
    bpy.context.view_layer.update()
    corners = [obj.matrix_world @ Vector(c) for c in obj.bound_box]
    center = sum(corners, Vector()) / 8
    return center
```

## 常见陷阱

### 陷阱1：用4x4矩阵转换3D向量
```python
# 错误：4x4矩阵 × 3D向量 → 维度不匹配
result = obj.matrix_world @ Vector((1, 2, 3))  # ← 错误！

# 正确：使用4D向量（w=1为点，w=0为方向）
point_world = obj.matrix_world @ Vector((1, 2, 3, 1))  # 点
dir_world = obj.matrix_world.to_3x3() @ Vector((1, 2, 3))  # 方向
```

### 陷阱2：忘记update导致矩阵过期
```python
obj.location = (5, 0, 0)
# matrix_world还是旧值！
world_pos = obj.matrix_world @ Vector((0,0,0,1))  # ← 错误

# 正确：
obj.location = (5, 0, 0)
bpy.context.view_layer.update()
world_pos = obj.matrix_world @ Vector((0,0,0,1))  # ← 正确
```
