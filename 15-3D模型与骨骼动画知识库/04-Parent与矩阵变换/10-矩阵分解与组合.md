# 矩阵分解与组合（Decompose/Compose）

## 问题场景

有时需要从matrix_world中提取位移/旋转/缩放分量，或从分量重新构建矩阵。Blender的mathutils提供了decompose()和compose()方法。

## 核心原理

### 矩阵分解

```python
from mathutils import Matrix, Vector, Quaternion

def decompose_matrix(obj):
    """从matrix_world提取TRS分量"""
    loc, rot, scale = obj.matrix_world.decompose()
    
    print(f"Location: {loc.to_tuple()}")
    print(f"Rotation (quat): {rot.to_tuple()}")  # (w, x, y, z)
    print(f"Scale: {scale.to_tuple()}")
    
    # 四元数转欧拉角
    euler = rot.to_euler()
    print(f"Rotation (euler): {euler.to_tuple()}")  # (rx, ry, rz) 弧度
    
    return loc, rot, scale
```

### 矩阵组合

```python
from mathutils import Matrix, Vector, Quaternion, Euler
import math

def compose_matrix(location, rotation_euler, scale):
    """从TRS分量构建4x4矩阵"""
    loc = Vector(location)
    rot = Euler(rotation_euler).to_matrix().to_4x4()
    scl = Matrix.Diagonal(Vector(scale).to_4d())
    
    # TRS组合：T × R × S
    return Matrix.Translation(loc) @ rot @ scl

# 使用示例
M = compose_matrix(
    location=(1, 2, 3),
    rotation_euler=(0, 0, math.radians(45)),
    scale=(1, 1, 1)
)
```

### 仅提取旋转（忽略缩放）

```python
def get_rotation_only(obj):
    """提取纯旋转矩阵（去除缩放影响）"""
    # 方法1：从decompose
    _, rot_quat, _ = obj.matrix_world.decompose()
    return rot_quat.to_matrix().to_4x4()
    
    # 方法2：正交化3x3部分
    mat3 = obj.matrix_world.to_3x3()
    mat3.normalize()  # 去除缩放
    return mat3.to_4x4()
```

### 插值变换（动画用）

```python
def lerp_transform(obj_a, obj_b, t):
    """在两个物体的变换之间插值 (t: 0~1)"""
    loc_a, rot_a, scl_a = obj_a.matrix_world.decompose()
    loc_b, rot_b, scl_b = obj_b.matrix_world.decompose()
    
    # 位移：线性插值
    loc = loc_a.lerp(loc_b, t)
    
    # 旋转：球面插值（slerp）
    rot = rot_a.slerp(rot_b, t)
    
    # 缩放：线性插值
    scl = scl_a.lerp(scl_b, t)
    
    # 组合
    return Matrix.LocRotScale(loc, rot, scl)

# 应用插值结果
result_matrix = lerp_transform(obj_start, obj_end, 0.5)
obj_current.matrix_world = result_matrix
```

## 常见陷阱

### 陷阱1：非均匀缩放时旋转提取错误
```python
# 非均匀缩放(1,2,1)会"污染"旋转矩阵
# decompose()能正确处理，但to_3x3()不能
# 推荐：始终用decompose()
```

### 陷阱2：四元数插值方向
```python
# slerp可能走"长路"（>180°）
# 检查点积，如果为负则翻转一个四元数
if rot_a.dot(rot_b) < 0:
    rot_b.negate()
result = rot_a.slerp(rot_b, t)
```
