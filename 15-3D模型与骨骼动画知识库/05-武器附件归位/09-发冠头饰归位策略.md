# 发冠/头饰归位策略

## 问题场景

发冠（Object001）需要固定在头顶，跟随Head骨骼运动。与武器不同，发冠不需要复杂的角度调整，但需要精确的parent到BONE TAIL。

## 核心原理

### 发冠归位流程

```python
import bpy
from mathutils import Matrix, Vector

def position_crown(crown_obj, armature_obj, head_bone_name='Head'):
    """
    发冠归位：parent到Head骨骼TAIL
    保持发冠FBX原始位置（假设已在头顶）
    """
    bpy.context.view_layer.update()
    
    # 1. 获取Head骨骼的pose矩阵
    pb = armature_obj.pose.bones[head_bone_name]
    bone_world = armature_obj.matrix_world @ pb.matrix
    
    # 2. 计算TAIL世界矩阵
    bone_length = pb.bone.length
    tail_matrix = bone_world @ Matrix.Translation(Vector((0, bone_length, 0)))
    
    # 3. 记录发冠当前世界矩阵
    crown_world = crown_obj.matrix_world.copy()
    
    # 4. 设置parent
    crown_obj.parent = armature_obj
    crown_obj.parent_type = 'BONE'
    crown_obj.parent_bone = head_bone_name
    
    # 5. 计算matrix_parent_inverse（保持原位）
    crown_obj.matrix_parent_inverse = tail_matrix.inverted() @ crown_world
    
    bpy.context.view_layer.update()
    
    # 6. 验证
    new_pos = crown_obj.matrix_world.translation
    old_pos = crown_world.translation
    diff = (new_pos - old_pos).length
    print(f"[Crown] Position diff after parent: {diff:.6f}")
    print(f"[Crown] World pos: ({new_pos.x:.2f}, {new_pos.y:.2f}, {new_pos.z:.2f})")
    
    return crown_obj
```

### 发冠位置验证

```python
def validate_crown_position(crown_obj, armature_obj):
    """验证发冠是否在头顶"""
    bpy.context.view_layer.update()
    
    crown_pos = crown_obj.matrix_world.translation
    
    # Head骨骼世界位置
    pb = armature_obj.pose.bones['Head']
    head_world = armature_obj.matrix_world @ pb.matrix
    head_pos = head_world.translation
    
    # 发冠应该在Head上方
    z_diff = crown_pos.z - head_pos.z
    
    if z_diff > 0:
        print(f"[OK] Crown is {z_diff:.2f} above Head")
    else:
        print(f"[WARN] Crown is {abs(z_diff):.2f} BELOW Head!")
    
    # 水平偏移
    xy_diff = ((crown_pos.x - head_pos.x)**2 + (crown_pos.y - head_pos.y)**2) ** 0.5
    print(f"[Crown] Horizontal offset from Head: {xy_diff:.2f}")
```

# 如果Head bone length=0.3，TAIL只在head上方0.3
