# 包围盒与特征点计算

## 问题场景

武器归位需要找到"握点"（handle的-X端）和"连接点"（blade的-X端）。这些特征点通过包围盒(bound_box)计算。理解bound_box的局部/世界坐标区别至关重要。

## 核心原理

### bound_box基础

```python
import bpy
from mathutils import Vector

def get_bbox_info(obj):
    """获取物体包围盒信息"""
    # bound_box是局部坐标！（8个角点）
    local_bbox = [Vector(v) for v in obj.bound_box]
    
    # 转世界坐标
    world_bbox = [obj.matrix_world @ Vector(v) for v in obj.bound_box]
    
    # 计算尺寸
    x_min = min(v.x for v in world_bbox)
    x_max = max(v.x for v in world_bbox)
    y_min = min(v.y for v in world_bbox)
    y_max = max(v.y for v in world_bbox)
    z_min = min(v.z for v in world_bbox)
    z_max = max(v.z for v in world_bbox)
    
    return {
        'x_range': (x_min, x_max),
        'y_range': (y_min, y_max),
        'z_range': (z_min, z_max),
        'size': (x_max-x_min, y_max-y_min, z_max-z_min),
        'center': Vector((
            (x_min+x_max)/2,
            (y_min+y_max)/2,
            (z_min+z_max)/2
        ))
    }
```

### 特征点计算

```python
def get_grip_point(handle_obj):
    """计算扇柄握点（-X端中心）"""
    # 局部坐标
    local_bbox = [Vector(v) for v in handle_obj.bound_box]
    
    grip_local = Vector((
        min(v.x for v in local_bbox),  # -X端
        sum(v.y for v in local_bbox) / 8,  # Y中心
        sum(v.z for v in local_bbox) / 8   # Z中心
    ))
    
    # 转世界坐标
    grip_world = handle_obj.matrix_world @ grip_local
    return grip_local, grip_world

def get_connection_point(blade_obj):
    """计算扇面连接点（-X端中心）"""
    local_bbox = [Vector(v) for v in blade_obj.bound_box]
    
    connect_local = Vector((
        min(v.x for v in local_bbox),
        sum(v.y for v in local_bbox) / 8,
        sum(v.z for v in local_bbox) / 8
    ))
    
    connect_world = blade_obj.matrix_world @ connect_local
    return connect_local, connect_world

def get_tip_point(handle_obj):
    """计算扇柄tip（+X端中心）"""
    # 注意：tip用世界坐标计算（因为handle已旋转）
    world_bbox = [handle_obj.matrix_world @ Vector(v) for v in handle_obj.bound_box]
    
    tip_world = Vector((
        max(v.x for v in world_bbox),  # +X端
        sum(v.y for v in world_bbox) / 8,
        sum(v.z for v in world_bbox) / 8
    ))
    return tip_world
```

### 注意事项

```python
# bound_box的8个角点顺序（局部坐标）：
# 0: (-x, -y, -z)  1: (-x, -y, +z)
# 2: (-x, +y, +z)  3: (-x, +y, -z)
# 4: (+x, -y, -z)  5: (+x, -y, +z)
# 6: (+x, +y, +z)  7: (+x, +y, -z)

# 重要：bound_box是AXIS-ALIGNED的（轴对齐）
# 旋转物体后，bound_box不会旋转！
# 它始终是世界轴对齐的包围盒

# 如果需要旋转后的精确包围盒：
def get_rotated_bbox(obj):
    """获取考虑旋转的包围盒（用mesh顶点）"""
    world_matrix = obj.matrix_world
    verts = [world_matrix @ v.co for v in obj.data.vertices]
    
    x_min = min(v.x for v in verts)
    x_max = max(v.x for v in verts)
    # ... 同理y, z
    return (x_min, x_max)
```

## 常见陷阱

### 陷阱1：bound_box是局部坐标
```python
# 错误：直接用bound_box作为世界坐标
grip = Vector(obj.bound_box[0])  # ← 局部坐标！

# 正确：转换到世界
grip_world = obj.matrix_world @ Vector(obj.bound_box[0])
```

### 陷阱2：旋转后bound_box不更新
```python
# bound_box是对象空间的AABB
# 旋转物体后，bound_box的局部值不变
# 但matrix_world变了，所以世界坐标会变
```
