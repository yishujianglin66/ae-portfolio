# 可视化调试（生成调试网格）

## 问题场景

纯文本日志难以直观理解3D空间中的问题（如骨骼位置、权重分布、物体偏移）。通过生成调试网格、标记点、辅助线，可以渲染出可视化的调试图像。

## 核心原理

### 调试标记点

```python
import bpy
from mathutils import Vector

def create_point_marker(name, location, size=0.05, 
                        color=(1, 0, 0, 1)):
    """创建点标记（小球）"""
    # 创建UV球
    mesh = bpy.data.meshes.new(f"marker_{name}")
    obj = bpy.data.objects.new(f"DEBUG_{name}", mesh)
    bpy.context.collection.objects.link(obj)
    
    # 使用内置球体
    import bmesh
    bm = bmesh.new()
    bmesh.ops.create_uvsphere(bm, u_segments=8, v_segments=6, 
                               radius=size)
    bm.to_mesh(mesh)
    bm.free()
    
    obj.location = location
    
    # 材质
    mat = bpy.data.materials.new(f"marker_mat_{name}")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get('Principled BSDF')
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        bsdf.inputs['Emission Color'].default_value = color
        bsdf.inputs['Emission Strength'].default_value = 5.0
    obj.data.materials.append(mat)
    
    return obj
```

```python
def create_line_marker(name, start, end, color=(1, 1, 0, 1), 
                       thickness=0.01):
    """创建线段标记"""
    # 使用曲线
    curve = bpy.data.curves.new(f"line_{name}", 'CURVE')
    curve.dimensions = '3D'
    curve.bevel_depth = thickness  # 线段粗细
    
    spline = curve.splines.new('POLY')
    spline.points.add(1)
    spline.points[0].co = (*start, 1)
    spline.points[1].co = (*end, 1)
    
    obj = bpy.data.objects.new(f"DEBUG_LINE_{name}", curve)
    bpy.context.collection.objects.link(obj)
    
    # 材质
    mat = bpy.data.materials.new(f"line_mat_{name}")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get('Principled BSDF')
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        bsdf.inputs['Emission Color'].default_value = color
        bsdf.inputs['Emission Strength'].default_value = 3.0
    obj.data.materials.append(mat)
    
    return obj
```

### 骨骼可视化

```python
def visualize_armature(arm_obj, show_names=True):
    """可视化骨骼结构"""
    armature = arm_obj.data
    
    for bone in armature.bones:
        # 骨骼主线（head -> tail）
        create_line_marker(
            f"skeleton_{bone.name}",
            arm_obj.matrix_world @ bone.head_local,
            arm_obj.matrix_world @ bone.tail_local,
            color=(0, 1, 0, 1),
            thickness=0.02
        )
        
        # 关节点
        create_point_marker(
            f"joint_{bone.name}",
            arm_obj.matrix_world @ bone.head_local,
            size=0.03,
            color=(1, 1, 0, 1)
        )
        
        # 连接到父骨骼
        if bone.parent:
            create_line_marker(
                f"connect_{bone.name}",
                arm_obj.matrix_world @ bone.parent.tail_local,
                arm_obj.matrix_world @ bone.head_local,
                color=(0.5, 0.5, 0.5, 1),
                thickness=0.005
            )

# 使用
visualize_armature(armature_obj)
```

### 权重可视化

```python
def visualize_weights(obj, bone_name, output_path):
    """渲染权重分布图"""
    # 创建权重可视化材质
    mat = bpy.data.materials.new(f"weight_viz_{bone_name}")
    mat.use_nodes = True
    tree = mat.node_tree
    nodes = tree.nodes
    links = tree.links
    nodes.clear()
    
    # Vertex Color/Weight节点
    # Blender 4.0+使用Color Attribute
    attr_node = nodes.new('ShaderNodeVertexColor')
    attr_node.layer_name = bone_name  # 顶点组名
    attr_node.location = (-200, 0)
    
    # ColorRamp（蓝->绿->红）
    ramp = nodes.new('ShaderNodeValToRGB')
    ramp.color_ramp.elements[0].color = (0, 0, 1, 1)  # 蓝（无权重）
    ramp.color_ramp.elements[1].color = (1, 0, 0, 1)  # 红（全权重）
    ramp.location = (0, 0)
    
    # Emission（自发光，不受灯光影响）
    emission = nodes.new('ShaderNodeEmission')
    emission.location = (200, 0)
    
    # Output
    output = nodes.new('ShaderNodeOutputMaterial')
    output.location = (400, 0)
    
    # 连接
    links.new(attr_node.outputs['Color'], ramp.inputs['Fac'])
    links.new(ramp.outputs['Color'], emission.inputs['Color'])
    links.new(emission.outputs['Emission'], output.inputs['Surface'])
    
    # 临时替换材质
    original_mats = list(obj.data.materials)
    obj.data.materials.clear()
    obj.data.materials.append(mat)
    
    # 渲染
    scene = bpy.context.scene
    orig_path = scene.render.filepath
    scene.render.filepath = output_path
    bpy.ops.render.render(write_still=True)
    scene.render.filepath = orig_path
    
    # 恢复材质
    obj.data.materials.clear()
    for m in original_mats:
        obj.data.materials.append(m)
    
    print(f"[DEBUG] Weight visualization: {output_path}")
```

### 包围盒可视化

```python
def visualize_bounds(obj, color=(1, 1, 0, 0.3)):
    """可视化物体包围盒"""
    # 获取世界坐标包围盒
    bbox_corners = [obj.matrix_world @ Vector(c) 
                    for c in obj.bound_box]
    
    # 创建线框
    edges = [
        (0,1), (1,2), (2,3), (3,0),  # 底面
        (4,5), (5,6), (6,7), (7,4),  # 顶面
        (0,4), (1,5), (2,6), (3,7),  # 竖边
    ]
    
    mesh = bpy.data.meshes.new(f"bbox_{obj.name}")
    mesh.from_pydata(
        [c[:] for c in bbox_corners],
        edges,
        []
    )
    
    bbox_obj = bpy.data.objects.new(f"DEBUG_BBOX_{obj.name}", mesh)
    bpy.context.collection.objects.link(bbox_obj)
    
    # 线框材质
    mat = bpy.data.materials.new("bbox_mat")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get('Principled BSDF')
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        bsdf.inputs['Alpha'].default_value = color[3]
    mat.blend_method = 'BLEND'
    bbox_obj.data.materials.append(mat)
    
    return bbox_obj
```

### 调试渲染合成

```python
def render_debug_composite(output_path, show_skeleton=True, 
                           show_bounds=True, show_markers=True):
    """渲染带调试信息的合成图"""
    debug_objects = []
    
    # 收集调试对象
    for obj in bpy.data.objects:
        if obj.name.startswith("DEBUG_"):
            obj.hide_render = False  # 确保可见
            debug_objects.append(obj)
    
    # 渲染
    scene = bpy.context.scene
    orig_path = scene.render.filepath
    scene.render.filepath = output_path
    bpy.ops.render.render(write_still=True)
    scene.render.filepath = orig_path
    
    # 恢复隐藏
    for obj in debug_objects:
        obj.hide_render = True
    
    print(f"[DEBUG] Composite: {output_path}")

# 清理所有调试对象
def cleanup_debug_objects():
    to_delete = [o for o in bpy.data.objects 
                 if o.name.startswith("DEBUG_")]
    for obj in to_delete:
        bpy.data.objects.remove(obj, do_unlink=True)
    print(f"[DEBUG] Cleaned up {len(to_delete)} debug objects")
```

## 常见陷阱

### 陷阱1：调试对象影响正式渲染
```python
# 解决：设置hide_render=True
debug_obj.hide_render = True
# 或：渲染后清理
cleanup_debug_objects()
```
