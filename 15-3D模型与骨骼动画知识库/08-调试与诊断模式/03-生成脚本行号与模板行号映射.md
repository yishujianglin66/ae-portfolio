# 生成脚本行号与模板行号映射

## 问题场景

本项目使用Python模板生成Blender脚本（cel_shading.py生成临时脚本执行）。当生成的脚本报错时，错误行号是生成脚本的行号，而非模板源码的行号。需要建立映射关系快速定位问题。

## 核心原理

### 问题示例

```python
# 模板文件: cel_shading.py
# 生成脚本: temp/generated_script.py

# 错误信息：
# File "temp/generated_script.py", line 1523, in <module>
#   _bc = calculate_bounds_center(weapon_obj)
# NameError: name '_bc' is not defined

# 问题：line 1523是生成脚本的行号
# 需要找到cel_shading.py中对应的代码位置
```

### 行号映射表

```python
class ScriptLineMapper:
    """生成脚本行号映射器"""
    
    def __init__(self):
        self.mappings = []  # [(gen_line, template_file, template_line)]
    
    def add_mapping(self, gen_line, template_file, template_line):
        self.mappings.append((gen_line, template_file, template_line))
    
    def lookup(self, gen_line):
        """查找生成行号对应的模板位置"""
        # 找最近的映射点
        best = None
        for g, tf, tl in self.mappings:
            if g <= gen_line:
                if best is None or g > best[0]:
                    best = (g, tf, tl)
        
        if best:
            offset = gen_line - best[0]
            return best[1], best[2] + offset
        return None, None
```

### 代码生成时记录映射

```python
class ScriptGenerator:
    """带行号追踪的脚本生成器"""
    
    def __init__(self, template_file):
        self.template_file = template_file
        self.output_lines = []
        self.mapper = ScriptLineMapper()
        self.current_line = 1
    
    def emit(self, code, template_line=None):
        """输出一行代码"""
        self.output_lines.append(code)
        if template_line:
            self.mapper.add_mapping(
                self.current_line, 
                self.template_file, 
                template_line
            )
        self.current_line += 1
    
    def emit_block(self, code_block, start_template_line):
        """输出代码块"""
        lines = code_block.split('\n')
        for i, line in enumerate(lines):
            self.emit(line, start_template_line + i)
    
    def generate(self):
        return '\n'.join(self.output_lines)
    
    def save_with_map(self, output_file):
        """保存脚本和映射文件"""
        # 保存脚本
        with open(output_file, 'w', encoding='utf-8') as f:
            f.write(self.generate())
        
        # 保存映射
        import json
        map_file = output_file + '.map'
        with open(map_file, 'w') as f:
            json.dump(self.mapper.mappings, f)
```

### 错误行号解析

```python
import re
import traceback

def parse_error_line(error_msg):
    """从错误信息中提取行号"""
    # 匹配: File "xxx.py", line 123
    match = re.search(r'line (\d+)', error_msg)
    if match:
        return int(match.group(1))
    return None

def resolve_error_location(error_msg, mapper):
    """解析错误的真实位置"""
    gen_line = parse_error_line(error_msg)
    if gen_line:
        template_file, template_line = mapper.lookup(gen_line)
        if template_file:
            return f"{template_file}:{template_line}"
    return f"generated_script:{gen_line}"

# 使用示例
error = '''
Traceback (most recent call last):
  File "temp/generated_script.py", line 1523, in <module>
    _bc = calculate_bounds_center(weapon_obj)
NameError: name '_bc' is not defined
'''

location = resolve_error_location(error, mapper)
print(f"Error at: {location}")
# 输出: cel_shading.py:1637
```

### 内联注释标记

```python
# 在生成的脚本中添加注释标记
def generate_with_markers(template_lines):
    """生成带标记的脚本"""
    output = []
    
    for i, line in enumerate(template_lines, 1):
        # 添加行号注释
        output.append(f"# [L{i}]")
        output.append(line)
    
    return '\n'.join(output)

# 生成的脚本：
# # [L1]
# import bpy
# # [L2]
# import os
# ...

# 错误时可以看到最近的标记
```

```python
# 在生成脚本开头添加调试信息
DEBUG_HEADER = '''
import sys
import traceback

def debug_exception_handler(exc_type, exc_value, exc_tb):
    """自定义异常处理器"""
    print("[EXCEPTION]", file=sys.stderr)
    print(f"Type: {exc_type.__name__}", file=sys.stderr)
    print(f"Value: {exc_value}", file=sys.stderr)
    
    # 打印完整traceback
    tb_lines = traceback.format_exception(exc_type, exc_value, exc_tb)
    for line in tb_lines:
        print(line, end='', file=sys.stderr)
    
    # 打印局部变量
    if exc_tb:
        frame = exc_tb.tb_frame
        print("[LOCALS]", file=sys.stderr)
        for k, v in frame.f_locals.items():
            if not k.startswith('__'):
                print(f"  {k} = {repr(v)[:100]}", file=sys.stderr)

sys.excepthook = debug_exception_handler
'''
```

## 常见陷阱

### 陷阱1：条件代码导致行号偏移
```python
# 模板中有if/else，生成时只取一个分支
# 行号映射需要考虑跳过的代码

# 解决：记录实际生成的行，而非模板行
```

### 陷阱2：多文件模板
```python
# 生成脚本可能合并多个模板文件
# 映射需要记录来源文件

mapper.add_mapping(gen_line, "cel_shading.py", 1637)
mapper.add_mapping(gen_line, "utils.py", 45)
```
