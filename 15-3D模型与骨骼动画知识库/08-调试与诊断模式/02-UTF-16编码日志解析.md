# UTF-16编码日志解析

## 问题场景

Windows下Blender的某些输出（特别是通过subprocess捕获时）可能使用UTF-16编码。直接读取会出现乱码或解析错误。本项目v12_full_log.txt就是UTF-16编码的典型案例。

## 核心原理

### 编码问题来源

```python
# Windows控制台默认编码：
# - 中文系统：GBK/CP936
# - 英文系统：CP1252
# - PowerShell：可能UTF-16 LE

# Blender输出编码：
# - stdout：通常是UTF-8
# - 某些Windows API调用：UTF-16
# - 日志文件：取决于写入方式

# subprocess捕获时的编码问题
import subprocess

# 错误：不指定编码
result = subprocess.run(cmd, capture_output=True)
# result.stdout是bytes，需要解码

# 正确：指定编码
result = subprocess.run(cmd, capture_output=True, text=True, 
                        encoding='utf-8')
```

### UTF-16日志读取

```python
def read_utf16_log(filepath):
    """读取UTF-16编码的日志文件"""
    # 方法1：自动检测BOM
    with open(filepath, 'r', encoding='utf-16') as f:
        content = f.read()
    
    # 方法2：指定具体编码
    with open(filepath, 'r', encoding='utf-16-le') as f:  # Little Endian
        content = f.read()
    
    with open(filepath, 'r', encoding='utf-16-be') as f:  # Big Endian
        content = f.read()
    
    return content

def detect_encoding(filepath):
    """检测文件编码"""
    with open(filepath, 'rb') as f:
        raw = f.read(4)
    
    # BOM检测
    if raw.startswith(b'\xff\xfe'):
        return 'utf-16-le'
    elif raw.startswith(b'\xfe\xff'):
        return 'utf-16-be'
    elif raw.startswith(b'\xef\xbb\xbf'):
        return 'utf-8-sig'
    
    # 无BOM，尝试UTF-8
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            f.read()
        return 'utf-8'
    except UnicodeDecodeError:
        pass
    
    # 尝试UTF-16
    try:
        with open(filepath, 'r', encoding='utf-16') as f:
            f.read()
        return 'utf-16'
    except:
        pass
    
    return 'unknown'
```

### subprocess编码处理

```python
import subprocess
import sys

def run_blender_script(script_path, blender_exe="blender"):
    """运行Blender脚本并正确处理输出编码"""
    cmd = [
        blender_exe,
        "--background",
        "--python", script_path
    ]
    
    # 方法1：让Python处理编码
    result = subprocess.run(
        cmd,
        capture_output=True,
        text=True,
        encoding='utf-8',
        errors='replace'  # 无法解码的字符用?替代
    )
    
    # 方法2：手动解码（更灵活）
    result = subprocess.run(cmd, capture_output=True)
    
    # 尝试多种编码
    for encoding in ['utf-8', 'utf-16', 'gbk', 'cp936']:
        try:
            stdout = result.stdout.decode(encoding)
            stderr = result.stderr.decode(encoding)
            break
        except UnicodeDecodeError:
            continue
    else:
        # 都失败，用errors='replace'
        stdout = result.stdout.decode('utf-8', errors='replace')
        stderr = result.stderr.decode('utf-8', errors='replace')
    
    return stdout, stderr
```

### 日志解析

```python
import re

def parse_blender_log(log_content):
    """解析Blender日志，提取关键信息"""
    lines = log_content.split('\n')
    
    result = {
        'errors': [],
        'warnings': [],
        'phases': [],
        'progress': [],
    }
    
    for line in lines:
        # 错误
        if '[ERROR]' in line or 'Error:' in line:
            result['errors'].append(line.strip())
        
        # 警告
        if '[WARN]' in line or 'Warning:' in line:
            result['warnings'].append(line.strip())
        
        # 阶段
        if '[PHASE' in line:
            result['phases'].append(line.strip())
        
        # 进度
        if '[PROGRESS]' in line:
            result['progress'].append(line.strip())
        
        # Python异常
        if 'Traceback' in line:
            result['errors'].append(line.strip())
    
    return result

# 使用
log_content = read_utf16_log("v12_full_log.txt")
parsed = parse_blender_log(log_content)
print(f"Errors: {len(parsed['errors'])}")
print(f"Warnings: {len(parsed['warnings'])}")
```

### 编码转换工具

```python
def convert_log_encoding(input_file, output_file, 
                         from_encoding='utf-16', 
                         to_encoding='utf-8'):
    """转换日志文件编码"""
    with open(input_file, 'r', encoding=from_encoding) as f:
        content = f.read()
    
    with open(output_file, 'w', encoding=to_encoding) as f:
        f.write(content)
    
    print(f"Converted: {input_file} ({from_encoding}) -> "
          f"{output_file} ({to_encoding})")

# 批量转换
import glob
for log_file in glob.glob("logs/*.log"):
    encoding = detect_encoding(log_file)
    if encoding != 'utf-8':
        convert_log_encoding(log_file, log_file + '.utf8',
                            encoding, 'utf-8')
```

## 常见陷阱

### 陷阱1：BOM字符
```python
# UTF-16文件开头有BOM（FF FE或FE FF）
# 读取时可能包含BOM字符
content = content.lstrip('\ufeff')  # 移除BOM
```

### 陷阱2：混合编码
```python
# 日志中可能有不同编码的内容
# 解决：使用errors='replace'或'reignore'
content = raw.decode('utf-8', errors='replace')
```

### 陷阱3：Windows路径中的中文
```python
# 路径包含中文时可能编码错误
# 解决：使用原始字符串或正斜杠
path = r"C:\用户\文档\model.fbx"
# 或
path = "C:/用户/文档/model.fbx"
```
