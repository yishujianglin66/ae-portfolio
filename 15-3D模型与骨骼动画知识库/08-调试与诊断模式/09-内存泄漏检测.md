# 内存泄漏检测

## 问题场景

长时间运行的Blender脚本（如批量渲染）可能出现内存持续增长，最终导致OOM崩溃。需要检测内存泄漏来源：未释放的BMesh、残留的数据块、循环引用等。

## 核心原理

### 内存监控

```python
import tracemalloc
import gc

class MemoryMonitor:
    """内存监控器"""
    
    def __init__(self):
        tracemalloc.start()
        self.snapshots = []
    
    def checkpoint(self, label):
        """创建内存快照"""
        snapshot = tracemalloc.take_snapshot()
        self.snapshots.append((label, snapshot))
        
        current, peak = tracemalloc.get_traced_memory()
        print(f"[MEMORY] {label}: {current/1024/1024:.1f}MB "
              f"(peak: {peak/1024/1024:.1f}MB)")
    
    def compare(self, idx1, idx2):
        """对比两个快照"""
        label1, snap1 = self.snapshots[idx1]
        label2, snap2 = self.snapshots[idx2]
        
        stats = snap2.compare_to(snap1, 'lineno')
        
        print(f"\n[MEMORY DIFF] {label1} -> {label2}")
        print("Top 10 memory increases:")
        for stat in stats[:10]:
            print(f"  {stat}")
    
    def stop(self):
        tracemalloc.stop()

# 使用
monitor = MemoryMonitor()

monitor.checkpoint("Start")
import_fbx()
monitor.checkpoint("After Import")
create_bones()
monitor.checkpoint("After Bones")

monitor.compare(0, 2)  # 对比Start和After Bones
monitor.stop()
```

### 数据块泄漏检测

```python
def check_data_leaks():
    """检查未使用的数据块"""
    leaks = {
        'meshes': [],
        'materials': [],
        'textures': [],
        'images': [],
        'armatures': [],
    }
    
    # 检查users=0的数据块
    for mesh in bpy.data.meshes:
        if mesh.users == 0:
            leaks['meshes'].append(mesh.name)
    
    for mat in bpy.data.materials:
        if mat.users == 0 and not mat.use_fake_user:
            leaks['materials'].append(mat.name)
    
    for tex in bpy.data.textures:
        if tex.users == 0:
            leaks['textures'].append(tex.name)
    
    for img in bpy.data.images:
        if img.users == 0:
            leaks['images'].append(img.name)
    
    for arm in bpy.data.armatures:
        if arm.users == 0:
            leaks['armatures'].append(arm.name)
    
    # 报告
    total = sum(len(v) for v in leaks.values())
    if total > 0:
        print(f"[LEAK] Found {total} unused data blocks:")
        for dtype, names in leaks.items():
            if names:
                print(f"  {dtype}: {len(names)}")
                for name in names[:5]:  # 只显示前5个
                    print(f"    - {name}")
    else:
        print("[OK] No data leaks detected")
    
    return leaks

def cleanup_leaks():
    """清理泄漏的数据块"""
    cleaned = 0
    
    for mesh in list(bpy.data.meshes):
        if mesh.users == 0:
            bpy.data.meshes.remove(mesh)
            cleaned += 1
    
    for mat in list(bpy.data.materials):
        if mat.users == 0 and not mat.use_fake_user:
            bpy.data.materials.remove(mat)
            cleaned += 1
    
    for img in list(bpy.data.images):
        if img.users == 0:
            bpy.data.images.remove(img)
            cleaned += 1
    
    gc.collect()
    print(f"[CLEANUP] Removed {cleaned} unused data blocks")
```

### BMesh泄漏

```python
# 错误：BMesh未释放
def bad_bmesh_usage(obj):
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    # 操作...
    # 忘记bm.free()！内存泄漏！

# 正确：使用try/finally确保释放
def good_bmesh_usage(obj):
    bm = bmesh.new()
    try:
        bm.from_mesh(obj.data)
        # 操作...
        bm.to_mesh(obj.data)
    finally:
        bm.free()  # 总是释放

# 或使用上下文管理器（自定义）
from contextlib import contextmanager

@contextmanager
def bmesh_from_object(obj):
    """BMesh上下文管理器"""
    bm = bmesh.new()
    try:
        bm.from_mesh(obj.data)
        yield bm
        bm.to_mesh(obj.data)
    finally:
        bm.free()

# 使用
with bmesh_from_object(obj) as bm:
    bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
```

```python
    depsgraph = bpy.context.evaluated_depsgraph_get()

    mesh = obj_eval.to_mesh()
```

```python
def manage_images():
    """图像内存管理"""
    
    # 检查加载的图像
    for img in bpy.data.images:
        print(f"{img.name}: {img.size[0]}x{img.size[1]}, "
              f"users={img.users}")
        
        # 大图像占用大量内存
        if img.size[0] * img.size[1] > 4096 * 4096:
            print(f"  [WARN] Large image: {img.name}")
    
    # 释放未使用的图像
    for img in list(bpy.data.images):
        if img.users == 0 and img.name != 'Render Result':
```

### 循环引用检测

```python
import gc

def detect_circular_references():
    """检测循环引用"""
    # 强制垃圾回收
    gc.collect()
    
    # 获取无法回收的对象
    garbage = gc.garbage
    if garbage:
        print(f"[WARN] {len(garbage)} uncollectable objects:")
        for obj in garbage[:10]:
            print(f"  {type(obj)}: {repr(obj)[:100]}")
    else:
        print("[OK] No circular references")
    
    return len(garbage)

# 启用调试
gc.set_debug(gc.DEBUG_SAVEALL)
```

### 批量渲染内存管理

```python
def batch_render_with_cleanup(frame_ranges, cleanup_interval=10):
    """批量渲染（定期清理）"""
    scene = bpy.context.scene
    
    for i, (start, end) in enumerate(frame_ranges):
        scene.frame_start = start
        scene.frame_end = end
        
        print(f"[BATCH] Rendering frames {start}-{end}")
        bpy.ops.render.render(animation=True)
        
        # 定期清理
        if (i + 1) % cleanup_interval == 0:
            print(f"[CLEANUP] After {i+1} batches")
            
            # 清理未使用数据
            cleanup_leaks()
            
            # 强制GC
            gc.collect()
            
            # 检查内存
            current, peak = tracemalloc.get_traced_memory()
            print(f"[MEMORY] Current: {current/1024/1024:.1f}MB")
```

## 常见陷阱

### 陷阱1：list()避免迭代问题
```python
# 错误：迭代时删除
for mesh in bpy.data.meshes:
    if mesh.users == 0:
        bpy.data.meshes.remove(mesh)  # RuntimeError!

# 正确：
for mesh in list(bpy.data.meshes):
    if mesh.users == 0:
        bpy.data.meshes.remove(mesh)
```

### 陷阱2：fake_user阻止清理
```python
# use_fake_user=True的数据块不会被自动清理
```
