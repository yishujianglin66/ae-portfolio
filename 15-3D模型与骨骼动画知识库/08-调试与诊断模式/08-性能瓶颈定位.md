## 问题场景

Blender脚本执行缓慢时，需要定位性能瓶颈：是FBX导入慢、权重计算慢、还是渲染慢？通过计时和性能分析，找到优化方向。

import time
from functools import wraps

def timer(func):
    """函数计时装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"[TIMER] {func.__name__}: {elapsed:.3f}s")
        return result
    return wrapper

### 阶段计时器

```python
class PhaseTimer:
    """阶段计时器"""
    
    def __init__(self):
        self.phases = {}
        self.current_phase = None
        self.start_time = None
    
    def start(self, phase_name):
        if self.current_phase:
            self.end()  # 结束上一个阶段
        
        self.current_phase = phase_name
        self.start_time = time.perf_counter()
        print(f"[PHASE START] {phase_name}")
    
    def end(self):
        if self.current_phase:
            elapsed = time.perf_counter() - self.start_time
            self.phases[self.current_phase] = elapsed
            print(f"[PHASE END] {self.current_phase}: {elapsed:.3f}s")
            self.current_phase = None
    
    def report(self):
        """输出计时报告"""
        print("\n" + "="*50)
        print("PERFORMANCE REPORT")
        print("="*50)
        
        total = sum(self.phases.values())
        for phase, elapsed in self.phases.items():
            percent = elapsed / total * 100 if total > 0 else 0
            bar = "█" * int(percent / 5)
            print(f"  {phase:30s} {elapsed:8.3f}s ({percent:5.1f}%) {bar}")
        
        print("-"*50)
        print(f"  {'TOTAL':30s} {total:8.3f}s")
        print("="*50 + "\n")

# 使用
timer = PhaseTimer()

timer.start("FBX Import")
import_fbx()
timer.end()

timer.start("Bone Creation")
create_bones()
timer.end()

timer.start("Weight Assignment")
assign_weights()
timer.end()

timer.start("Render")
render_scene()
timer.end()

timer.report()
```

### 循环性能分析

```python
def profile_loop(iterations, func, *args):
    """分析循环性能"""
    times = []
    
    for i in range(iterations):
        start = time.perf_counter()
        func(i, *args)
        elapsed = time.perf_counter() - start
        times.append(elapsed)
        
        if (i + 1) % 100 == 0:
            avg = sum(times[-100:]) / 100
            print(f"  Iteration {i+1}: avg {avg*1000:.2f}ms/iter")
    
    total = sum(times)
    avg = total / iterations
    
    print(f"\n[PROFILE] {iterations} iterations")
    print(f"  Total: {total:.3f}s")
    print(f"  Average: {avg*1000:.2f}ms")
    print(f"  Min: {min(times)*1000:.2f}ms")
    print(f"  Max: {max(times)*1000:.2f}ms")
    
    return times
```

import sys
import gc

def get_memory_usage():
    """获取Python内存使用"""
    import tracemalloc
    
    if not tracemalloc.is_tracing():
        tracemalloc.start()
    
    current, peak = tracemalloc.get_traced_memory()
    return current / 1024 / 1024, peak / 1024 / 1024  # MB

def memory_checkpoint(label):
    """内存检查点"""
    current, peak = get_memory_usage()
    print(f"[MEMORY] {label}: {current:.1f}MB (peak: {peak:.1f}MB)")

### Blender对象统计

```python
def scene_statistics():
    """场景统计信息"""
    stats = {
        'objects': len(bpy.data.objects),
        'meshes': len(bpy.data.meshes),
        'materials': len(bpy.data.materials),
        'textures': len(bpy.data.textures),
        'images': len(bpy.data.images),
    }
    
    # 网格统计
    total_verts = 0
    total_faces = 0
    for obj in bpy.data.objects:
        if obj.type == 'MESH':
            total_verts += len(obj.data.vertices)
            total_faces += len(obj.data.polygons)
    
    stats['total_vertices'] = total_verts
    stats['total_faces'] = total_faces
    
    # 打印
    print("\n[SCENE STATS]")
    for key, value in stats.items():
        print(f"  {key}: {value}")
    
    return stats
```

### 渲染性能分析

```python
def analyze_render_performance(frame_count=5):
    """分析渲染性能"""
    scene = bpy.context.scene
    
    # 保存设置
    orig_end = scene.frame_end
    orig_start = scene.frame_start
    
    # 渲染测试帧
    scene.frame_start = 1
    scene.frame_end = frame_count
    
    frame_times = []
    for frame in range(1, frame_count + 1):
        scene.frame_set(frame)
        
        start = time.perf_counter()
        bpy.ops.render.render(write_still=False)
        elapsed = time.perf_counter() - start
        
        frame_times.append(elapsed)
        print(f"  Frame {frame}: {elapsed:.2f}s")
    
    # 恢复
    scene.frame_start = orig_start
    scene.frame_end = orig_end
    
    # 统计
    avg = sum(frame_times) / len(frame_times)
    print(f"\n[RENDER PERF]")
    print(f"  Average: {avg:.2f}s/frame")
    print(f"  Estimated total ({orig_end - orig_start + 1} frames): "
          f"{avg * (orig_end - orig_start + 1) / 60:.1f}min")
    
    return frame_times
```

import io

def profile_function(func, *args, **kwargs):
    """使用cProfile分析函数"""
    profiler = cProfile.Profile()

    result = func(*args, **kwargs)

    stats.sort_stats('cumulative')

    return result

## 常见陷阱

### 陷阱1：计时包含无关操作
```python
# 错误：计时包含print
start = time.time()
result = operation()
print(f"Result: {result}")  # 这也被计时了！
elapsed = time.time() - start

# 正确：只计时核心操作
start = time.perf_counter()
result = operation()
elapsed = time.perf_counter() - start
print(f"Result: {result}, Time: {elapsed:.3f}s")
```

### 陷阱2：首次运行慢（冷启动）
```python
# 第一次渲染通常比后续慢（shader编译等）
# 解决：预热后再计时
bpy.ops.render.render(write_still=False)  # 预热
# 然后正式计时
```
