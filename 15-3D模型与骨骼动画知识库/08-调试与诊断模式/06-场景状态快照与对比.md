# 场景状态快照与对比

## 问题场景

调试时需要对比不同阶段的场景状态：导入后、骨骼创建后、权重分配后等。通过状态快照和差异对比，可以快速定位哪个步骤引入了问题。

## 核心原理

### 状态快照

```python
import bpy
import json
import hashlib
from datetime import datetime

class SceneSnapshot:
    """场景状态快照"""
    
    def __init__(self, name):
        self.name = name
        self.timestamp = datetime.now().isoformat()
        self.data = self._capture()
    
    def _capture(self):
        """捕获当前场景状态"""
        data = {
            'name': self.name,
            'timestamp': self.timestamp,
            'objects': {},
            'materials': [],
            'render_settings': {},
        }
        
        # 对象状态
        for obj in bpy.data.objects:
            obj_data = {
                'type': obj.type,
                'location': [round(v, 6) for v in obj.location],
                'rotation': [round(v, 6) for v in obj.rotation_euler],
                'scale': [round(v, 6) for v in obj.scale],
                'parent': obj.parent.name if obj.parent else None,
                'visible': not obj.hide_get(),
            }
            
            if obj.type == 'MESH':
                mesh = obj.data
                obj_data['vertices'] = len(mesh.vertices)
                obj_data['faces'] = len(mesh.polygons)
                obj_data['vertex_groups'] = len(obj.vertex_groups)
                obj_data['modifiers'] = [m.type for m in obj.modifiers]
                
                # 顶点位置哈希（检测网格变化）
                verts_str = str([v.co[:] for v in mesh.vertices[:100]])
                obj_data['verts_hash'] = hashlib.md5(
                    verts_str.encode()).hexdigest()[:8]
            
            if obj.type == 'ARMATURE':
                obj_data['bones'] = len(obj.data.bones)
                obj_data['bone_names'] = [b.name for b in obj.data.bones]
            
            data['objects'][obj.name] = obj_data
        
        # 材质
        for mat in bpy.data.materials:
            data['materials'].append(mat.name)
        
        # 渲染设置
        scene = bpy.context.scene
        data['render_settings'] = {
            'engine': scene.render.engine,
            'resolution': [scene.render.resolution_x, 
                          scene.render.resolution_y],
            'frame': scene.frame_current,
        }
        
        return data
    
    def save(self, filepath):
        """保存快照到文件"""
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(self.data, f, indent=2, ensure_ascii=False)
        print(f"[SNAPSHOT] Saved: {filepath}")
    
    @staticmethod
    def load(filepath):
        """从文件加载快照"""
        with open(filepath, 'r', encoding='utf-8') as f:
            data = json.load(f)
        snapshot = SceneSnapshot(data['name'])
        snapshot.data = data
        return snapshot
```

### 状态对比

```python
def compare_snapshots(snap1, snap2):
    """对比两个快照的差异"""
    diff = {
        'added_objects': [],
        'removed_objects': [],
        'modified_objects': {},
        'summary': [],
    }
    
    objs1 = set(snap1.data['objects'].keys())
    objs2 = set(snap2.data['objects'].keys())
    
    # 新增对象
    diff['added_objects'] = list(objs2 - objs1)
    for name in diff['added_objects']:
        obj = snap2.data['objects'][name]
        diff['summary'].append(
            f"+ {name} ({obj['type']})")
    
    # 删除对象
    diff['removed_objects'] = list(objs1 - objs2)
    for name in diff['removed_objects']:
        diff['summary'].append(f"- {name}")
    
    # 修改对象
    for name in objs1 & objs2:
        obj1 = snap1.data['objects'][name]
        obj2 = snap2.data['objects'][name]
        
        changes = []
        
        # 位置变化
        if obj1['location'] != obj2['location']:
            changes.append(
                f"location: {obj1['location']} -> {obj2['location']}")
        
        # Parent变化
        if obj1['parent'] != obj2['parent']:
            changes.append(
                f"parent: {obj1['parent']} -> {obj2['parent']}")
        
        # 顶点组变化
        if obj1.get('vertex_groups') != obj2.get('vertex_groups'):
            changes.append(
                f"vertex_groups: {obj1.get('vertex_groups')} -> "
                f"{obj2.get('vertex_groups')}")
        
        # 修改器变化
        if obj1.get('modifiers') != obj2.get('modifiers'):
            changes.append(
                f"modifiers: {obj1.get('modifiers')} -> "
                f"{obj2.get('modifiers')}")
        
        if changes:
            diff['modified_objects'][name] = changes
            for c in changes:
                diff['summary'].append(f"~ {name}: {c}")
    
    return diff

def print_diff(diff):
    """打印差异报告"""
    print("\n" + "="*60)
    print("SNAPSHOT DIFF")
    print("="*60)
    
    for line in diff['summary']:
        print(f"  {line}")
    
    print(f"\nTotal changes: {len(diff['summary'])}")
    print("="*60 + "\n")
```

### 管线检查点

```python
class PipelineCheckpoints:
    """管线检查点管理器"""
    
    def __init__(self, output_dir):
        self.output_dir = output_dir
        self.snapshots = {}
        import os
        os.makedirs(output_dir, exist_ok=True)
    
    def checkpoint(self, name):
        """创建检查点"""
        snap = SceneSnapshot(name)
        filepath = f"{self.output_dir}/checkpoint_{name}.json"
        snap.save(filepath)
        self.snapshots[name] = snap
        return snap
    
    def compare(self, name1, name2):
        """对比两个检查点"""
        if name1 not in self.snapshots or name2 not in self.snapshots:
            print(f"[ERROR] Checkpoint not found: {name1} or {name2}")
            return None
        
        diff = compare_snapshots(
            self.snapshots[name1], self.snapshots[name2])
        print_diff(diff)
        return diff
    
    def compare_all(self):
        """对比所有相邻检查点"""
        names = list(self.snapshots.keys())
        for i in range(len(names) - 1):
            print(f"\n--- {names[i]} -> {names[i+1]} ---")
            self.compare(names[i], names[i+1])

# 使用
checkpoints = PipelineCheckpoints("debug/checkpoints")

# 管线各阶段
import_fbx()
checkpoints.checkpoint("after_import")

create_bones()
checkpoints.checkpoint("after_bones")

assign_weights()
checkpoints.checkpoint("after_weights")

# 对比
checkpoints.compare("after_import", "after_bones")
checkpoints.compare_all()
```

### 快速状态验证

```python
def validate_state(expected):
    """验证当前状态是否符合预期"""
    errors = []
    
    # 检查对象存在
    for obj_name in expected.get('objects', []):
        if obj_name not in bpy.data.objects:
            errors.append(f"Missing object: {obj_name}")
    
    # 检查对象数量
    if 'object_count' in expected:
        actual = len(bpy.data.objects)
        if actual != expected['object_count']:
            errors.append(
                f"Object count: {actual} != {expected['object_count']}")
    
    # 检查Armature
    if expected.get('has_armature'):
        armatures = [o for o in bpy.data.objects if o.type == 'ARMATURE']
        if not armatures:
            errors.append("No armature found")
    
    # 检查顶点组
    if 'min_vertex_groups' in expected:
        for obj in bpy.data.objects:
            if obj.type == 'MESH':
                if len(obj.vertex_groups) < expected['min_vertex_groups']:
                    errors.append(
                        f"{obj.name}: too few vertex groups")
    
    if errors:
        print("[VALIDATE] FAILED:")
        for e in errors:
            print(f"  - {e}")
        return False
    else:
        print("[VALIDATE] PASSED")
        return True

# 使用
validate_state({
    'has_armature': True,
    'min_vertex_groups': 10,
    'objects': ['Armature', 'Body', 'Hair'],
})
```
