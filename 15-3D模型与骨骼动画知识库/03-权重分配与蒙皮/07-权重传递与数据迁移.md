# 权重传递与数据迁移

## 问题场景

当模型拓扑改变（如添加Subdivision、合并部件）或需要将权重从一个mesh复制到另一个mesh时，需要权重传递（Data Transfer）技术。

## 核心原理

### Data Transfer Modifier

```python
import bpy

def transfer_weights(source_obj, target_obj):
    """将source的权重传递到target（基于最近顶点）"""
    
    # 添加Data Transfer modifier
    mod = target_obj.modifiers.new("WeightTransfer", 'DATA_TRANSFER')
    mod.object = source_obj
    mod.use_object_transform = True  # 考虑对象变换
    
    # 配置顶点数据传递
    mod.data_types = {'VGROUP_WEIGHTS'}
    mod.vert_mapping = 'NEAREST'  # 最近顶点匹配
    # 其他选项: 'EDGE_INTERP_NEAREST', 'TOPOLOGY'
    
    # 应用modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier="WeightTransfer")
```

### 手动权重复制（基于空间位置）

```python
from mathutils import Vector
import bpy

def copy_weights_by_proximity(source_obj, target_obj, radius=0.1):
    """基于空间邻近度复制权重"""
    src_mesh = source_obj.data
    tgt_mesh = target_obj.data
    src_world = source_obj.matrix_world
    tgt_world = target_obj.matrix_world
    
    # 确保target有相同的vertex_groups
    for vg in source_obj.vertex_groups:
        if vg.name not in target_obj.vertex_groups:
            target_obj.vertex_groups.new(name=vg.name)
    
    # 对target的每个顶点，找source中最近的顶点
    for tgt_v in tgt_mesh.vertices:
        tgt_world_co = tgt_world @ tgt_v.co
        
        # 找最近的source顶点
        min_dist = float('inf')
        nearest_src = None
        for src_v in src_mesh.vertices:
            src_world_co = src_world @ src_v.co
            d = (tgt_world_co - src_world_co).length
            if d < min_dist:
                min_dist = d
                nearest_src = src_v
        
        if nearest_src and min_dist < radius:
            # 复制权重
            for g in nearest_src.groups:
                group_name = source_obj.vertex_groups[g.group].name
                tgt_vg = target_obj.vertex_groups[group_name]
                tgt_vg.add([tgt_v.index], g.weight, 'REPLACE')
```

### 拓扑相同的直接复制

```python
def copy_weights_same_topology(source_obj, target_obj):
    """拓扑完全相同时直接复制权重"""
    assert len(source_obj.data.vertices) == len(target_obj.data.vertices)
    
    # 确保vertex_groups存在
    for vg in source_obj.vertex_groups:
        if vg.name not in target_obj.vertex_groups:
            target_obj.vertex_groups.new(name=vg.name)
    
    # 逐顶点复制
    for v_idx, src_v in enumerate(source_obj.data.vertices):
        for g in src_v.groups:
            group_name = source_obj.vertex_groups[g.group].name
            target_obj.vertex_groups[group_name].add([v_idx], g.weight, 'REPLACE')
```

## 常见陷阱

### 陷阱1：对象变换未对齐
```python
# 如果source和target的location/rotation/scale不同
# 必须启用 use_object_transform = True
# 否则顶点位置不匹配
```

### 陷阱2：vertex_group名称不一致
```python
# Data Transfer按名称匹配vertex_group
# source有"Hand_R"，target有"Hand.R" → 不传递
```
