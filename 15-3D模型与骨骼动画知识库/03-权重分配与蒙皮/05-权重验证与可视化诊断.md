# 权重验证与可视化诊断

## 问题场景

权重分配后需要验证质量：是否有未加权顶点、权重是否归一化、特定骨骼是否获得了合理的顶点。Blender提供Weight Paint模式可视化，但background模式下需要程序化诊断。

## 核心原理

### 程序化权重验证

```python
import bpy

def validate_weights(mesh_obj, armature_obj):
    """完整权重质量验证"""
    mesh = mesh_obj.data
    issues = []
    
    # 1. 检查未加权顶点
    unweighted = [v.index for v in mesh.vertices if len(v.groups) == 0]
    if unweighted:
        issues.append(f"未加权顶点: {len(unweighted)}个 (例: {unweighted[:5]})")
    
    # 2. 检查权重归一化
    bad_norm = []
    for v in mesh.vertices:
        total = sum(g.weight for g in v.groups)
        if v.groups and abs(total - 1.0) > 0.01:
            bad_norm.append((v.index, total))
    if bad_norm:
        issues.append(f"权重未归一化: {len(bad_norm)}个 (例: v{bad_norm[0][0]}={bad_norm[0][1]:.3f})")
    
    # 3. 检查空骨骼（无顶点分配的骨骼）
    bone_names = set(b.name for b in armature_obj.data.bones)
    used_groups = set()
    for v in mesh.vertices:
        for g in v.groups:
            if g.weight > 0.01:
                used_groups.add(mesh_obj.vertex_groups[g.group].name)
    
    empty_bones = bone_names - used_groups
    if empty_bones:
        issues.append(f"空骨骼(无权重): {empty_bones}")
    
    # 4. 检查权重范围
    for v in mesh.vertices:
        for g in v.groups:
            if g.weight < 0 or g.weight > 1:
                issues.append(f"权重越界: v{v.index} group{g.group} = {g.weight}")
                break
    
    return issues

def print_weight_report(mesh_obj, armature_obj):
    """输出权重分配报告"""
    mesh = mesh_obj.data
    
    print(f"\n{'='*50}")
    print(f"Weight Report: {mesh_obj.name}")
    print(f"{'='*50}")
    print(f"Vertices: {len(mesh.vertices)}")
    print(f"Vertex Groups: {len(mesh_obj.vertex_groups)}")
    print(f"Bones: {len(armature_obj.data.bones)}")
    
    # 每根骨骼的权重统计
    print(f"\n{'Bone':<15} {'Verts':>6} {'Avg W':>7} {'Max W':>7}")
    print("-" * 40)
    
    for bone in armature_obj.data.bones:
        vg = mesh_obj.vertex_groups.get(bone.name)
        if not vg:
            print(f"{bone.name:<15} {'N/A':>6}")
            continue
        
        weights = []
        for v in mesh.vertices:
            try:
                w = vg.weight(v.index)
                if w > 0.001:
                    weights.append(w)
            except RuntimeError:
                pass
        
        if weights:
            avg = sum(weights) / len(weights)
            mx = max(weights)
            print(f"{bone.name:<15} {len(weights):>6} {avg:>7.3f} {mx:>7.3f}")
        else:
            print(f"{bone.name:<15} {0:>6} {'---':>7} {'---':>7}")
    
    # 验证
    issues = validate_weights(mesh_obj, armature_obj)
    if issues:
        print(f"\n[ISSUES FOUND]")
        for issue in issues:
            print(f"  ! {issue}")
    else:
        print(f"\n[ALL CHECKS PASSED]")
```

### 权重热力图导出（用于调试）

```python
def export_weight_as_vertex_colors(mesh_obj, bone_name):
    """将特定骨骼的权重导出为顶点颜色（可在viewport查看）"""
    mesh = mesh_obj.data
    
    # 创建顶点颜色层
    color_layer = mesh.vertex_colors.new(name=f"weight_{bone_name}")
    
    vg = mesh_obj.vertex_groups.get(bone_name)
    if not vg:
        return
    
    # 对每个loop设置颜色（权重→灰度）
    for poly in mesh.polygons:
        for loop_idx in poly.loop_indices:
            vert_idx = mesh.loops[loop_idx].vertex_index
            try:
                w = vg.weight(vert_idx)
            except RuntimeError:
                w = 0.0
            
            # 权重0=蓝色, 权重1=红色（类似Blender Weight Paint）
            r = w
            g = 1.0 - abs(w - 0.5) * 2
            b = 1.0 - w
            color_layer.data[loop_idx].color = (r, g, b, 1.0)
```

### Z-slice权重分析（本项目使用）

```python
def z_slice_weight_analysis(mesh_obj, bone_name, num_slices=10):
    """按Z高度分层分析权重分布"""
    mesh = mesh_obj.data
    world_matrix = mesh_obj.matrix_world
    
    # 获取Z范围
    z_values = [(world_matrix @ v.co).z for v in mesh.vertices]
    z_min, z_max = min(z_values), max(z_values)
    slice_height = (z_max - z_min) / num_slices
    
    vg = mesh_obj.vertex_groups.get(bone_name)
    if not vg:
        return
    
    print(f"\nZ-Slice Analysis for '{bone_name}':")
    for i in range(num_slices):
        z_low = z_min + i * slice_height
        z_high = z_low + slice_height
        
        # 找该层的顶点
        slice_verts = []
        for v in mesh.vertices:
            wz = (world_matrix @ v.co).z
            if z_low <= wz < z_high:
                try:
                    w = vg.weight(v.index)
                    slice_verts.append(w)
                except RuntimeError:
                    pass
        
        if slice_verts:
            avg_w = sum(slice_verts) / len(slice_verts)
            bar = '#' * int(avg_w * 40)
            print(f"  Z[{z_low:.1f},{z_high:.1f}]: {len(slice_verts):4d} verts, avg={avg_w:.3f} |{bar}")
```
