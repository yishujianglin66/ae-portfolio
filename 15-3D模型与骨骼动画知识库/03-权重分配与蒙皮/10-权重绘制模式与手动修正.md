# 权重绘制模式与手动修正

## 问题场景

自动权重分配后，某些区域需要手动微调（如手指、面部边缘）。Blender的Weight Paint模式提供交互式绘制，但background模式下需要用脚本模拟。

## 核心原理

### Weight Paint模式基础

```python
import bpy

def enter_weight_paint(mesh_obj, armature_obj):
    """进入Weight Paint模式（需要GUI）"""
    bpy.context.view_layer.objects.active = armature_obj
    armature_obj.select_set(True)
    mesh_obj.select_set(True)
    
    # 进入Pose模式选择骨骼
    bpy.ops.object.mode_set(mode='POSE')
    # 选择目标骨骼
    armature_obj.data.bones['Hand_R'].select = True
    
    # 切换到Weight Paint
    bpy.ops.object.mode_set(mode='WEIGHT_PAINT')
```

### 脚本化权重绘制（Brush模拟）

```python
def script_weight_brush(mesh_obj, bone_name, vertex_indices, weight, mode='REPLACE'):
    """
    模拟Weight Paint的笔刷操作
    mode: 'REPLACE', 'ADD', 'SUBTRACT', 'MIX'
    """
    vg = mesh_obj.vertex_groups.get(bone_name)
    if not vg:
        vg = mesh_obj.vertex_groups.new(name=bone_name)
    
    if mode == 'REPLACE':
        vg.add(vertex_indices, weight, 'REPLACE')
    elif mode == 'ADD':
        vg.add(vertex_indices, weight, 'ADD')
    elif mode == 'SUBTRACT':
        vg.add(vertex_indices, weight, 'SUBTRACT')
    elif mode == 'MIX':
        # MIX: 向weight值插值
        for idx in vertex_indices:
            try:
                current = vg.weight(idx)
            except RuntimeError:
                current = 0.0
            new_w = current + (weight - current) * 0.5  # 50%混合
            vg.add([idx], new_w, 'REPLACE')
```

### 区域选择与批量修正

```python
def select_vertices_by_region(mesh_obj, region_fn):
    """
    按区域条件选择顶点
    region_fn: 接收顶点世界坐标，返回True/False
    """
    world_matrix = mesh_obj.matrix_world
    selected = []
    
    for v in mesh_obj.data.vertices:
        world_co = world_matrix @ v.co
        if region_fn(world_co):
            selected.append(v.index)
    
    return selected

# 使用示例：选择右手区域（X < -0.5, Z在2.0-3.0之间）
hand_verts = select_vertices_by_region(
    mesh_obj,
    lambda co: co.x < -0.5 and 2.0 < co.z < 3.0
)

# 将这些顶点的Hand_R权重设为1.0
script_weight_brush(mesh_obj, 'Hand_R', hand_verts, 1.0, 'REPLACE')
```

### 权重平滑

```python
def smooth_weights(mesh_obj, iterations=3, factor=0.5):
    """平滑权重（减少突变）"""
    bpy.context.view_layer.objects.active = mesh_obj
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    bpy.ops.object.mode_set(mode='WEIGHT_PAINT')
    
    for _ in range(iterations):
        bpy.ops.object.vertex_group_smooth(factor=factor, repeat=1)
    
    bpy.ops.object.mode_set(mode='OBJECT')
```
