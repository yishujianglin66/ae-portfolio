```typescript
// 在 Phase 6 中调用 Python 分析器
import { spawn } from 'child_process';

async function analyzeMusic(musicFile: string): Promise<MusicAtom> {
    return new Promise((resolve, reject) => {
        const python = spawn('python', [
            'analyzers/music_analyzer.py',
            musicFile,
            '--output', 'json'
        ]);
        
        let output = '';
        python.stdout.on('data', (data) => output += data);
        python.stderr.on('data', (data) => console.error(`Python error: ${data}`));
        python.on('close', (code) => {
            if (code !== 0) return reject(new Error(`Python exited with ${code}`));
            try {
                resolve(JSON.parse(output));
            } catch (e) {
                reject(new Error(`Invalid JSON output: ${e}`));
            }
        });
    });
}
```
