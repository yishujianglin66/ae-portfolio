```javascript
class AEPerformanceOptimizer {

        app.preferences.setPrefAsLong(
            "Preview", 
            "Draft3D", 
            1
        );
        
        app.preferences.setPrefAsLong(
            "Preview", 
            "AdaptiveResolution", 
            1
        );
        
        app.preferences.setPrefAsLong(
            "Preview", 
            "AdaptiveResolutionLimit", 
            1
        );
    }

    static optimizeMultiprocessing() {
        app.preferences.setPrefAsLong(
            "Memory", 
            "EnableMultiProcessing", 
            1
        );
        
        app.preferences.setPrefAsLong(
            "Memory", 
            "NumCPUs", 
            Math.max(1, Math.floor(navigator.hardwareConcurrency * 0.75))
        );
    }

    static clearMediaCache() {
        var cachePath = app.preferences.getPrefAsString(
            "MediaCache", 
            "MediaCacheFolder"
        );
        
        var folder = new Folder(cachePath);
        if (folder.exists) {
            var files = folder.getFiles();
            files.forEach(function(file) {
                if (file.constructor.name === 'File' && !file.hidden) {
                    try {
                        file.remove();
                    } catch (e) {
                        // File in use, skip
                    }
                }
            });
        }
    }

    static renderAndReplaceComplexLayers(comp) {
        for (var i = 1; i <= comp.numLayers; i++) {
            var layer = comp.layer(i);
            if (layer.effects.length > 5 || layer.source instanceof FootageItem) {
                try {
                    layer.renderAndReplace();
                } catch (e) {
                    // Skip locked or pre-compose layers
                }
            }
        }
    }

    static flattenCompositions(comps) {
        comps.forEach(function(comp) {
            var flattened = comp.flatten();
            flattened.name = comp.name + ' (Flattened)';
        });
    }

    static getPerformanceMetrics() {
        return {
            memoryUsage: app.preferences.getPrefAsLong("Memory", "CurrentMemoryUsage"),
            diskCacheSize: app.preferences.getPrefAsLong("Memory", "DiskCacheSize"),
            numEffects: app.project.activeItem ? app.project.activeItem.numEffects : 0,
            numLayers: app.project.activeItem ? app.project.activeItem.numLayers : 0,
            compSize: app.project.activeItem ? {
                width: app.project.activeItem.width,
                height: app.project.activeItem.height
            } : null
        };
    }

    static recommendOptimizations(comp) {
        var recommendations = [];
        
        if (comp.numLayers > 50) {
            recommendations.push('图层数量过多，建议使用预合成');
        }
        
        if (comp.numEffects > 10) {
            recommendations.push('效果数量过多，建议渲染替换');
        }
        
        if (comp.width > 3840 || comp.height > 2160) {
            recommendations.push('超高分辨率合成，建议降低分辨率或使用代理');
        }
        
        return recommendations;
    }
}
```

```python
        project.SetPreference("Fusion", "CacheLimit", 512)
        project.SetPreference("Fusion", "UseGPU", True)
        project.SetPreference("Fusion", "RenderMode", "OpenGL")
        
        return True

    def clear_cache(resolve):
        project = resolve.GetProjectManager().GetCurrentProject()
        
        project.DeleteCacheFiles(True)
        
        return True

            timeline.GenerateProxies(
                "DNxHR LB",
                "1280x720",
                "Half",
                "/path/to/proxies"

            timeline.SetRenderCachePolicy("Smart")
            timeline.SetOptimizedMedia("On")
```

```javascript
class PPPerformanceOptimizer {
    static optimizeSequence(sequence) {
        sequence.renderPreview();
    }

    static clearPreviewCache() {
        var cacheFolder = PP.project.preferences.previewCacheFolder;
        var folder = new Folder(cacheFolder);
        if (folder.exists) {
            var files = folder.getFiles();
            files.forEach(function(file) {
                if (file.constructor.name === 'File') {
                    try {
                        file.remove();
                    } catch (e) {
                        // Skip locked files
                    }
                }
            });
        }
    }

    static disableGPUAcceleration() {
        PP.project.preferences.gpuAcceleration = false;
    }

    static enableGPUAcceleration() {
        PP.project.preferences.gpuAcceleration = true;
    }

    static setPreviewResolution(resolution) {
        PP.project.preferences.previewResolution = resolution;
    }

    static setPreviewQuality(quality) {
        PP.project.preferences.previewQuality = quality;
    }

    static proxyMedia(sequence, proxySettings) {
        for (var i = 0; i < sequence.videoTracks.length; i++) {
            var track = sequence.videoTracks[i];
            for (var j = 0; j < track.clips.length; j++) {
                var clip = track.clips[j];
                clip.createProxy(proxySettings);
            }
        }
    }

    static toggleProxy(sequence, enabled) {
        for (var i = 0; i < sequence.videoTracks.length; i++) {
            var track = sequence.videoTracks[i];
            for (var j = 0; j < track.clips.length; j++) {
                var clip = track.clips[j];
                clip.useProxy = enabled;
            }
        }
    }

    static renderAndReplaceEffects(clip) {
        clip.renderAndReplace();
    }

    static flattenSequence(sequence) {
        sequence.flatten();
    }

    static getPerformanceMetrics() {
        return {
            memoryUsage: 0,
            cpuUsage: 0,
            gpuUsage: 0,
            diskUsage: 0,
            previewQuality: PP.project.preferences.previewQuality,
            gpuAcceleration: PP.project.preferences.gpuAcceleration
        };
    }

    static recommendOptimizations(sequence) {
        var recommendations = [];
        var clipCount = 0;
        
        for (var i = 0; i < sequence.videoTracks.length; i++) {
            clipCount += sequence.videoTracks[i].clips.length;
        }
        
        if (clipCount > 100) {
            recommendations.push('剪辑数量过多，建议使用嵌套序列');
        }
        
        if (!PP.project.preferences.gpuAcceleration) {
            recommendations.push('建议启用GPU加速');
```

```python
                })
```

```python
import time
import json
import threading

class PerformanceMonitor:
    def __init__(self):
        self.metrics = []
        self.is_running = False
        self.interval = 5
        
    def start_monitoring(self):
        self.is_running = True
        thread = threading.Thread(target=self._monitor_loop)
        thread.daemon = True
        thread.start()
        
    def stop_monitoring(self):
        self.is_running = False
        
    def _monitor_loop(self):
        while self.is_running:
            metric = self._collect_metrics()
            self.metrics.append(metric)
            
            if len(self.metrics) > 1000:
                self.metrics = self.metrics[-1000:]
            
            time.sleep(self.interval)
            
    def _collect_metrics(self):
        return {
            'timestamp': time.time(),
            'memory_usage': psutil.virtual_memory().percent,
            'cpu_usage': psutil.cpu_percent(),
            'disk_io': {
                'read_bytes': psutil.disk_io_counters().read_bytes,
                'write_bytes': psutil.disk_io_counters().write_bytes
            },
            'network_io': {
                'bytes_sent': psutil.net_io_counters().bytes_sent,
                'bytes_recv': psutil.net_io_counters().bytes_recv
            }
        }
        
    def get_metrics(self, limit=100):
        return self.metrics[-limit:]
    
    def get_average_metrics(self, minutes=5):
        cutoff_time = time.time() - (minutes * 60)
        recent_metrics = [m for m in self.metrics if m['timestamp'] > cutoff_time]
        
        if not recent_metrics:
            return None
            
        return {
            'avg_memory': sum(m['memory_usage'] for m in recent_metrics) / len(recent_metrics),
            'avg_cpu': sum(m['cpu_usage'] for m in recent_metrics) / len(recent_metrics),
            'sample_count': len(recent_metrics)
        }
        
    def save_metrics_to_file(self, filepath):
        with open(filepath, 'w') as f:
            json.dump(self.metrics, f, indent=2)
            
    def load_metrics_from_file(self, filepath):
        with open(filepath, 'r') as f:
            self.metrics = json.load(f)
```

            }

            self.log_manager.warning(alert['message'])
            
        self.alerts.extend(triggered_alerts)
        
        if len(self.alerts) > 100:
            self.alerts = self.alerts[-100:]
            
        return triggered_alerts
    
    def get_alerts(self, severity=None):
        if severity:
            return [a for a in self.alerts if a['severity'] == severity]
        return self.alerts

        self.alerts = []
