## 第1章 系统总览

音画匹配引擎是整个风格化剪辑系统的**核心决策中枢**，它承接上游的音乐分析与片段分析结果，通过多层匹配算法生成精准的剪辑方案，最终通过参数反推与桥接层交付给下游的编译器执行。

## 第2章 节拍-片段对齐引擎

### 2.1 问题定义

节拍-片段对齐的核心目标是：**将每个视频片段的起始点锚定到音乐的节拍点上，使画面切换与音乐节奏产生感官同步**。

数学表述为：给定节拍序列 $B = \{b_1, b_2, ..., b_m\}$ 和片段序列 $C = \{c_1, c_2, ..., c_n\}$，寻找对齐映射 $f: C \rightarrow B$，使得总对齐代价最小：

$$
\min_f \sum_{i=1}^{n} cost(c_i, f(c_i)) + \lambda \cdot smoothness(f)
$$

### 2.2 动态规划对齐算法

#### 2.2.1 算法原理

动态规划是解决序列对齐问题的经典方法。定义 $dp[i][j]$ 表示将前 $i$ 个片段对齐到前 $j$ 个节拍点的最小代价。

状态转移方程：

$$
dp[i][j] = \min_{k < j} \left( dp[i-1][k] + align\_cost(c_i, b_j, b_k) \right)
$$

其中 $align\_cost$ 包含三部分：
- **位置偏差代价**：片段时长与节拍间距的匹配程度
- **能量相关代价**：片段视觉能量与节拍强度的相关性
- **跳跃惩罚**：避免跳过过多节拍导致的节奏断裂

#### 2.2.2 Python 完整实现

```python
import numpy as np
from typing import List, Tuple, Optional
from dataclasses import dataclass

@dataclass
class AlignResult:
    clip_id: str
    beat_time: float
    beat_confidence: float
    score: float
    next_beat_time: Optional[float] = None

class BeatClipAligner:
    def __init__(
        self,
        position_weight: float = 0.4,
        energy_weight: float = 0.3,
        duration_weight: float = 0.3,
        max_beat_skip: int = 3,
        min_clip_duration_ratio: float = 0.5,
        max_clip_duration_ratio: float = 2.0
    ):
        self.position_weight = position_weight
        self.energy_weight = energy_weight
        self.duration_weight = duration_weight
        self.max_beat_skip = max_beat_skip
        self.min_clip_duration_ratio = min_clip_duration_ratio
        self.max_clip_duration_ratio = max_clip_duration_ratio

    def _duration_cost(self, clip_duration: float, beat_interval: float) -> float:
        ratio = clip_duration / max(beat_interval, 0.001)
        if self.min_clip_duration_ratio <= ratio <= self.max_clip_duration_ratio:
            return 1.0 - abs(ratio - 1.0) / (self.max_clip_duration_ratio - 1.0)
        return 0.0

    def _energy_cost(self, clip_energy: float, beat_confidence: float) -> float:
        diff = abs(clip_energy - beat_confidence)
        return max(0.0, 1.0 - diff)

    def _position_cost(self, offset_ratio: float) -> float:
        return max(0.0, 1.0 - abs(offset_ratio) * 2.0)

    def _compute_cost_matrix(
        self,
        clips: List[dict],
        beats: List[float],
        beat_confidences: List[float]
    ) -> np.ndarray:
        n_clips = len(clips)
        n_beats = len(beats)
        cost_matrix = np.full((n_clips, n_beats), np.inf)

        beat_intervals = np.diff(beats) if n_beats > 1 else np.array([1.0])

        for i in range(n_clips):
            clip_duration = clips[i]["duration"]
            clip_energy = clips[i]["visual_energy"]

            for j in range(n_beats):
                if j == 0:
                    duration_score = self._duration_cost(clip_duration, beat_intervals[0] if len(beat_intervals) > 0 else 1.0)
                else:
                    duration_score = self._duration_cost(clip_duration, beat_intervals[j-1] if j-1 < len(beat_intervals) else beat_intervals[-1])

                energy_score = self._energy_cost(clip_energy, beat_confidences[j])
                position_score = 1.0

                total_cost = 1.0 - (
                    self.duration_weight * duration_score +
                    self.energy_weight * energy_score +
                    self.position_weight * position_score
                )
                cost_matrix[i, j] = total_cost

        return cost_matrix

    def align_dp(
        self,
        clips: List[dict],
        beats: List[float],
        beat_confidences: Optional[List[float]] = None
    ) -> List[AlignResult]:
        if not clips or not beats:
            return []

        n_clips = len(clips)
        n_beats = len(beats)

        if beat_confidences is None:
            beat_confidences = [1.0] * n_beats

        cost_matrix = self._compute_cost_matrix(clips, beats, beat_confidences)

        dp = np.full((n_clips, n_beats), np.inf)
        prev = np.full((n_clips, n_beats), -1, dtype=int)

        for j in range(min(n_beats, self.max_beat_skip + 1)):
            dp[0, j] = cost_matrix[0, j]

        for i in range(1, n_clips):
            for j in range(n_beats):
                min_cost = np.inf
                best_k = -1
                start_k = max(0, j - self.max_beat_skip)
                for k in range(start_k, j):
                    if dp[i-1, k] + cost_matrix[i, j] < min_cost:
                        min_cost = dp[i-1, k] + cost_matrix[i, j]
                        best_k = k
                if best_k >= 0:
                    dp[i, j] = min_cost
                    prev[i, j] = best_k

        results = []
        if n_clips > 0:
            j = np.argmin(dp[-1])
            for i in range(n_clips - 1, -1, -1):
                next_beat = beats[j + 1] if j + 1 < n_beats else None
                results.append(AlignResult(
                    clip_id=clips[i].get("clip_id", f"clip_{i}"),
                    beat_time=beats[j],
                    beat_confidence=beat_confidences[j],
                    score=1.0 - dp[i, j],
                    next_beat_time=next_beat
                ))
                j = prev[i, j]
            results.reverse()

        return results
```

### 2.3 DTW（Dynamic Time Warping）实现

DTW 是一种更灵活的序列对齐方法，允许时间轴的非线性伸缩，特别适合音乐节拍与视频片段之间存在节奏漂移的场景。

#### 2.3.1 使用 dtw-python 库实现

```python
try:
    from dtw import dtw, rabinerJuangStepPattern
    DTW_AVAILABLE = True
except ImportError:
    DTW_AVAILABLE = False
    print("Warning: dtw-python not installed. Using numpy fallback.")

class DTWBeatClipAligner(BeatClipAligner):
    def align_dtw(
        self,
        clips: List[dict],
        beats: List[float],
        beat_confidences: Optional[List[float]] = None
    ) -> List[AlignResult]:
        if not DTW_AVAILABLE:
            return self.align_dp(clips, beats, beat_confidences)

        if not clips or not beats:
            return []

        n_clips = len(clips)
        n_beats = len(beats)

        if beat_confidences is None:
            beat_confidences = [1.0] * n_beats

        clip_energies = np.array([c["visual_energy"] for c in clips])
        beat_energies = np.array(beat_confidences)

        clip_times = np.cumsum([0.0] + [c["duration"] for c in clips])[:-1]
        beat_times = np.array(beats)

        def custom_distance(i, j):
            ci = min(int(i), n_clips - 1)
            bj = min(int(j), n_beats - 1)

            energy_diff = abs(clip_energies[ci] - beat_energies[bj])
            time_diff = abs(clip_times[ci] - beat_times[bj]) / max(beat_times[-1], 0.001)
            duration_diff = abs(clips[ci]["duration"] - (beat_times[min(bj+1, n_beats-1)] - beat_times[bj]))

            return self.energy_weight * energy_diff + \
                   self.position_weight * time_diff + \
                   self.duration_weight * min(duration_diff / 2.0, 1.0)

        alignment = dtw(
            np.arange(n_clips).reshape(-1, 1),
            np.arange(n_beats).reshape(-1, 1),
            dist_method=custom_distance,
            step_pattern=rabinerJuangStepPattern(6, "c"),
            keep_internals=True
        )

        results = []
        used_beats = set()
        for ci, bj in zip(alignment.index1, alignment.index2):
            if bj not in used_beats:
                used_beats.add(bj)
                next_beat = beats[bj + 1] if bj + 1 < n_beats else None
                results.append(AlignResult(
                    clip_id=clips[ci].get("clip_id", f"clip_{ci}"),
                    beat_time=beats[bj],
                    beat_confidence=beat_confidences[bj],
                    score=1.0 - alignment.distance / max(n_clips, n_beats),
                    next_beat_time=next_beat
                ))

        return results
```

#### 2.3.2 纯 NumPy 手动实现（无依赖版本）

```python
class PureNumpyDTWAligner(BeatClipAligner):
    def align_numpy_dtw(
        self,
        clips: List[dict],
        beats: List[float],
        beat_confidences: Optional[List[float]] = None
    ) -> List[AlignResult]:
        if not clips or not beats:
            return []

        n_clips = len(clips)
        n_beats = len(beats)

        if beat_confidences is None:
            beat_confidences = [1.0] * n_beats

        clip_energies = np.array([c["visual_energy"] for c in clips])
        beat_energies = np.array(beat_confidences)

        cost_matrix = np.zeros((n_clips, n_beats))
        for i in range(n_clips):
            for j in range(n_beats):
                energy_diff = abs(clip_energies[i] - beat_energies[j])
                duration_diff = abs(
                    clips[i]["duration"] -
                    (beats[min(j+1, n_beats-1)] - beats[j])
                ) / max(clips[i]["duration"], 0.001)
                cost_matrix[i, j] = (
                    self.energy_weight * energy_diff +
                    self.duration_weight * min(duration_diff, 1.0)
                )

        dtw_matrix = np.full((n_clips + 1, n_beats + 1), np.inf)
        dtw_matrix[0, 0] = 0

        for i in range(1, n_clips + 1):
            for j in range(1, n_beats + 1):
                cost = cost_matrix[i-1, j-1]
                dtw_matrix[i, j] = cost + min(
                    dtw_matrix[i-1, j],
                    dtw_matrix[i, j-1],
                    dtw_matrix[i-1, j-1]
                )

        path = []
        i, j = n_clips, n_beats
        while i > 0 and j > 0:
            path.append((i-1, j-1))
            candidates = [
                (dtw_matrix[i-1, j], i-1, j),
                (dtw_matrix[i, j-1], i, j-1),
                (dtw_matrix[i-1, j-1], i-1, j-1)
            ]
            _, i, j = min(candidates, key=lambda x: x[0])
        path.reverse()

        results = []
        used_beats = set()
        for ci, bj in path:
            if bj not in used_beats:
                used_beats.add(bj)
                next_beat = beats[bj + 1] if bj + 1 < n_beats else None
                results.append(AlignResult(
                    clip_id=clips[ci].get("clip_id", f"clip_{ci}"),
                    beat_time=beats[bj],
                    beat_confidence=beat_confidences[bj],
                    score=1.0 - dtw_matrix[n_clips, n_beats] / max(n_clips, n_beats),
                    next_beat_time=next_beat
                ))

        return results
```

### 2.4 对齐评分函数

对齐质量通过多维度评分综合评估：

$$
AlignScore = \alpha \cdot E_{corr} + \beta \cdot D_{match} + \gamma \cdot B_{coverage}
$$

其中：
- $E_{corr}$：能量相关性（Pearson 相关系数）
- $D_{match}$：时长匹配度（平均时长比值）
- $B_{coverage}$：节拍覆盖率（被使用的节拍占比）

```python
def compute_alignment_score(
    results: List[AlignResult],
    clips: List[dict],
    beats: List[float],
    alpha: float = 0.4,
    beta: float = 0.35,
    gamma: float = 0.25
) -> dict:
    clip_energies = [c["visual_energy"] for c in clips[:len(results)]]
    beat_confs = [r.beat_confidence for r in results]

    if len(results) > 1 and np.std(clip_energies) > 0 and np.std(beat_confs) > 0:
        energy_corr = np.corrcoef(clip_energies, beat_confs)[0, 1]
        energy_corr = max(0.0, energy_corr)
    else:
        energy_corr = 0.5

    duration_scores = []
    for i, r in enumerate(results):
        if r.next_beat_time:
            beat_interval = r.next_beat_time - r.beat_time
            clip_dur = clips[i]["duration"]
            ratio = min(beat_interval, clip_dur) / max(beat_interval, clip_dur)
            duration_scores.append(ratio)
    duration_match = np.mean(duration_scores) if duration_scores else 0.5

    beat_coverage = len(set(r.beat_time for r in results)) / len(beats) if beats else 0

    total_score = alpha * energy_corr + beta * duration_match + gamma * beat_coverage

    return {
        "total_score": total_score,
        "energy_correlation": energy_corr,
        "duration_match": duration_match,
        "beat_coverage": beat_coverage,
        "n_aligned": len(results),
        "n_clips": len(clips),
        "n_beats": len(beats)
    }
```

## 第3章 段落-片段配对引擎

### 3.1 问题定义

段落-片段配对解决的是**粗粒度语义匹配**问题：将音乐的段落（intro/verse/chorus/bridge/outro）与视频片段进行最佳配对，使音乐情绪与画面情绪在宏观层面保持一致。

这是一个典型的**二分图最大权匹配**问题，可以用匈牙利算法（Kuhn-Munkres）求解。

### 3.2 匈牙利算法实现

使用 `scipy.optimize.linear_sum_assignment` 作为底层实现，封装为业务友好的接口。

```python
import numpy as np
from typing import List, Dict, Tuple, Optional
from dataclasses import dataclass
from scipy.optimize import linear_sum_assignment

@dataclass
class SegmentPair:
    segment_label: str
    segment_start: float
    segment_end: float
    segment_energy: float
    clip_id: str
    clip_duration: float
    clip_energy: float
    match_score: float
    energy_score: float
    duration_score: float
    semantic_score: float

class SegmentClipMatcher:
    def __init__(
        self,
        alpha: float = 0.4,
        beta: float = 0.3,
        gamma: float = 0.3,
        use_semantic: bool = False
    ):
        self.alpha = alpha
        self.beta = beta
        self.gamma = gamma
        self.use_semantic = use_semantic

    def _energy_match_score(self, seg_energy: float, clip_energy: float) -> float:
        diff = abs(seg_energy - clip_energy)
        return max(0.0, 1.0 - diff * 1.5)

    def _duration_match_score(
        self,
        seg_duration: float,
        clip_duration: float
    ) -> float:
        if seg_duration <= 0 or clip_duration <= 0:
            return 0.0
        ratio = clip_duration / seg_duration
        if ratio <= 1.0:
            return ratio
        else:
            return max(0.0, 1.0 - (ratio - 1.0) * 0.5)

    def _semantic_match_score(
        self,
        seg_label: str,
        clip_tag: Optional[str]
    ) -> float:
        if not clip_tag or not self.use_semantic:
            return 0.5

        semantic_map = {
            "intro": ["calm", "ambient", "establishing", "wide"],
            "verse": ["narrative", "story", "medium", "character"],
            "chorus": ["energetic", "climax", "action", "closeup"],
            "bridge": ["transition", "buildup", "tension", "abstract"],
            "outro": ["calm", "resolution", "fade", "wide"]
        }

        seg_tags = set(semantic_map.get(seg_label, []))
        clip_tags = set([clip_tag.lower()])
        overlap = len(seg_tags & clip_tags)
        return min(1.0, overlap * 0.3 + 0.5)

    def _compute_score_matrix(
        self,
        segments: List[dict],
        clips: List[dict]
    ) -> np.ndarray:
        n_seg = len(segments)
        n_clip = len(clips)
        n = max(n_seg, n_clip)
        score_matrix = np.zeros((n, n))

        for i in range(n):
            for j in range(n):
                if i < n_seg and j < n_clip:
                    seg = segments[i]
                    clip = clips[j]

                    e_score = self._energy_match_score(
                        seg.get("energy", 0.5),
                        clip.get("visual_energy", 0.5)
                    )
                    d_score = self._duration_match_score(
                        seg.get("duration", 1.0),
                        clip.get("duration", 1.0)
                    )
                    s_score = self._semantic_match_score(
                        seg.get("label", ""),
                        clip.get("semantic_tag")
                    )

                    total = (
                        self.alpha * e_score +
                        self.beta * d_score +
                        self.gamma * s_score
                    )
                    score_matrix[i, j] = total
                else:
                    score_matrix[i, j] = -1.0

        return score_matrix

    def match(
        self,
        segments: List[dict],
        clips: List[dict]
    ) -> List[SegmentPair]:
        if not segments or not clips:
            return []

        n_seg = len(segments)
        n_clip = len(clips)

        score_matrix = self._compute_score_matrix(segments, clips)
        cost_matrix = -score_matrix

        row_ind, col_ind = linear_sum_assignment(cost_matrix)

        pairs = []
        for r, c in zip(row_ind, col_ind):
            if r < n_seg and c < n_clip and score_matrix[r, c] >= 0:
                seg = segments[r]
                clip = clips[c]
                e_score = self._energy_match_score(
                    seg.get("energy", 0.5),
                    clip.get("visual_energy", 0.5)
                )
                d_score = self._duration_match_score(
                    seg.get("duration", 1.0),
                    clip.get("duration", 1.0)
                )
                s_score = self._semantic_match_score(
                    seg.get("label", ""),
                    clip.get("semantic_tag")
                )

                pairs.append(SegmentPair(
                    segment_label=seg.get("label", f"seg_{r}"),
                    segment_start=seg.get("start", 0.0),
                    segment_end=seg.get("end", 0.0),
                    segment_energy=seg.get("energy", 0.5),
                    clip_id=clip.get("clip_id", f"clip_{c}"),
                    clip_duration=clip.get("duration", 1.0),
                    clip_energy=clip.get("visual_energy", 0.5),
                    match_score=score_matrix[r, c],
                    energy_score=e_score,
                    duration_score=d_score,
                    semantic_score=s_score
                ))

        pairs.sort(key=lambda p: p.segment_start)
        return pairs
```

### 3.3 综合评分公式详解

综合评分公式：

$$
S_{match} = \alpha \cdot S_{energy} + \beta \cdot S_{duration} + \gamma \cdot S_{semantic}
$$

各分量计算方法：

**能量匹配分**：

$$
S_{energy} = \max(0, 1 - 1.5 \cdot |E_{seg} - E_{clip}|)
$$

使用 1.5 作为系数是为了让能量差异 0.67 以上时得分归零，强化高区分度。

**时长匹配分**：

$$
S_{duration} = \begin{cases}
\frac{D_{clip}}{D_{seg}}, & D_{clip} \leq D_{seg} \\
\max(0, 1 - 0.5 \cdot (\frac{D_{clip}}{D_{seg}} - 1)), & D_{clip} > D_{seg}
\end{cases}
$$

片段时长不足时线性下降，过长时更宽容（0.5 系数）。

**语义匹配分**（可选）：

基于标签集合的 Jaccard 相似度，需要 CLIP 模型提供语义标签时启用。

### 3.4 CLIP 语义匹配增强（可选模块）

当需要更高精度的语义匹配时，可以接入 CLIP 模型计算音乐段落标签与视频帧的语义相似度。

```python
class CLIPEnhancedMatcher(SegmentClipMatcher):
    def __init__(self, clip_model_name: str = "ViT-B/32", **kwargs):
        super().__init__(use_semantic=True, **kwargs)
        self.clip_model = None
        self.clip_preprocess = None
        self.clip_tokenizer = None
        self._model_loaded = False
        self.clip_model_name = clip_model_name

    def _load_model(self):
        if self._model_loaded:
            return
        try:
            import clip
            import torch
            self.device = "cuda" if torch.cuda.is_available() else "cpu"
            self.clip_model, self.clip_preprocess = clip.load(
                self.clip_model_name, device=self.device
            )
            self._model_loaded = True
        except ImportError:
            print("CLIP not available, falling back to keyword matching")
            self.use_semantic = False

    def compute_clip_similarity(
        self,
        image_path: str,
        text_labels: List[str]
    ) -> np.ndarray:
        self._load_model()
        if not self._model_loaded:
            return np.ones(len(text_labels)) / len(text_labels)

        import clip
        import torch
        from PIL import Image

        image = self.clip_preprocess(Image.open(image_path)).unsqueeze(0).to(self.device)
        text = clip.tokenize(text_labels).to(self.device)

        with torch.no_grad():
            image_features = self.clip_model.encode_image(image)
            text_features = self.clip_model.encode_text(text)
            image_features /= image_features.norm(dim=-1, keepdim=True)
            text_features /= text_features.norm(dim=-1, keepdim=True)
            similarity = (image_features @ text_features.T).squeeze(0).cpu().numpy()

        return similarity

    def match_with_clip(
        self,
        segments: List[dict],
        clips: List[dict],
        clip_frame_paths: Dict[str, str]
    ) -> List[SegmentPair]:
        self._load_model()
        augmented_clips = []

        for clip_data in clips:
            clip_id = clip_data.get("clip_id")
            frame_path = clip_frame_paths.get(clip_id)
            if frame_path and self._model_loaded:
                seg_labels = [s.get("label", "") for s in segments]
                sims = self.compute_clip_similarity(frame_path, seg_labels)
                best_idx = np.argmax(sims)
                clip_data = dict(clip_data)
                clip_data["semantic_tag"] = seg_labels[best_idx]
            augmented_clips.append(clip_data)

        return self.match(segments, augmented_clips)
```

### 3.5 多对多配对扩展

当段落数与片段数不相等时，需要支持**多对多**配对：一个段落可以包含多个片段，一个片段也可以跨段落使用（不推荐）。

```python
class MultiPairMatcher(SegmentClipMatcher):
    def match_multi(
        self,
        segments: List[dict],
        clips: List[dict],
        max_clips_per_segment: int = 5,
        min_clips_per_segment: int = 1
    ) -> Dict[str, List[SegmentPair]]:
        all_pairs = []
        for seg in segments:
            seg_duration = seg.get("duration", 1.0)
            candidates = []

            for clip in clips:
                e_score = self._energy_match_score(
                    seg.get("energy", 0.5),
                    clip.get("visual_energy", 0.5)
                )
                d_score = self._duration_match_score(
                    seg_duration,
                    clip.get("duration", 1.0)
                )
                s_score = self._semantic_match_score(
                    seg.get("label", ""),
                    clip.get("semantic_tag")
                )
                total = self.alpha * e_score + self.beta * d_score + self.gamma * s_score
                candidates.append((clip, total, e_score, d_score, s_score))

            candidates.sort(key=lambda x: x[1], reverse=True)
            n_select = min(max_clips_per_segment, max(min_clips_per_segment, int(seg_duration / 2.0)))
            selected = candidates[:n_select]

            seg_pairs = []
            for clip, total, e_s, d_s, s_s in selected:
                seg_pairs.append(SegmentPair(
                    segment_label=seg.get("label", "unknown"),
                    segment_start=seg.get("start", 0.0),
                    segment_end=seg.get("end", 0.0),
                    segment_energy=seg.get("energy", 0.5),
                    clip_id=clip.get("clip_id", "unknown"),
                    clip_duration=clip.get("duration", 1.0),
                    clip_energy=clip.get("visual_energy", 0.5),
                    match_score=total,
                    energy_score=e_s,
                    duration_score=d_s,
                    semantic_score=s_s
                ))
            all_pairs.append((seg.get("label", "unknown"), seg_pairs))

        return dict(all_pairs)
```

### 4.4 效果堆叠策略

多个效果同时作用时需要考虑**视觉叠加优先级**，避免效果互相抵消或过度叠加导致画面混乱。
>
```python

    DOMINANT_ONLY = "dominant_only"
    WEIGHTED_SUM = "weighted_sum"
    LAYERED = "layered"
>
EFFECT_LAYER_PRIORITY = {
    EffectCategory.PARTICLE: 1,
    EffectCategory.FLASH: 2,
    EffectCategory.GLOW: 3,
    EffectCategory.COLOR: 4,
    EffectCategory.BLUR: 5,
    EffectCategory.SHAKE: 6,
}
>
def stack_effects(
    bindings: List[EffectBinding],
    strategy: EffectStackingStrategy = EffectStackingStrategy.LAYERED
) -> List[EffectBinding]:
    if not bindings:
        return []
>
    if strategy == EffectStackingStrategy.DOMINANT_ONLY:
        best = max(bindings, key=lambda b: b.intensity)
        return [best]
>
    elif strategy == EffectStackingStrategy.WEIGHTED_SUM:
        total_intensity = sum(b.intensity for b in bindings)
        if total_intensity == 0:
            return bindings
        normalized = []
        for b in bindings:
            scaled_keyframes = [(t, v / total_intensity) for t, v in b.keyframes]
            normalized.append(EffectBinding(
                clip_id=b.clip_id,
                effect_category=b.effect_category,
                band_name=b.band_name,
                intensity=b.intensity / total_intensity,
                keyframes=scaled_keyframes,
                param_name=b.param_name
            ))
        return normalized
>
    elif strategy == EffectStackingStrategy.LAYERED:
        sorted_bindings = sorted(
            bindings,
            key=lambda b: EFFECT_LAYER_PRIORITY.get(b.effect_category, 99)
        )
        return sorted_bindings
>
    return bindings
```
>
> [!info] 效果堆叠原则
> 1. **优先级分层**：粒子 > 闪烁 > 发光 > 色彩 > 模糊 > 震动
> 2. **强度衰减**：非主导效果强度乘以 0.6 的衰减系数
> 3. **同类互斥**：同一效果类别最多保留 1 个最强绑定
>

### 5.3 转场时长计算

$$

其中 $n_{beats}$ 是转场跨越的拍数，$\alpha$ 是能量影响系数（通常取 0.5）。

```python
            energy = np.array(fb.get("energy", []))
            if len(times) < 2:
                continue
            sample_times = np.linspace(0, duration, n_samples)
            interp_energy = np.interp(sample_times, times, energy, left=0, right=0)
>
            if band_name in ["sub_bass", "bass"]:
                bass_energy += interp_energy
            elif band_name in ["high_mid", "high"]:
                high_energy += interp_energy
>

        max_high = np.max(high_energy) if np.max(high_energy) > 0 else 1.0
>
        arousal = (bass_energy / max_bass) * 0.6 + (high_energy / max_high) * 0.4
        valence = 0.5 + (high_energy / max_high - bass_energy / max_bass) * 0.3
        valence = np.clip(valence, 0, 1)
>
        for i in range(n_samples):
            emotions.append((float(valence[i]), float(arousal[i])))
>
        return emotions
```

## 第8章 参数反推引擎

### 8.1 设计目标
>
参数反推引擎（Parameter Inferrer）是连接**抽象匹配决策**与**具体效果操作**的桥梁。它将 MatchPlan 中的高层语义（如"shake intensity 0.8"）转换为编译器可以直接执行的原子操作参数。
>
核心职责：
1. **强度映射**：将 0-1 的强度值映射为具体效果参数值
2. **关键帧生成**：根据关键帧模板生成时间-值序列
3. **转场操作生成**：将转场计划转换为效果操作链
4. **CompilerInput 组装**：输出标准化的编译器输入
>
相关参考：[[参数反向推演与原子输出规范]]

关键帧模板定义了效果强度随时间变化的曲线形态，是实现动态效果的核心。

```python
class KeyframeTemplate(str, Enum):
    BEAT_PULSE = "beat_pulse"
    PARAGRAPH_HOLD = "paragraph_hold"
    PEAK_BUILD = "peak_build"
    BREATHING = "breathing"
    CRESCENDO = "crescendo"
    DECRESCENDO = "decrescendo"
>
def generate_keyframes(
    template: KeyframeTemplate,
    duration: float,
    intensity: float,
    n_samples: int = 32,
    bpm: float = 120.0
) -> List[Tuple[float, float]]:
    times = np.linspace(0, duration, n_samples)
    values = np.zeros(n_samples)
>
    if template == KeyframeTemplate.BEAT_PULSE:
        beat_period = 60.0 / bpm
        n_beats = max(1, int(duration / beat_period))
        for i, t in enumerate(times):
            phase = (t % beat_period) / beat_period
            pulse = np.exp(-(phase - 0.0) ** 2 / (2 * 0.1 ** 2))
            values[i] = intensity * pulse
>
    elif template == KeyframeTemplate.PARAGRAPH_HOLD:
        attack = 0.1
        release = 0.1
        for i, t in enumerate(times):
            progress = t / max(duration, 0.001)
            if progress < attack:
                values[i] = intensity * (progress / attack)
            elif progress > 1.0 - release:
                values[i] = intensity * ((1.0 - progress) / release)
            else:
                values[i] = intensity
>
    elif template == KeyframeTemplate.PEAK_BUILD:
        peak_time = duration * 0.7
        for i, t in enumerate(times):
            if t <= peak_time:
                progress = t / max(peak_time, 0.001)
                values[i] = intensity * (progress ** 1.5)
            else:
                progress = (t - peak_time) / max(duration - peak_time, 0.001)
                values[i] = intensity * (1.0 - progress * 0.5)
>
    elif template == KeyframeTemplate.BREATHING:
        breath_period = max(duration / 4.0, 1.0)
        for i, t in enumerate(times):
            phase = (t / breath_period) * 2 * np.pi
            breath = 0.5 + 0.5 * np.sin(phase)
            values[i] = intensity * (0.6 + 0.4 * breath)
>
    elif template == KeyframeTemplate.CRESCENDO:
        for i, t in enumerate(times):
            progress = t / max(duration, 0.001)
            values[i] = intensity * (progress ** 2)
>
    elif template == KeyframeTemplate.DECRESCENDO:
        for i, t in enumerate(times):
            progress = t / max(duration, 0.001)
            values[i] = intensity * ((1.0 - progress) ** 2)
>
    return [(float(t), float(v)) for t, v in zip(times, values)]
```

```python
def map_intensity_to_param(
    intensity: float,
    param_range: ParamRange
) -> float:
    intensity = max(0.0, min(1.0, intensity))
>
    if param_range.curve == "linear":
        mapped = intensity
    elif param_range.curve == "quadratic":
        mapped = intensity ** 2
    elif param_range.curve == "sqrt":
        mapped = np.sqrt(intensity)
    elif param_range.curve == "log":
        mapped = np.log10(1 + 9 * intensity) / np.log10(10)
    else:
        mapped = intensity
>
    value = param_range.min_value + mapped * (param_range.max_value - param_range.min_value)
    return float(value)
>
def reverse_param_to_intensity(
    value: float,
    param_range: ParamRange
) -> float:
    ratio = (value - param_range.min_value) / max(param_range.max_value - param_range.min_value, 0.001)

>
    if param_range.curve == "linear":
        intensity = ratio
    elif param_range.curve == "quadratic":
        intensity = np.sqrt(ratio)
    elif param_range.curve == "sqrt":
        intensity = ratio ** 2
    elif param_range.curve == "log":
        intensity = (10 ** ratio - 1) / 9
```

### 9.1 方案对比与选型
>
Python-TS 桥接层负责连接 Python 侧的算法引擎与 TypeScript 侧的应用层。三种主流方案对比：
>
| 方案 | 延迟 | 并发 | 部署复杂度 | 适用场景 |
|------|------|------|------------|----------|
| Subprocess | 中（启动开销） | 低（进程级） | 低 | 轻量级、单机 |
| FastAPI HTTP | 低（连接复用） | 高（异步） | 中 | 中大规模、微服务 |
| gRPC | 最低（二进制） | 最高 | 高 | 大规模、高性能 |
>
**选型建议**：
- MVP / 单机版：Subprocess 方案（零配置、易调试）
- 生产环境 / 多用户：FastAPI 方案（平衡性能与复杂度）
- 超大规模 / 低延迟：gRPC 方案（极致性能）
>

### 10.5 并发控制与缓存策略
>
并发控制采用**令牌桶 + 队列**模式：
>
```
请求进入
   │
   ├─→ 缓存命中？→ 返回缓存结果
   │
   └─→ 活跃请求 < 上限？
           │
           ├─ 是 → 立即执行
           └─ 否 → 加入队列等待
```

### 11.1 错误分类体系

音画匹配系统中的错误按照**发生阶段 + 严重程度**两个维度进行分类，形成完整的错误码体系。

#### 11.1.1 错误分类矩阵

| 错误类别 | 错误码前缀 | 严重程度 | 是否可重试 | 典型场景 |
|----------|-----------|----------|-----------|----------|
| 输入错误 | E1000 | Warning | 否（需修正输入） | 缺少必要参数、格式错误 |
| 分析失败 | E2000 | Error | 是 | 音乐分析超时、片段解析失败 |
| 匹配失败 | E3000 | Error | 是 | 对齐算法不收敛、配对矩阵奇异 |
| 生成失败 | E4000 | Error | 是 | 参数反推异常、序列化失败 |
| 桥接错误 | E5000 | Critical | 是 | 进程通信失败、HTTP 5xx |
| 系统错误 | E9000 | Critical | 否 | 内存不足、磁盘满 |

### 11.2 降级策略

当某些模块不可用或性能不足时，系统采用**分级降级**策略，确保核心功能始终可用。

#### 11.2.1 降级层级定义

| 降级级别 | 触发条件 | 影响范围 | 用户体验 |
|----------|---------|---------|---------|
| L0 正常 | 所有模块正常工作 | 无 | 完整功能 |
| L1 轻度 | CLIP 模型不可用 / 非核心模块故障 | 语义匹配降级为关键词匹配 | 质量微降，不影响节奏 |
| L2 中度 | 音乐分析降级 / 频段数据缺失 | 效果绑定减少、转场类型简化 | 功能可用，效果数量减少 |
| L3 重度 | 匹配算法部分失败 | 使用默认模板匹配 | 基础剪辑可用，个性化降低 |
| L4 紧急 | Python 侧完全不可用 | TS 侧纯规则引擎 | 仅保证基本剪辑功能 |

#### 11.2.2 各模块降级方案

```python
from typing import Callable, Optional, Any
import time
import logging

logger = logging.getLogger(__name__)

class FallbackStrategy:
    def __init__(self):
        self.fallback_chains = {
            "semantic_matching": ["clip", "keyword", "random"],
            "beat_alignment": ["dtw", "dp", "even_spacing"],
            "effect_binding": ["full_band", "main_band", "fixed_pattern"],
            "transition_gen": ["smart", "random", "fixed_cut"],
        }

    def execute_with_fallback(
        self,
        operation_name: str,
        primary_fn: Callable,
        *args,
        **kwargs
    ) -> tuple[Any, str]:
        chain = self.fallback_chains.get(operation_name, ["primary"])
        last_error = None

        for level in chain:
            try:
                if level == "primary" or level == chain[0]:
                    result = primary_fn(*args, **kwargs)
                    return result, level
                else:
                    fallback_fn = self._get_fallback_fn(operation_name, level)
                    if fallback_fn:
                        result = fallback_fn(*args, **kwargs)
                        return result, level
            except Exception as e:
                last_error = e
                logger.warning(f"Fallback level {level} failed for {operation_name}: {e}")
                continue

        raise RuntimeError(f"All fallbacks failed for {operation_name}: {last_error}")

    def _get_fallback_fn(self, operation: str, level: str) -> Optional[Callable]:
        fallback_map = {
            ("semantic_matching", "keyword"): self._keyword_semantic_fallback,
            ("semantic_matching", "random"): self._random_semantic_fallback,
            ("beat_alignment", "even_spacing"): self._even_spacing_fallback,
            ("effect_binding", "main_band"): self._main_band_fallback,
            ("effect_binding", "fixed_pattern"): self._fixed_pattern_fallback,
            ("transition_gen", "random"): self._random_transition_fallback,
            ("transition_gen", "fixed_cut"): self._fixed_cut_fallback,
        }
        return fallback_map.get((operation, level))

    def _keyword_semantic_fallback(self, segments, clips, **kwargs):
        logger.info("Using keyword semantic fallback")
        from random import shuffle
        shuffled_clips = clips.copy()
        shuffle(shuffled_clips)
        return shuffled_clips

    def _random_semantic_fallback(self, segments, clips, **kwargs):
        logger.info("Using random semantic fallback")
        return clips

    def _even_spacing_fallback(self, clips, total_duration, **kwargs):
        logger.info("Using even spacing fallback for beat alignment")
        n_clips = len(clips)
        if n_clips == 0:
            return []
        interval = total_duration / n_clips
        return [{"beat_time": i * interval, "confidence": 0.5} for i in range(n_clips)]

    def _main_band_fallback(self, freq_bands, **kwargs):
        logger.info("Using main band fallback for effect binding")
        if freq_bands and len(freq_bands) > 0:
            return [freq_bands[len(freq_bands) // 2]]
        return []

    def _fixed_pattern_fallback(self, **kwargs):
        logger.info("Using fixed pattern fallback")
        return [{"pattern": "pulse", "frequency": 2.0}]

    def _random_transition_fallback(self, n_transitions, **kwargs):
        logger.info("Using random transition fallback")
        from random import choice
        types = ["cut", "dissolve", "fade"]
        return [choice(types) for _ in range(n_transitions)]

    def _fixed_cut_fallback(self, n_transitions, **kwargs):
        logger.info("Using fixed cut fallback")
        return ["cut"] * n_transitions
```

### 11.3 超时处理机制

系统对每个处理阶段设置独立的超时时间，确保单个环节故障不会导致整体阻塞。

#### 11.3.1 超时配置

```python
@dataclass
class TimeoutConfig:
    audio_analysis: float = 30.0
    clip_analysis: float = 60.0
    beat_alignment: float = 10.0
    segment_matching: float = 5.0
    effect_binding: float = 10.0
    transition_generation: float = 3.0
    narrative_planning: float = 5.0
    parameter_inference: float = 5.0
    total_pipeline: float = 120.0

class TimeoutManager:
    def __init__(self, config: TimeoutConfig = None):
        self.config = config or TimeoutConfig()
        self._start_times: Dict[str, float] = {}
        self._durations: Dict[str, float] = {}

    def start(self, stage: str) -> None:
        self._start_times[stage] = time.time()

    def check(self, stage: str) -> tuple[bool, float]:
        if stage not in self._start_times:
            return False, 0.0

        elapsed = time.time() - self._start_times[stage]
        timeout = getattr(self.config, stage, 60.0)
        return elapsed > timeout, elapsed

    def end(self, stage: str) -> float:
        if stage in self._start_times:
            duration = time.time() - self._start_times[stage]
            self._durations[stage] = duration
            return duration
        return 0.0

    def get_all_durations(self) -> Dict[str, float]:
        return dict(self._durations)
```

#### 11.3.2 带超时的函数执行器

```python
import signal
import functools
from typing import Callable, Any, TypeVar

T = TypeVar('T')

class TimeoutError(Exception):
    def __init__(self, timeout: float, stage: str = ""):
        self.timeout = timeout
        self.stage = stage
        super().__init__(f"Operation '{stage}' timed out after {timeout}s")

def with_timeout(timeout: float, stage: str = ""):
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @functools.wraps(func)
        def wrapper(*args, **kwargs) -> T:
            def _handle_timeout(signum, frame):
                raise TimeoutError(timeout, stage)

            old_handler = signal.signal(signal.SIGALRM, _handle_timeout)
            signal.alarm(int(timeout))
            try:
                result = func(*args, **kwargs)
            finally:
                signal.alarm(0)
                signal.signal(signal.SIGALRM, old_handler)
            return result
        return wrapper
    return decorator
```

> [!warning] Windows 兼容性
> `signal.SIGALRM` 在 Windows 上不可用。Windows 环境下请使用 `threading + Event` 方案实现超时。

#### 11.4.1 指数退避重试

```python
import random
import math
from typing import Callable, TypeVar

T = TypeVar('T')

@dataclass
class RetryConfig:
    max_retries: int = 3
    base_delay: float = 1.0
    max_delay: float = 30.0
    backoff_factor: float = 2.0
    jitter: bool = True
    retry_on_exceptions: tuple = (Exception,)

class RetryExecutor:
    def __init__(self, config: RetryConfig = None):
        self.config = config or RetryConfig()

    def execute(self, fn: Callable[..., T], *args, **kwargs) -> T:
        last_exception = None
        config = self.config

        for attempt in range(config.max_retries + 1):
            try:
                return fn(*args, **kwargs)
            except config.retry_on_exceptions as e:
                last_exception = e
                if attempt == config.max_retries:
                    break

                delay = self._calculate_delay(attempt)
                logger.warning(
                    f"Attempt {attempt + 1}/{config.max_retries + 1} failed: {e}. "
                    f"Retrying in {delay:.2f}s..."
                )
                time.sleep(delay)

        raise last_exception

    def _calculate_delay(self, attempt: int) -> float:
        config = self.config
        delay = config.base_delay * (config.backoff_factor ** attempt)
        delay = min(delay, config.max_delay)

        if config.jitter:
            delay = delay * (0.5 + random.random() * 0.5)

        return delay

    async def execute_async(self, fn: Callable[..., Any], *args, **kwargs) -> Any:
        import asyncio
        last_exception = None
        config = self.config

        for attempt in range(config.max_retries + 1):
            try:
                return await fn(*args, **kwargs)
            except config.retry_on_exceptions as e:
                last_exception = e
                if attempt == config.max_retries:
                    break

                delay = self._calculate_delay(attempt)
                logger.warning(f"Attempt {attempt + 1} failed: {e}. Retrying in {delay:.2f}s...")
                await asyncio.sleep(delay)

        raise last_exception
```

#### 11.4.2 错误处理器完整实现

```python
class ErrorHandler:
    def __init__(
        self,
        timeout_config: TimeoutConfig = None,
        retry_config: RetryConfig = None,
        enable_fallback: bool = True
    ):
        self.timeout_manager = TimeoutManager(timeout_config)
        self.retry_executor = RetryExecutor(retry_config)
        self.fallback_strategy = FallbackStrategy() if enable_fallback else None
        self.error_log: list = []
        self.enable_fallback = enable_fallback

    def safe_execute(
        self,
        stage: str,
        operation: str,
        fn: Callable,
        *args,
        **kwargs
    ) -> Dict[str, Any]:
        result = {
            "success": False,
            "data": None,
            "error": None,
            "fallback_used": None,
            "duration": 0.0,
            "stage": stage,
        }

        self.timeout_manager.start(stage)

        try:
            data = self.retry_executor.execute(fn, *args, **kwargs)
            result["success"] = True
            result["data"] = data
            result["duration"] = self.timeout_manager.end(stage)
            return result

        except Exception as e:
            result["error"] = self._format_error(e, stage)
            result["duration"] = self.timeout_manager.end(stage)
            self.error_log.append(result["error"])

            if self.enable_fallback and self.fallback_strategy:
                try:
                    fallback_data, fallback_level = self.fallback_strategy.execute_with_fallback(
                        operation, fn, *args, **kwargs
                    )
                    result["success"] = True
                    result["data"] = fallback_data
                    result["fallback_used"] = fallback_level
                    logger.info(f"Fallback to {fallback_level} for {operation}")
                except Exception as fe:
                    result["error"] = self._format_error(fe, stage, "fallback")
                    self.error_log.append(result["error"])

            return result

    def _format_error(
        self,
        exception: Exception,
        stage: str,
        context: str = "primary"
    ) -> Dict[str, Any]:
        return {
            "stage": stage,
            "context": context,
            "type": type(exception).__name__,
            "message": str(exception),
            "timestamp": time.time(),
        }

    def get_error_summary(self) -> Dict[str, Any]:
        errors_by_stage = {}
        for err in self.error_log:
            stage = err.get("stage", "unknown")
            if stage not in errors_by_stage:
                errors_by_stage[stage] = []
            errors_by_stage[stage].append(err)

        return {
            "total_errors": len(self.error_log),
            "errors_by_stage": errors_by_stage,
            "error_types": list(set(e["type"] for e in self.error_log)),
        }

    def has_critical_errors(self) -> bool:
        critical_types = {ErrorCategory.SYSTEM_ERROR, ErrorCategory.BRIDGE_ERROR}
        return any(
            err.get("category") in critical_types
            for err in self.error_log
        )
```

```typescript
export class BridgeError extends Error {
  readonly code: string;
  readonly severity: 'warning' | 'error' | 'critical';
  readonly stage: string;
  readonly retryable: boolean;
  readonly details?: Record<string, any>;

  constructor(
    code: string,
    message: string,
    options: {
      severity?: 'warning' | 'error' | 'critical';
      stage?: string;
      retryable?: boolean;
      details?: Record<string, any>;
    } = {}
  ) {
    super(message);
    this.name = 'BridgeError';
    this.code = code;
    this.severity = options.severity || 'error';
    this.stage = options.stage || 'unknown';
    this.retryable = options.retryable ?? true;
    this.details = options.details;
  }

  toJSON() {
    return {
      code: this.code,
      message: this.message,
      severity: this.severity,
      stage: this.stage,
      retryable: this.retryable,
      details: this.details,
    };
  }
}

export class ErrorHandlerTS {
  private retryConfig = {
    maxRetries: 3,
    baseDelay: 1000,
    maxDelay: 30000,
    backoffFactor: 2,
  };

  async withRetry<T>(
    fn: () => Promise<T>,
    options: Partial<typeof this.retryConfig> = {}
  ): Promise<T> {
    const config = { ...this.retryConfig, ...options };
    let lastError: Error | null = null;

    for (let attempt = 0; attempt <= config.maxRetries; attempt++) {
      try {
        return await fn();
      } catch (e) {
        lastError = e as Error;

        if (attempt === config.maxRetries) break;
        if (e instanceof BridgeError && !e.retryable) break;

        const delay = Math.min(
          config.baseDelay * Math.pow(config.backoffFactor, attempt),
          config.maxDelay
        );
        await this.delay(delay);
      }
    }

    throw lastError;
  }

  async withTimeout<T>(
    fn: Promise<T>,
    timeoutMs: number,
    stage: string = 'unknown'
  ): Promise<T> {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), timeoutMs);

    try {
      const result = await fn;
      clearTimeout(timeoutId);
      return result;
    } catch (e) {
      clearTimeout(timeoutId);
      if ((e as Error).name === 'AbortError') {
        throw new BridgeError('E5001', `Operation timed out after ${timeoutMs}ms`, {
          severity: 'critical',
          stage,
          retryable: true,
        });
      }
      throw e;
    }
  }

  private delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  parseError(error: any): BridgeError {
    if (error instanceof BridgeError) return error;

    if (error.code) {
      return new BridgeError(error.code, error.message || 'Unknown error', {
        severity: error.severity || 'error',
        stage: error.stage,
        retryable: error.retryable ?? true,
        details: error.details,
      });
    }

    return new BridgeError('E9999', error.message || String(error), {
      severity: 'critical',
      retryable: false,
    });
  }
}
```

> [!tip] 错误处理最佳实践
> 1. **尽早失败**：输入验证在最外层完成，避免无效计算
> 2. **保留上下文**：每个错误都应记录发生的阶段和关键参数
> 3. **优雅降级**：始终提供至少一个可用的降级方案
> 4. **可观测性**：所有错误都应结构化输出，便于监控分析

#### 12.2.1 向量化优化

将 Python 循环替换为 NumPy 向量化操作，可以获得 10-100 倍的性能提升。

```python
import numpy as np

class VectorizedBeatClipAligner:
    def __init__(
        self,
        position_weight: float = 0.4,
        energy_weight: float = 0.3,
        duration_weight: float = 0.3,
        max_beat_skip: int = 3,
    ):
        self.position_weight = position_weight
        self.energy_weight = energy_weight
        self.duration_weight = duration_weight
        self.max_beat_skip = max_beat_skip

    def _compute_cost_matrix_vectorized(
        self,
        clip_durations: np.ndarray,
        clip_energies: np.ndarray,
        beats: np.ndarray,
        beat_confidences: np.ndarray,
    ) -> np.ndarray:
        n_clips = len(clip_durations)
        n_beats = len(beats)

        beat_intervals = np.diff(beats, prepend=beats[0])

        clip_dur_2d = clip_durations.reshape(-1, 1)
        beat_int_2d = beat_intervals.reshape(1, -1)

        ratio = clip_dur_2d / np.maximum(beat_int_2d, 0.001)
        duration_score = np.where(
            (ratio >= 0.5) & (ratio <= 2.0),
            1.0 - np.abs(ratio - 1.0) / 1.0,
            0.0
        )

        clip_energy_2d = clip_energies.reshape(-1, 1)
        beat_conf_2d = beat_confidences.reshape(1, -1)
        energy_diff = np.abs(clip_energy_2d - beat_conf_2d)
        energy_score = np.maximum(0.0, 1.0 - energy_diff)

        position_score = np.ones((n_clips, n_beats))

        total_cost = 1.0 - (
            self.duration_weight * duration_score +
            self.energy_weight * energy_score +
            self.position_weight * position_score
        )

        return total_cost

    def align_dp_vectorized(
        self,
        clips: List[dict],
        beats: List[float],
        beat_confidences: List[float] = None,
    ) -> List[AlignResult]:
        if not clips or not beats:
            return []

        n_clips = len(clips)
        n_beats = len(beats)

        clip_durations = np.array([c["duration"] for c in clips])
        clip_energies = np.array([c["visual_energy"] for c in clips])
        beats_arr = np.array(beats)
        beat_confs = np.array(beat_confidences) if beat_confidences else np.ones(n_beats)

        cost_matrix = self._compute_cost_matrix_vectorized(
            clip_durations, clip_energies, beats_arr, beat_confs
        )

        dp = np.full((n_clips, n_beats), np.inf)
        prev = np.full((n_clips, n_beats), -1, dtype=int)

        dp[0, :min(n_beats, self.max_beat_skip + 1)] = cost_matrix[0, :min(n_beats, self.max_beat_skip + 1)]

        for i in range(1, n_clips):
            for j in range(n_beats):
                start_k = max(0, j - self.max_beat_skip)
                if start_k < j:
                    window = dp[i-1, start_k:j] + cost_matrix[i, j]
                    min_idx = np.argmin(window)
                    dp[i, j] = window[min_idx]
                    prev[i, j] = start_k + min_idx

        results = []
        j = np.argmin(dp[-1])
        for i in range(n_clips - 1, -1, -1):
            next_beat = beats[j + 1] if j + 1 < n_beats else None
            results.append(AlignResult(
                clip_id=clips[i].get("clip_id", f"clip_{i}"),
                beat_time=beats[j],
                beat_confidence=beat_confs[j],
                score=1.0 - dp[i, j],
                next_beat_time=next_beat,
            ))
            j = prev[i, j]
        results.reverse()

        return results
```

```python
class TTLCache:
    def __init__(self, maxsize: int = 128, ttl: float = 3600.0):
        self.maxsize = maxsize
        self.ttl = ttl
        self._cache: Dict[str, tuple[Any, float]] = {}
        self._access_order: List[str] = []

    def get(self, key: str) -> tuple[bool, Any]:
        if key not in self._cache:
            return False, None

        value, timestamp = self._cache[key]
        if time.time() - timestamp > self.ttl:
            del self._cache[key]
            self._access_order.remove(key)
            return False, None

        self._access_order.remove(key)
        self._access_order.append(key)
        return True, value

    def set(self, key: str, value: Any) -> None:
        if key in self._cache:
            self._access_order.remove(key)
        elif len(self._cache) >= self.maxsize:
            oldest = self._access_order.pop(0)
            del self._cache[oldest]

        self._cache[key] = (value, time.time())
        self._access_order.append(key)

    def clear(self) -> None:
        self._cache.clear()
        self._access_order.clear()

    def __len__(self) -> int:
        return len(self._cache)
```

```python
class IncrementalMatcher:
    def __init__(self, base_matcher: SegmentClipMatcher):
        self.base_matcher = base_matcher
        self._previous_result = None
        self._previous_segments_hash = None
        self._previous_clips_hash = None

    def _hash_data(self, data: List[dict]) -> str:
        return hashlib.md5(json.dumps(data, sort_keys=True).encode()).hexdigest()

    def match_incremental(
        self,
        segments: List[dict],
        clips: List[dict],
    ) -> List[SegmentPair]:
        seg_hash = self._hash_data(segments)
        clip_hash = self._hash_data(clips)

        if (self._previous_segments_hash == seg_hash and
            self._previous_clips_hash == clip_hash and
            self._previous_result is not None):
            return self._previous_result

        if self._previous_segments_hash == seg_hash and self._previous_result:
            new_clips = [c for c in clips if self._hash_data([c]) not in
                         {self._hash_data([pc]) for pc in
                          [c for c in self._get_prev_clips()]}]
            if new_clips and len(new_clips) < len(clips) * 0.3:
                return self._partial_rematch(segments, clips, new_clips)

        result = self.base_matcher.match(segments, clips)
        self._previous_result = result
        self._previous_segments_hash = seg_hash
        self._previous_clips_hash = clip_hash
        return result

    def _get_prev_clips(self) -> List[dict]:
        if not self._previous_result:
            return []
        return [{"clip_id": p.clip_id} for p in self._previous_result]

    def _partial_rematch(
        self,
        segments: List[dict],
        clips: List[dict],
        new_clips: List[dict],
    ) -> List[SegmentPair]:
        existing_pairs = self._previous_result or []
        return self.base_matcher.match(segments, clips)
```

```python
def optimize_array_memory(arr: np.ndarray, target_dtype: str = None) -> np.ndarray:
    if target_dtype:
        return arr.astype(target_dtype)

    if np.issubdtype(arr.dtype, np.floating):
        if arr.min() >= 0 and arr.max() <= 1:
            return arr.astype(np.float16)
        return arr.astype(np.float32)

    if np.issubdtype(arr.dtype, np.integer):
        if arr.min() >= 0:
            if arr.max() < 256:
                return arr.astype(np.uint8)
            if arr.max() < 65536:
                return arr.astype(np.uint16)
        else:
            if arr.min() > -128 and arr.max() < 127:
                return arr.astype(np.int8)

    return arr

class MemoryOptimizedAligner(BeatClipAligner):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def align_low_memory(
        self,
        clips: List[dict],
        beats: List[float],
        beat_confidences: List[float] = None,
    ) -> List[AlignResult]:
        n_clips = len(clips)
        n_beats = len(beats)

        if beat_confidences is None:
            beat_confidences = [1.0] * n_beats

        dp_prev = np.full(n_beats, np.inf, dtype=np.float32)
        dp_curr = np.full(n_beats, np.inf, dtype=np.float32)

        prev_indices = np.full((n_clips, n_beats), -1, dtype=np.int16)

        cost_row = np.zeros(n_beats, dtype=np.float32)
        dp_prev[:min(n_beats, self.max_beat_skip + 1)] = self._compute_row_cost(
            clips[0], beats, beat_confidences, min(n_beats, self.max_beat_skip + 1)
        )

        for i in range(1, n_clips):
            dp_curr[:] = np.inf
            clip = clips[i]

            for j in range(n_beats):
                cost = self._compute_single_cost(clip, beats, beat_confidences, j)
                start_k = max(0, j - self.max_beat_skip)

                if start_k < j:
                    min_val = dp_prev[start_k:j].min()
                    min_idx = start_k + dp_prev[start_k:j].argmin()
                    dp_curr[j] = min_val + cost
                    prev_indices[i, j] = min_idx

            dp_prev[:] = dp_curr

        results = []
        j = int(np.argmin(dp_prev))
        for i in range(n_clips - 1, -1, -1):
            next_beat = beats[j + 1] if j + 1 < n_beats else None
            results.append(AlignResult(
                clip_id=clips[i].get("clip_id", f"clip_{i}"),
                beat_time=beats[j],
                beat_confidence=beat_confidences[j],
                score=float(1.0 - dp_prev[j]),
                next_beat_time=next_beat,
            ))
            if i > 0:
                j = int(prev_indices[i, j])

        results.reverse()
        return results

    def _compute_row_cost(self, clip, beats, confidences, n):
        row = np.zeros(n, dtype=np.float32)
        for j in range(n):
            row[j] = self._compute_single_cost(clip, beats, confidences, j)
        return row

    def _compute_single_cost(self, clip, beats, confidences, j):
        return 0.5
```

### 13.4 准确率测试（vs 人工标注）

```python
import numpy as np
from typing import List, Dict, Tuple

class AccuracyBenchmark:
    def __init__(self):
        self.ground_truth: List[Dict] = []

    def load_ground_truth(self, annotations: List[Dict]):
        self.ground_truth = annotations

    def beat_alignment_accuracy(
        self,
        predicted: List[AlignResult],
        ground_truth: List[Dict],
        tolerance_ms: float = 50.0,
    ) -> Dict[str, float]:
        pred_times = [r.beat_time for r in predicted]
        gt_times = [g["beat_time"] for g in ground_truth]

        if len(pred_times) == 0 or len(gt_times) == 0:
            return {"precision": 0, "recall": 0, "f1": 0, "mean_offset_ms": 0}

        tolerance = tolerance_ms / 1000.0
        matches = 0
        used_gt = set()
        offsets = []

        for p_time in pred_times:
            best_gt = None
            best_diff = float('inf')
            for j, g_time in enumerate(gt_times):
                if j in used_gt:
                    continue
                diff = abs(p_time - g_time)
                if diff < best_diff and diff <= tolerance:
                    best_diff = diff
                    best_gt = j
            if best_gt is not None:
                matches += 1
                used_gt.add(best_gt)
                offsets.append(best_diff * 1000)

        precision = matches / len(pred_times)
        recall = matches / len(gt_times)
        f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0
        mean_offset = np.mean(offsets) if offsets else 0

        return {
            "precision": precision,
            "recall": recall,
            "f1": f1,
            "mean_offset_ms": mean_offset,
            "n_predictions": len(pred_times),
            "n_groundtruth": len(gt_times),
        }

    def segment_pairing_accuracy(
        self,
        predicted: List[SegmentPair],
        ground_truth: List[Dict],
    ) -> Dict[str, float]:
        pred_pairs = {(p.segment_label, p.clip_id) for p in predicted}
        gt_pairs = {(g["segment_label"], g["clip_id"]) for g in ground_truth}

        correct = len(pred_pairs & gt_pairs)
        precision = correct / len(pred_pairs) if pred_pairs else 0
        recall = correct / len(gt_pairs) if gt_pairs else 0
        f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0

        return {
            "precision": precision,
            "recall": recall,
            "f1": f1,
            "n_predictions": len(pred_pairs),
            "n_groundtruth": len(gt_pairs),
            "correct": correct,
        }
```

```python
class IntegratedEditEngine:
    def __init__(self, config: dict = None):
        self.config = config or {}
        self._init_phase_engines()

    def _init_phase_engines(self):
        self.phase1_music_analyzer = None
        self.phase2_clip_analyzer = None
        self.phase3_match_engine = None
        self.phase4_param_inferrer = None
        self.phase5_compiler = None

    def full_pipeline(self, request: dict) -> dict:
        music_atom = self._phase1_analyze_music(request["audio"])
        clip_atoms = self._phase2_analyze_clips(request["clips"])
        match_plan = self._phase3_match(music_atom, clip_atoms, request.get("style"))
        compiler_input = self._phase4_infer_params(match_plan)
        result = self._phase5_compile(compiler_input, request.get("output"))
        return result

    def _phase1_analyze_music(self, audio_input: dict) -> dict:
        if self.phase1_music_analyzer:
            return self.phase1_music_analyzer.analyze(audio_input)
        return self._mock_music_atom(audio_input)

    def _phase2_analyze_clips(self, clips_input: list) -> list:
        if self.phase2_clip_analyzer:
            return [self.phase2_clip_analyzer.analyze(c) for c in clips_input]
        return [self._mock_clip_atom(c) for c in clips_input]

    def _phase3_match(self, music_atom: dict, clip_atoms: list, style: str) -> dict:
        aligner = BeatClipAligner()
        matcher = SegmentClipMatcher()
        binder = EffectBandBinder(style_preset=style or "cinematic")
        transition_gen = TransitionGenerator()
        planner = NarrativePlanner()

        beats = music_atom.get("beats", [])
        beat_times = [b["time"] if isinstance(b, dict) else b for b in beats]
        beat_confs = [b.get("confidence", 1.0) if isinstance(b, dict) else 1.0 for b in beats]
        segments = music_atom.get("segments", [])
        freq_bands = music_atom.get("freq_bands", [])

        align_results = aligner.align_dp(clip_atoms, beat_times, beat_confs)
        segment_pairs = matcher.match(segments, clip_atoms)
        clip_starts = [r.beat_time for r in align_results]
        effect_bindings = binder.bind_all_clips(clip_atoms, clip_starts, freq_bands, beat_times)
        transitions = transition_gen.generate(
            segments, segment_pairs, music_atom.get("bpm", 120.0)
        )
        narrative = planner.plan(music_atom, clip_atoms, align_results, style)

        match_plan = MatchPlan(
            music_id=music_atom.get("id", "unknown"),
            clip_ids=[c.get("clip_id", f"clip_{i}") for i, c in enumerate(clip_atoms)],
            beat_alignments=[{
                "clip_id": r.clip_id,
                "beat_time": r.beat_time,
                "score": r.score,
            } for r in align_results],
            segment_pairs=segment_pairs,
            effect_bindings=effect_bindings,
            transitions=transitions,
            narrative=narrative,
            style_preset=style or "cinematic",
            overall_score=sum(r.score for r in align_results) / max(len(align_results), 1),
        )

        return match_plan.to_dict()

    def _phase4_infer_params(self, match_plan: dict) -> dict:
        inferrer = ParameterInferrer()
        return inferrer.generate_compiler_input(match_plan)

    def _phase5_compile(self, compiler_input: dict, output_config: dict) -> dict:
        return {
            "status": "success",
            "compiler_input": compiler_input,
            "output_config": output_config,
            "project_file": "output.aep",
        }

    def _mock_music_atom(self, audio_input: dict) -> dict:
        return {
            "id": audio_input.get("id", "music"),
            "duration": audio_input.get("duration", 180.0),
            "bpm": 120.0,
            "beats": [],
            "segments": [],
            "freq_bands": [],
        }

    def _mock_clip_atom(self, clip_input: dict) -> dict:
        return {
            "clip_id": clip_input.get("id", "clip"),
            "duration": clip_input.get("duration", 5.0),
            "visual_energy": 0.5,
        }
```

```python
class MatchEngineConfig(BaseModel):
    server: ServerConfig = Field(default_factory=ServerConfig)
    timeouts: TimeoutConfigDTO = Field(default_factory=TimeoutConfigDTO)
    cache: CacheConfig = Field(default_factory=CacheConfig)
    default_style: str = "cinematic"
    max_clips_per_request: int = 100
    min_clips_per_request: int = 1

    @classmethod
    def from_env(cls) -> "MatchEngineConfig":
        config = cls()
        for field_name in cls.model_fields:
            env_var = f"MATCH_ENGINE_{field_name.upper()}"
            if env_var in os.environ:
                setattr(config, field_name, os.environ[env_var])
        return config

    @classmethod
    def from_file(cls, path: str) -> "MatchEngineConfig":
        with open(path, 'r') as f:
            data = json.load(f)
        return cls(**data)

    def to_dict(self) -> dict:
        return self.model_dump()
```

#### 14.5.1 结构化日志

```python
import logging
import json
import time
from typing import Any, Dict

class StructuredLogger:
    def __init__(self, name: str, level: str = "INFO"):
        self.logger = logging.getLogger(name)
        self.logger.setLevel(getattr(logging, level.upper()))

        if not self.logger.handlers:
            handler = logging.StreamHandler()
            formatter = JsonFormatter()
            handler.setFormatter(formatter)
            self.logger.addHandler(handler)

    def _log(self, level: str, message: str, **kwargs):
        extra = {
            "timestamp": time.time(),
            "level": level,
            "message": message,
        }
        extra.update(kwargs)
        getattr(self.logger, level.lower())(json.dumps(extra))

    def info(self, message: str, **kwargs):
        self._log("INFO", message, **kwargs)

    def warning(self, message: str, **kwargs):
        self._log("WARNING", message, **kwargs)

    def error(self, message: str, **kwargs):
        self._log("ERROR", message, **kwargs)

    def debug(self, message: str, **kwargs):
        self._log("DEBUG", message, **kwargs)

class JsonFormatter(logging.Formatter):
    def format(self, record):
        try:
            return record.getMessage()
        except Exception:
            return json.dumps({"message": str(record.msg)})
```

#### 14.5.2 性能指标收集

```python
import time
from collections import defaultdict
from typing import Dict, List

class MetricsCollector:
    def __init__(self):
        self._counters: Dict[str, int] = defaultdict(int)
        self._histograms: Dict[str, List[float]] = defaultdict(list)
        self._gauges: Dict[str, float] = {}

    def increment(self, name: str, value: int = 1, labels: dict = None):
        key = self._make_key(name, labels)
        self._counters[key] += value

    def observe(self, name: str, value: float, labels: dict = None):
        key = self._make_key(name, labels)
        self._histograms[key].append(value)
        if len(self._histograms[key]) > 10000:
            self._histograms[key] = self._histograms[key][-5000:]

    def set_gauge(self, name: str, value: float, labels: dict = None):
        key = self._make_key(name, labels)
        self._gauges[key] = value

    def _make_key(self, name: str, labels: dict = None) -> str:
        if not labels:
            return name
        label_str = ",".join(f"{k}={v}" for k, v in sorted(labels.items()))
        return f"{name}[{label_str}]"

    def get_summary(self) -> dict:
        histo_summary = {}
        for name, values in self._histograms.items():
            if values:
                sorted_vals = sorted(values)
                histo_summary[name] = {
                    "count": len(values),
                    "mean": sum(values) / len(values),
                    "p50": sorted_vals[len(sorted_vals) // 2],
                    "p95": sorted_vals[int(len(sorted_vals) * 0.95)],
                    "p99": sorted_vals[int(len(sorted_vals) * 0.99)],
                    "min": min(values),
                    "max": max(values),
                }

        return {
            "counters": dict(self._counters),
            "histograms": histo_summary,
            "gauges": dict(self._gauges),
        }

metrics = MetricsCollector()

def timed_metric(metric_name: str):
    def decorator(fn):
        def wrapper(*args, **kwargs):
            start = time.time()
            try:
                result = fn(*args, **kwargs)
                metrics.increment(f"{metric_name}_success")
                return result
            except Exception as e:
                metrics.increment(f"{metric_name}_errors", labels={"error": type(e).__name__})
                raise
            finally:
                duration = time.time() - start
                metrics.observe(f"{metric_name}_duration", duration)
        return wrapper
    return decorator
```
