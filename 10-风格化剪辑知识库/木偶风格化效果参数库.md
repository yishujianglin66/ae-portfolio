## 二、木质木偶风格参数详解

### 2.1 核心参数配置

#### 调色参数
```jsx
// 木质风格调色配置
function applyWoodenColorGrade(comp, layer) {
    // 1. 曲线调整
    var curves = layer.Effects.addProperty("ADBE Curves");
    curves.property("ADBE Curves Channel").setValue("RGB");
    curves.property("ADBE Curves Curve").setValue([
        [0, 0], [30, 20], [60, 50], [100, 85],
        [150, 130], [200, 180], [255, 255]
    ]);
    
    // 红色通道 - 增加暖色调
    curves.property("ADBE Curves Channel").setValue("Red");
    curves.property("ADBE Curves Curve").setValue([
        [0, 0], [50, 55], [128, 140], [200, 215], [255, 255]
    ]);
    
    // 绿色通道 - 轻微降低
    curves.property("ADBE Curves Channel").setValue("Green");
    curves.property("ADBE Curves Curve").setValue([
        [0, 0], [50, 45], [128, 120], [200, 190], [255, 255]
    ]);
    
    // 2. 色相饱和度
    var hueSat = layer.Effects.addProperty("ADBE HSL2");
    hueSat.property("ADBE HSL Saturation").setValue(10);
    hueSat.property("ADBE HSL Lightness").setValue(-5);
    
    // 3. 色阶
    var levels = layer.Effects.addProperty("ADBE Levels");
    levels.property("ADBE Levels Input White").setValue(230);
    levels.property("ADBE Levels Gamma").setValue(1.1);
}
```

function applyWoodenTexture(comp, layer) {
    // 木纹纹理叠加

#### 关节接缝参数
```jsx
function createWoodenJoints(comp, layer) {
    // 创建木榫关节效果
    var jointPositions = {
        "neck": [0.5, 0.28],
        "left_shoulder": [0.32, 0.35],
        "right_shoulder": [0.68, 0.35],
        "left_elbow": [0.22, 0.5],
        "right_elbow": [0.78, 0.5],
        "waist": [0.5, 0.55],
        "left_hip": [0.4, 0.58],
        "right_hip": [0.6, 0.58],
        "left_knee": [0.38, 0.72],
        "right_knee": [0.62, 0.72]
    };
    
    var jointLayer = comp.layers.addShape();
    jointLayer.name = "Wooden_Joints";
    jointLayer.parent = layer;
    
    var contents = jointLayer.property("Contents");
    
    for (var jointName in jointPositions) {
        var pos = jointPositions[jointName];
        var group = contents.addProperty("ADBE Vector Group");
        group.name = "Joint_" + jointName;
        
        // 木榫外圈
        var ellipse = group.property("Contents").addProperty("ADBE Vector Shape - Ellipse");
        ellipse.property("ADBE Ellipse Path Size").setValue([25, 20]);
        ellipse.property("ADBE Ellipse Path Position").setValue([
            pos[0] * comp.width,
            pos[1] * comp.height
        ]);
        
        var stroke = group.property("Contents").addProperty("ADBE Vector Graphic - Stroke");
        stroke.property("ADBE Stroke Color").setValue([0.15, 0.08, 0.02]);
        stroke.property("ADBE Stroke Width").setValue(3);
        
        var fill = group.property("Contents").addProperty("ADBE Vector Graphic - Fill");
        fill.property("ADBE Fill Color").setValue([0.25, 0.15, 0.05]);
        
        // 木榫内圈
        var innerGroup = group.property("Contents").addProperty("ADBE Vector Group");
        var innerEllipse = innerGroup.property("Contents").addProperty("ADBE Vector Shape - Ellipse");
        innerEllipse.property("ADBE Ellipse Path Size").setValue([15, 12]);
        innerEllipse.property("ADBE Ellipse Path Position").setValue([
            pos[0] * comp.width,
            pos[1] * comp.height
        ]);
        
        var innerFill = innerGroup.property("Contents").addProperty("ADBE Vector Graphic - Fill");
        innerFill.property("ADBE Fill Color").setValue([0.4, 0.25, 0.1]);
        
        // 高光
        var highlight = group.property("Contents").addProperty("ADBE Vector Group");
        var highlightEllipse = highlight.property("Contents").addProperty("ADBE Vector Shape - Ellipse");
        highlightEllipse.property("ADBE Ellipse Path Size").setValue([8, 5]);
        highlightEllipse.property("ADBE Ellipse Path Position").setValue([
            pos[0] * comp.width - 5,
            pos[1] * comp.height - 5
        ]);
        var highlightFill = highlight.property("Contents").addProperty("ADBE Vector Graphic - Fill");
        highlightFill.property("ADBE Fill Color").setValue([0.8, 0.65, 0.4]);
    }
}
```

### 2.2 完整风格脚本
```jsx
function applyWoodenPuppetStyle(comp, layer) {
    // Phase 1: 调色
    applyWoodenColorGrade(comp, layer);
    
    // Phase 2: 材质纹理
    applyWoodenTexture(comp, layer);
    
    // Phase 3: 关节接缝
    createWoodenJoints(comp, layer);
    
    // Phase 4: 定格动画效果
    var stopMotion = comp.layers.addSolid([1,1,1], "StopMotion_Wooden", comp.width, comp.height, 1);
    stopMotion.adjustmentLayer = true;
    
    var fps = comp.frameRate;
    var targetFps = 12;
    layer.property("Time Remap").expression = 
        "frame = Math.floor(time * " + fps + ") / " + fps + ";" +
        "stepped = Math.floor(frame * " + targetFps + ") / " + targetFps + ";" +
        "stepped;";
    
    var flicker = stopMotion.Effects.addProperty("ADBE Brightness & Contrast");
    flicker.property("ADBE Brightness-Contrast Brightness").expression = "random(-3, 3)";
    
    // Phase 5: 微缩场景效果
    var tiltShift = comp.layers.addSolid([1,1,1], "TiltShift_Wooden", comp.width, comp.height, 1);
    tiltShift.adjustmentLayer = true;
    var vignette = tiltShift.Effects.addProperty("ADBE Vignette");
    vignette.property("ADBE Vignette Amount").setValue(25);
}
```

#### 缝线接缝参数

function createClothStitches(comp, layer) {
    var stitchPositions = {
        "neck": [0.5, 0.28],
        "left_shoulder": [0.32, 0.35],
        "right_shoulder": [0.68, 0.35],
        "waist": [0.5, 0.55],
        "left_hip": [0.4, 0.58],
        "right_hip": [0.6, 0.58]
    };
    
    var stitchLayer = comp.layers.addShape();
    stitchLayer.name = "Cloth_Stitches";
    stitchLayer.parent = layer;
    
    var contents = stitchLayer.property("Contents");
    
    for (var posName in stitchPositions) {
        var pos = stitchPositions[posName];
        var group = contents.addProperty("ADBE Vector Group");
        
        // 缝线圆点
        for (var i = 0; i < 8; i++) {
            var dot = group.property("Contents").addProperty("ADBE Vector Shape - Ellipse");
            var angle = (i / 8) * Math.PI * 2;
            var radius = 12;
            dot.property("ADBE Ellipse Path Size").setValue([4, 4]);
            dot.property("ADBE Ellipse Path Position").setValue([
                pos[0] * comp.width + Math.cos(angle) * radius,
                pos[1] * comp.height + Math.sin(angle) * radius
            ]);
            
            var fill = group.property("Contents").addProperty("ADBE Vector Graphic - Fill");
            fill.property("ADBE Fill Color").setValue([0.3, 0.3, 0.3]);
        }
    }
}

        return self.presets.get(name)

        return [

                "display_name": p.display_name,
                "category": p.category,
                "description": p.description
            }
            for p in self.presets.values()
