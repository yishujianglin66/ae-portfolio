### 1.2 音效原子参数

每个音效原子具有以下参数：

```yaml
sound_atom:
  id: "SFX-01-001"
  name: "whoosh"
  category: "transition"
  file: "sfx/whoosh_01.wav"
  
  # 时长
  duration: 0.5
  
  # 音量包络
  envelope:
    attack: 0.01     # 起音
    decay: 0.1       # 衰减
    sustain: 0.8     # 持续
    release: 0.2     # 释放
  
  # 频率特征
  frequency:
    low_freq: 200    # 最低频率
    high_freq: 8000  # 最高频率
    dominant: 2000   # 主频
  
  # 能量特征
  energy:
    peak: 0.9        # 峰值能量
    rms: 0.5         # RMS能量
    dynamic_range: 12
  
  # 情绪标签
  mood: ["energetic", "dynamic"]
  
  # 视觉对应
  visual_association: ["motion", "transition", "speed"]
```

### 1.3 音效-音乐层级

```
音频混合层级（Audio Mix Hierarchy）
├── L1 主音乐（Music Bed）
│   └── 原始BGM，-3dB
│
├── L2 音效层（SFX Layer）
│   ├── 转场音效（-6dB）
│   ├── 卡点音效（-10dB）
│   └── 氛围音效（-18dB）
│
├── L3 环境音层（Ambience Layer）
│   └── 底噪/环境声（-20dB）
│
└── L4 特效音层（Designer Layer）
    └── 设计音效/特殊效果（-8dB）
```

## 第四章 音效时间轴自动生成

### 4.1 生成算法

```python
def generate_sound_timeline(music_atom, match_plan, clip_atoms):
    """
    根据音乐原子 + 匹配方案 + 片段原子，生成音效时间轴
    """
    sound_events = []
    
    # 1. 转场音效
    for transition in match_plan["transition_plan"]:
        sfx_type = transition_to_sfx[transition["type"]]
        sound_events.append({
            "time": transition["time"],
            "type": "transition",
            "sfx": sfx_type,
            "volume": transition["energy"] * 0.5 + -6,
            "duration": transition.get("duration", 0.5),
            "sync_point": "start"
        })
    
    # 2. 节拍卡点音效
    for beat in music_atom["rhythm"]["beat_grid"]:
        beat_index = get_beat_index(beat, music_atom)
        beat_type = classify_beat_type(beat_index)
        sfx_info = beat_to_sfx_map[beat_type]
        
        # 只在有视觉切镜的节拍加音效
        if has_visual_cut_at(beat, match_plan):
            sound_events.append({
                "time": beat,
                "type": "beat",
                "sfx": sfx_info["sound"],
                "volume": sfx_info["volume"],
                "duration": get_sfx_duration(sfx_info["sound"]),
                "sync_point": "peak"
            })
    
    # 3. 段落氛围音效
    for segment in music_atom["structure"]["segments"]:
        segment_rules = segment_sound_rules.get(segment["type"], {})
        for amb_sound in segment_rules.get("ambience", []):
            sound_events.append({
                "time": segment["start"],
                "type": "ambience",
                "sfx": amb_sound,
                "volume": segment_rules.get("ambience_volume", -18),
                "duration": segment["end"] - segment["start"],
                "loop": True,
                "fade_in": 0.5,
                "fade_out": 1.0
            })
    
    # 4. 效果触发音效
    for binding in match_plan["effect_binding"]:
        if binding["effect"] in visual_to_sound_map:
            sound_events.append({
                "time": binding["time_range"][0],
                "type": "effect",
                "sfx": visual_to_sound_map[binding["effect"]]["sound"],
                "volume": visual_to_sound_map[binding["effect"]]["volume"],
                "duration": binding["time_range"][1] - binding["time_range"][0],
                "sync_point": "peak"
            })
    
    # 5. 按时间排序
    sound_events.sort(key=lambda e: e["time"])
    
    return sound_events
```

### 5.2 AE 脚本生成音效关键帧

```typescript
// 音效驱动的视觉关键帧生成
interface SoundDrivenKeyframe {
  time: number;
  propertyPath: string;
  value: number | number[];
  easing: string;
  soundTrigger: string; // 触发的音效类型
}

function generateSoundDrivenKeyframes(
  soundTimeline: SoundTimeline,
  effectRef: string
): SoundDrivenKeyframe[] {
  const keyframes: SoundDrivenKeyframe[] = [];
  
  for (const event of soundTimeline.events) {
    if (event.type === "beat") {
      // 节拍触发缩放脉冲
      keyframes.push({
        time: event.time,
        propertyPath: `Effects/${effectRef}/Scale`,
        value: [105, 105],
        easing: "ease_out",
        soundTrigger: event.sfx
      });
      // 衰减回原位
      keyframes.push({
        time: event.time + 0.1,
        propertyPath: `Effects/${effectRef}/Scale`,
        value: [100, 100],
        easing: "ease_in",
        soundTrigger: event.sfx
      });
    }
    
    if (event.type === "effect" && event.sfx.includes("impact")) {
      // 冲击触发强烈抖动
      keyframes.push({
        time: event.time,
        propertyPath: `Effects/${effectRef}/Position`,
        value: [960 + 10, 540 + 5],
        easing: "hold",
        soundTrigger: "impact"
      });
    }
  }
  
  return keyframes;
}
```

```yaml
    tool: "python (librosa + soundfile)"
    purpose: "生成音高/时长变体，避免重复"
    method: "pitch_shift ±2 semitones, time_stretch 0.9-1.1x"
  
  step_4_concat_mix:
    tool: "ffmpeg (amix)"
    command: "ffmpeg -i music.wav -i sfx.wav -filter_complex amix=inputs=2:duration=longest output.wav"
    purpose: "音乐+音效混音"
```

## 第七章 QC 标准

### 7.1 音效质量标准

| 指标 | 标准 |
|------|------|
| 采样率 | 48kHz |
| 位深度 | 24-bit |
| 格式 | WAV |
| 峰值电平 | -1dBTP（不削波） |
| 集成响度 | -16 LUFS（短视频） |
| 响度范围 | < 10 LU |
| 声道 | 立体声 |

### 7.2 音画同步验收

| 指标 | 标准 |
|------|------|
| 视觉-音效延迟 | < 50ms（不可察觉） |
| 节拍同步精度 | < 20ms |
| 冲击效果同步 | < 1 帧 |
| 转场音效时长匹配 | ±10% 误差 |
| 整体混音平衡 | 音乐不被音效掩盖 |
