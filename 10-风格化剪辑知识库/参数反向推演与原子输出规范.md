# 参数反向推演与原子输出规范

> [!abstract] 文档摘要
> 本文档定义从 MatchPlan → CompilerInput 的参数反向推演规范。给定目标效果（如"Glow 强度0.8"），反推 AE 原子参数（matchName="ADBE Glo2", Glow Intensity=4, Glow Threshold=50, ...）。包括效果参数映射、关键帧模板库、置信度评估、与既有 Phase 1-5 引擎的接口对齐。

> [!tip] 使用指南
> - 参数推演开发者：第二至五章是核心
> - 编译器对接者：第六章是与 Phase 1 的接口
> - 案例参考者：第七章是常用反推模板

## 第二章 效果识别

### 2.1 自然语言效果名 → matchName

调用既有 [[参数-效果原子级映射库]]：

```python
from effect_mapping import find_by_display_name

def resolve_effect_match_name(effect_name):
    """
    把效果描述转换为 AE matchName
    """
    # 1. 直接查找
    entry = find_by_display_name(effect_name)
    if entry:
        return {
            "matchName": entry.matchName,
            "displayName": entry.displayName,
            "category": entry.category,
            "paramMap": entry.paramMap,
            "confidence": 1.0
        }
    
    # 2. 模糊匹配
    candidates = fuzzy_match(effect_name)
    if candidates:
        best = candidates[0]
        return {
            "matchName": best.matchName,
            "displayName": best.displayName,
            "confidence": 0.7
        }
    
    return None
```

## 第三章 参数反推

### 3.1 强度→具体参数值

每个效果的"强度"是 0-1 的归一化值，需要反推为具体参数：

```python
INTENSITY_TO_PARAMS = {
    "ADBE Glo2": {  # Glow
        "intensity_to_params": {
            "Glow Intensity": lambda i: i * 5,         # 0-5
            "Glow Threshold": lambda i: 60 - i * 20,   # 60→40
            "Glow Radius": lambda i: i * 30,            # 0-30
            "Glow Color": lambda i: [1, 1, 1]          # 默认白色
        },
        "value_ranges": {
            "Glow Intensity": [0, 5],
            "Glow Threshold": [0, 100],
            "Glow Radius": [0, 100]
        }
    },
    "ADBE Gaussian Blur 2": {  # Gaussian Blur
        "intensity_to_params": {
            "Blurriness": lambda i: i * 50,             # 0-50
            "Blur Dimensions": lambda i: 1              # 1=All
        }
    },
    "ADBE DisplacementMap": {  # Glitch / Displacement
        "intensity_to_params": {
            "Max Horizontal Displacement": lambda i: i * 50,
            "Max Vertical Displacement": lambda i: i * 20,
            "Displacement Map Layer": lambda i: "Precomp_RGB_Split"
        }
    },
    "Trapcode Particular": {
        "intensity_to_params": {
            "Particles/sec": lambda i: i * 500,
            "Size": lambda i: i * 3,
            "Opacity": lambda i: i * 100
        }
    },
    "Optical Flares": {
        "intensity_to_params": {
            "Brightness": lambda i: i * 200,
            "Scale": lambda i: 50 + i * 100,
            "Color": lambda i: [1, 1, 0.8]
        }
    }
}

def infer_params(match_name, intensity, additional_constraints=None):
    """
    强度 → 具体参数值
    """
    config = INTENSITY_TO_PARAMS.get(match_name)
    if not config:
        return None
    
    params = {}
    for param_name, formula in config["intensity_to_params"].items():
        value = formula(intensity)
        
        # 应用附加约束（如来自片段的色彩/亮度）
        if additional_constraints and param_name in additional_constraints:
            value = additional_constraints[param_name]
        
        # 限制到有效范围
        if param_name in config.get("value_ranges", {}):
            min_v, max_v = config["value_ranges"][param_name]
            value = max(min_v, min(max_v, value))
        
        params[param_name] = value
    
    return params
```

### 3.2 段落上下文修正

不同段落对参数有修正：

```python
SEGMENT_CONTEXT_MODIFIER = {
    "intro": {"intensity_multiplier": 0.5},   # 前奏效果更柔和
    "verse": {"intensity_multiplier": 0.7},
    "chorus": {"intensity_multiplier": 1.0},
    "bridge": {"intensity_multiplier": 0.8},
    "drop": {"intensity_multiplier": 1.2},    # 高潮更激进
    "outro": {"intensity_multiplier": 0.6}
}

def apply_segment_modifier(params, segment_type):
    modifier = SEGMENT_CONTEXT_MODIFIER.get(segment_type, {})
    multiplier = modifier.get("intensity_multiplier", 1.0)
    
    # 只对强度类参数做修正
    intensity_params = ["Glow Intensity", "Blurriness", "Brightness", "Particles/sec"]
    adjusted = params.copy()
    for p in intensity_params:
        if p in adjusted and isinstance(adjusted[p], (int, float)):
            adjusted[p] *= multiplier
    
    return adjusted
```

### 3.3 片段上下文修正

```python
def apply_clip_context(params, clip_atom):
    """
    根据片段特征调整参数
    """
    # 高饱和度片段 → 减少 Glow 强度（避免过曝）
    if clip_atom["color"]["saturation"] > 0.7:
        if "Glow Intensity" in params:
            params["Glow Intensity"] *= 0.8
    
    # 高对比度片段 → 增加 Glitch 位移
    if clip_atom["color"]["contrast"] > 0.7:
        if "Max Horizontal Displacement" in params:
            params["Max Horizontal Displacement"] *= 1.2
    
    # 暗色片段 → 增加 Glow 半径
    if clip_atom["color"]["brightness"] < 0.3:
        if "Glow Radius" in params:
            params["Glow Radius"] *= 1.3
    
    return params
```

## 第四章 关键帧模板库

### 4.1 节拍驱动的关键帧

```python
def generate_beat_driven_keyframes(beat_times, param_name, base_value, peak_value, easing="ease_in_out"):
    """
    生成节拍驱动的关键帧
    每个节拍点一个关键帧，在峰值和基线之间交替
    """
    keyframes = []
    for i, beat_time in enumerate(beat_times):
        # 强拍用峰值，弱拍用基线
        is_downbeat = (i % 4 == 0)
        value = peak_value if is_downbeat else base_value
        
        keyframes.append({
            "time": beat_time,
            "value": value,
            "easing": {"type": easing}
        })
    
    return keyframes
```

### 4.2 段落驱动的关键帧

```python
def generate_segment_keyframes(segments, param_name, value_curve):
    """
    生成段落驱动的关键帧
    每个段落一个关键帧
    """
    keyframes = []
    for i, seg in enumerate(segments):
        keyframes.append({
            "time": seg["start"],
            "value": value_curve[i],
            "easing": {"type": "ease_in_out", "bezier": [0.42, 0, 0.58, 1]}
        })
    
    return keyframes
```

### 4.3 峰值驱动的关键帧

```python
def generate_peak_keyframes(energy_peaks, param_name, base_value, peak_value, peak_duration=0.3):
    """
    生成峰值驱动的关键帧
    每个能量峰值前后插入关键帧，形成"爆发-恢复"曲线
    """
    keyframes = []
    
    for peak in energy_peaks:
        peak_time = peak["time"]
        peak_intensity = peak["intensity"]
        
        # 峰值前的"蓄势"关键帧
        keyframes.append({
            "time": peak_time - 0.1,
            "value": base_value,
            "easing": {"type": "ease_in"}
        })
        
        # 峰值"爆发"关键帧
        keyframes.append({
            "time": peak_time,
            "value": peak_value * peak_intensity,
            "easing": {"type": "ease_out"}
        })
        
        # 峰值后的"恢复"关键帧
        keyframes.append({
            "time": peak_time + peak_duration,
            "value": base_value,
            "easing": {"type": "ease_in_out"}
        })
    
    # 按时间排序
    keyframes.sort(key=lambda k: k["time"])
    return keyframes
```

## 第五章 转场操作生成

### 5.1 转场→操作序列

```python
def generate_transition_ops(transition, time, layer_refs):
    """
    把转场转换为具体的 AE 操作序列
    """
    transition_type = transition["type"]
    duration = transition.get("duration", 0.5)
    
    if transition_type == "hard_cut":
        # 硬切：直接切换图层可见性
        return [
            {"op": "setProperty", "ref": f"op_cut_{int(time*100)}",
             "layerRef": layer_refs[0], "propertyPath": "Opacity",
             "value": 100, "time": time},
            {"op": "setProperty", "ref": f"op_cut_{int(time*100)+1}",
             "layerRef": layer_refs[1], "propertyPath": "Opacity",
             "value": 0, "time": time - 0.01},
            {"op": "setProperty", "ref": f"op_cut_{int(time*100)+2}",
             "layerRef": layer_refs[1], "propertyPath": "Opacity",
             "value": 100, "time": time}
        ]
    
    elif transition_type == "flash":
        # 闪光转场：白色调整层快速闪现
        return [
            {"op": "addLayer", "ref": f"flash_layer_{int(time*100)}",
             "layerType": "adjustment", "color": [1, 1, 1]},
            {"op": "setKeyframe", "ref": f"flash_kf1_{int(time*100)}",
             "layerRef": f"flash_layer_{int(time*100)}", "propertyPath": "Opacity",
             "keyframes": [
                 {"time": time, "value": 0, "easing": {"type": "linear"}},
                 {"time": time + duration/2, "value": 100, "easing": {"type": "ease_in"}},
                 {"time": time + duration, "value": 0, "easing": {"type": "ease_out"}}
             ]}
        ]
    
    elif transition_type == "glitch_switch":
        # 故障切换：DisplacementMap + RGB Split + 快速切换
        return [
            {"op": "addEffect", "ref": f"glitch_fx_{int(time*100)}",
             "matchName": "ADBE DisplacementMap", "effectName": "Displacement Map"},
            {"op": "setKeyframe", "ref": f"glitch_kf1_{int(time*100)}",
             "propertyPath": "Effects/glitch_fx/Max Horizontal Displacement",
             "keyframes": [
                 {"time": time, "value": 0, "easing": {"type": "linear"}},
                 {"time": time + duration/2, "value": 50, "easing": {"type": "ease_in_out"}},
                 {"time": time + duration, "value": 0, "easing": {"type": "ease_out"}}
             ]}
        ]
    
    elif transition_type == "dissolve":
        # 溶解：交叉淡入淡出
        return [
            {"op": "setKeyframe", "ref": f"dissolve_out_{int(time*100)}",
             "layerRef": layer_refs[0], "propertyPath": "Opacity",
             "keyframes": [
                 {"time": time, "value": 100, "easing": {"type": "ease_in_out"}},
                 {"time": time + duration, "value": 0, "easing": {"type": "ease_in_out"}}
             ]},
            {"op": "setKeyframe", "ref": f"dissolve_in_{int(time*100)}",
             "layerRef": layer_refs[1], "propertyPath": "Opacity",
             "keyframes": [
                 {"time": time, "value": 0, "easing": {"type": "ease_in_out"}},
                 {"time": time + duration, "value": 100, "easing": {"type": "ease_in_out"}}
             ]}
        ]
    
    # ... 其他转场
```

## 第六章 与 Phase 1 编译器的接口

### 6.1 CompilerInput 输出

完整对齐既有 Phase 1 编译器接口：

```typescript
interface CompilerInput {
  operations: Array<
    | CreateCompOp
    | AddLayerOp
    | AddEffectOp
    | SetKeyframeOp
    | SetPropertyOp
  >;
  metadata: {
    source: "phase6-auto-edit";
    confidence: number;
    timestamp: string;
    description: string;
  };
}
```

### 6.2 操作生成示例

```python
def generate_compiler_input(match_plan, music_atom, clip_atoms, target_style):
    """
    生成完整的 CompilerInput
    """
    operations = []
    ref_counter = {"layer": 0, "fx": 0, "kf": 0, "prop": 0}
    
    # 1. 创建主合成
    operations.append({
        "op": "createComp",
        "ref": "comp_main",
        "name": f"Auto Edit - {target_style}",
        "width": 1920,
        "height": 1080,
        "pixelAspect": 1,
        "duration": music_atom["metadata"]["duration_seconds"],
        "frameRate": 30,
        "bgColor": [0, 0, 0]
    })
    
    # 2. 添加所有片段为图层
    layer_refs = {}
    for i, clip in enumerate(clip_atoms):
        layer_ref = f"layer_{i+1:03d}"
        layer_refs[i] = layer_ref
        operations.append({
            "op": "addLayer",
            "ref": layer_ref,
            "compRef": "comp_main",
            "layerType": "footage",
            "source": clip["metadata"]["file"],
            "name": f"Clip {i+1}",
            "duration": clip["metadata"]["duration_seconds"]
        })
    
    # 3. 应用匹配方案
    for binding in match_plan["effect_binding"]:
        # 解析效果
        effect_info = resolve_effect_match_name(binding["effect"])
        if not effect_info: continue
        
        # 反推参数
        params = infer_params(effect_info["matchName"], binding["intensity"])
        params = apply_segment_modifier(params, get_segment_type(binding["time_range"][0], music_atom))
        params = apply_clip_context(params, get_active_clip(binding, clip_atoms))
        
        # 为每个时间段添加效果
        target_layer = get_active_layer(binding, layer_refs)
        effect_ref = f"fx_{++ref_counter['fx']:03d}"
        
        operations.append({
            "op": "addEffect",
            "ref": effect_ref,
            "layerRef": target_layer,
            "compRef": "comp_main",
            "matchName": effect_info["matchName"],
            "effectName": effect_info["displayName"]
        })
        
        # 设置属性
        for param_name, value in params.items():
            operations.append({
                "op": "setProperty",
                "ref": f"prop_{++ref_counter['prop']:03d}",
                "layerRef": target_layer,
                "compRef": "comp_main",
                "propertyPath": f"Effects/{effect_ref}/{param_name}",
                "value": value
            })
        
        # 生成节拍驱动关键帧
        beats_in_range = get_beats_in_range(
            binding["time_range"], music_atom["rhythm"]["beat_grid"]
        )
        if beats_in_range and "primary_param" in effect_info:
            keyframes = generate_beat_driven_keyframes(
                beats_in_range,
                effect_info["primary_param"],
                base_value=params.get(effect_info["primary_param"], 0) * 0.3,
                peak_value=params.get(effect_info["primary_param"], 1)
            )
            operations.append({
                "op": "setKeyframe",
                "ref": f"kf_{++ref_counter['kf']:03d}",
                "layerRef": target_layer,
                "compRef": "comp_main",
                "propertyPath": f"Effects/{effect_ref}/{effect_info['primary_param']}",
                "keyframes": keyframes
            })
    
    # 4. 添加转场
    for transition in match_plan["transition_plan"]:
        transition_ops = generate_transition_ops(
            transition,
            transition["time"],
            [layer_refs[0], layer_refs[1]]  # 简化
        )
        operations.extend(transition_ops)
    
    return {
        "operations": operations,
        "metadata": {
            "source": "phase6-auto-edit",
            "confidence": 0.85,
            "timestamp": datetime.now().isoformat(),
            "description": f"Auto-generated from match plan, style={target_style}"
        }
    }
```

## 第八章 置信度评估

### 8.1 参数置信度

每个反推出的参数都有置信度：

```python
def assess_param_confidence(match_name, param_name, value, source):
    """
    评估单个参数的置信度
    source: "exact_match" | "fuzzy_match" | "intensity_inferred" | "default"
    """
    base_confidence = {
        "exact_match": 0.95,
        "fuzzy_match": 0.7,
        "intensity_inferred": 0.6,
        "default": 0.4
    }.get(source, 0.5)
    
    # 检查值是否在合理范围
    config = INTENSITY_TO_PARAMS.get(match_name, {})
    value_ranges = config.get("value_ranges", {})
    
    if param_name in value_ranges:
        min_v, max_v = value_ranges[param_name]
        if not (min_v <= value <= max_v):
            base_confidence *= 0.5  # 超出范围，置信度减半
    
    return base_confidence
```

### 8.2 整体置信度

```python
def assess_overall_confidence(compiler_input):
    """
    评估整个 CompilerInput 的置信度
    """
    confidences = []
    
    for op in compiler_input["operations"]:
        if op["op"] == "addEffect":
            # 效果置信度
            conf = op.get("confidence", 0.7)
            confidences.append(conf)
        elif op["op"] == "setProperty":
            # 属性置信度
            conf = op.get("confidence", 0.6)
            confidences.append(conf)
    
    return sum(confidences) / max(1, len(confidences))
```
