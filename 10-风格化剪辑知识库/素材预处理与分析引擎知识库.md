#### Proteus 模型（推荐）
**适用场景**：真人视频、写实素材、电影级画质增强
**核心优势**：
- 对人物面部细节保留最好
- 运动模糊处理最自然
- 噪点控制优秀
**参数调优**：

| 参数 | 推荐值 | 说明 |
|------|--------|------|
| 超分倍率 | 2x / 4x | 2x 质量最优，4x 速度较慢 |
| 降噪强度 | 30-50 | 过高会损失细节 |
| 去模糊强度 | 20-40 | 过高会产生伪影 |
| 去交错 | Auto | 隔行素材自动检测 |

### 1.4 批量处理策略

```python
class TopazBatchProcessor:
    """Topaz 批量处理器"""
    
    def __init__(self, cli_path: str, max_workers: int = 2):
        self.cli_path = cli_path
        self.max_workers = max_workers
        self.task_queue = []
        self.results = []
    
    def add_task(self, input_path: str, output_path: str,
                 **params):
        """添加任务"""
        self.task_queue.append({
            "input": input_path,
            "output": output_path,
            "params": params
        })
    
    def process_all(self, progress_callback=None) -> List[dict]:
        """批量处理"""
        results = []
        completed = 0
        total = len(self.task_queue)
        
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {}
            for task in self.task_queue:
                future = executor.submit(
                    self._process_single, task
                )
                futures[future] = task
            
            for future in as_completed(futures):
                task = futures[future]
                try:
                    result = future.result()
                    results.append({
                        "input": task["input"],
                        "output": task["output"],
                        "success": result
                    })
                except Exception as e:
                    results.append({
                        "input": task["input"],
                        "output": task["output"],
                        "success": False,
                        "error": str(e)
                    })
                
                completed += 1
                if progress_callback:
                    progress_callback(completed, total)
        
        return results
    
    def _process_single(self, task: dict) -> bool:
        """处理单个任务"""
        params = task["params"]
        enhancer = TopazProteusEnhancer(self.cli_path)
        return enhancer.enhance_portrait_video(
            task["input"], task["output"],
            scale=params.get("scale", 2.0),
            quality=params.get("quality", "medium")
        )
```

## 二、PySceneDetect 镜头分割引擎

### 2.1 检测算法对比

| 算法 | 原理 | 速度 | 精度 | 适用场景 |
|------|------|------|------|---------|
| **ContentDetector** | 比较相邻帧内容差异 | 中 | 高 | 通用场景，推荐 |
| **ThresholdDetector** | 基于亮度阈值 | 快 | 低 | 高对比度剪辑 |
| **AdaptiveDetector** | 自适应阈值 | 中 | 高 | 复杂场景 |
| **HistogramDetector** | 直方图差异 | 快 | 中 | 色彩变化明显 |

### 2.2 核心实现

```python
from scenedetect import SceneManager, open_video
from scenedetect.detectors import ContentDetector, AdaptiveDetector
from scenedetect.scene_manager import save_images
from typing import List, Dict, Tuple
import json

class SceneAnalyzer:
    """镜头分析器"""
    
    def __init__(self, threshold: float = 27.0,
                 min_scene_len: int = 15):
        self.threshold = threshold
        self.min_scene_len = min_scene_len
    
    def detect_scenes(self, video_path: str) -> List[Dict]:
        """检测镜头"""
        video = open_video(video_path)
        scene_manager = SceneManager()
        
        scene_manager.add_detector(
            ContentDetector(
                threshold=self.threshold,
                min_scene_len=self.min_scene_len
            )
        )
        
        scene_manager.detect_scenes(video)
        scene_list = scene_manager.get_scene_list()
        
        scenes = []
        for i, scene in enumerate(scene_list):
            start_time = scene[0].get_seconds()
            end_time = scene[1].get_seconds()
            duration = end_time - start_time
            
            scenes.append({
                "index": i,
                "start_time": start_time,
                "end_time": end_time,
                "duration": duration,
                "start_frame": scene[0].get_frames(),
                "end_frame": scene[1].get_frames(),
                "frame_count": scene[1].get_frames() - scene[0].get_frames()
            })
        
        return scenes
    
    def detect_with_adaptive(self, video_path: str) -> List[Dict]:
        """自适应检测 - 适合复杂场景"""
        video = open_video(video_path)
        scene_manager = SceneManager()
        
        scene_manager.add_detector(
            AdaptiveDetector(
                adaptive_threshold=3.0,
                min_scene_len=self.min_scene_len,
                window_width=2
            )
        )
        
        scene_manager.detect_scenes(video)
        scene_list = scene_manager.get_scene_list()
        
        return [
            {
                "index": i,
                "start_time": scene[0].get_seconds(),
                "end_time": scene[1].get_seconds(),
                "duration": scene[1].get_seconds() - scene[0].get_seconds()
            }
            for i, scene in enumerate(scene_list)
        ]
    
    def extract_scene_frames(self, video_path: str, 
                            output_dir: str,
                            num_images: int = 3) -> List[str]:
        """提取每个镜头的关键帧"""
        import os
        os.makedirs(output_dir, exist_ok=True)
        
        video = open_video(video_path)
        scene_manager = SceneManager()
        scene_manager.add_detector(
            ContentDetector(threshold=self.threshold)
        )
        scene_manager.detect_scenes(video)
        
        image_filenames = save_images(
            scene_manager.get_scene_list(),
            video,
            num_images=num_images,
            output_dir=output_dir,
            image_name_template="scene-$SCENE_NUMBER-frame-$IMAGE_NUMBER"
        )
        
        return [os.path.join(output_dir, f) for f in image_filenames]
    
    def get_scene_statistics(self, scenes: List[Dict]) -> Dict:
        """获取镜头统计信息"""
        if not scenes:
            return {}
        
        durations = [s["duration"] for s in scenes]
        
        return {
            "total_scenes": len(scenes),
            "total_duration": sum(durations),
            "avg_duration": sum(durations) / len(durations),
            "min_duration": min(durations),
            "max_duration": max(durations),
            "median_duration": sorted(durations)[len(durations)//2],
            "short_scenes_count": sum(1 for d in durations if d < 1.0),
            "long_scenes_count": sum(1 for d in durations if d > 10.0)
        }
    
    def detect_cuts_only(self, video_path: str) -> List[float]:
        """仅检测转场点（时间点列表）"""
        scenes = self.detect_scenes(video_path)
        return [s["end_time"] for s in scenes[:-1]]
```

```python
class AdvancedSceneAnalyzer(SceneAnalyzer):

            "-f", "null",
            "-"
        ]
        
        result = subprocess.run(cmd, capture_output=True, text=True)
        # 解析 stats 输出计算平均运动强度
        # ...
        return 0.5  # 示例值
    
    def classify_scene_type(self, scenes: List[Dict]) -> List[str]:
        """分类镜头类型"""
        types = []
        for scene in scenes:
            duration = scene["duration"]
            if duration < 0.5:
                types.append("flash")  # 闪回/快速剪辑
            elif duration < 2.0:
                types.append("fast")   # 快速节奏
            elif duration < 5.0:
                types.append("normal") # 正常节奏
            elif duration < 15.0:
                types.append("slow")   # 慢节奏
            else:
                types.append("static") # 静态长镜头
        return types
    
    def find_action_scenes(self, video_path: str,
                          min_duration: float = 3.0) -> List[Dict]:
        """找动作场景（连续快速剪辑）"""
        scenes = self.detect_scenes(video_path)
        action_scenes = []
        
        i = 0
        while i < len(scenes):
            if scenes[i]["duration"] < 1.0:
                # 找到快速剪辑序列
                start_idx = i
                while i < len(scenes) and scenes[i]["duration"] < 1.5:
                    i += 1
                
                sequence_duration = sum(
                    s["duration"] for s in scenes[start_idx:i]
                )
                if sequence_duration >= min_duration:
                    action_scenes.append({
                        "start_time": scenes[start_idx]["start_time"],
                        "end_time": scenes[i-1]["end_time"],
                        "duration": sequence_duration,
                        "scene_count": i - start_idx
                    })
            else:
                i += 1
        
        return action_scenes
```

```python
import librosa
import librosa.display
import numpy as np
from typing import Dict, List, Tuple
import matplotlib.pyplot as plt

class AudioAnalyzer:
    """音频分析器"""
    
    def __init__(self, sr: int = 22050):
        self.sr = sr
        self.y = None
        self.duration = None
    
    def load_audio(self, video_path: str) -> Tuple[np.ndarray, int]:
        """从视频中加载音频"""
        y, sr = librosa.load(video_path, sr=self.sr, mono=True)
        self.y = y
        self.duration = librosa.get_duration(y=y, sr=sr)
        return y, sr
    
    def get_basic_features(self) -> Dict:
        """获取基础音频特征"""
        if self.y is None:
            raise ValueError("Audio not loaded")
        
        # 时域特征
        rms = librosa.feature.rms(y=self.y)
        zcr = librosa.feature.zero_crossing_rate(self.y)
        
        # 频域特征
        spectral_centroid = librosa.feature.spectral_centroid(y=self.y, sr=self.sr)
        spectral_bandwidth = librosa.feature.spectral_bandwidth(y=self.y, sr=self.sr)
        spectral_rolloff = librosa.feature.spectral_rolloff(y=self.y, sr=self.sr)
        
        return {
            "duration": self.duration,
            "rms_mean": float(np.mean(rms)),
            "rms_std": float(np.std(rms)),
            "zcr_mean": float(np.mean(zcr)),
            "spectral_centroid_mean": float(np.mean(spectral_centroid)),
            "spectral_bandwidth_mean": float(np.mean(spectral_bandwidth)),
            "spectral_rolloff_mean": float(np.mean(spectral_rolloff)),
            "max_amplitude": float(np.max(np.abs(self.y))),
            "min_amplitude": float(np.min(np.abs(self.y)))
        }
```

### 3.2 BPM/节拍检测

```python
class BeatAnalyzer(AudioAnalyzer):
    """节拍分析器"""
    
    def detect_tempo(self) -> Dict:
        """检测 BPM 和节拍"""
        tempo, beat_frames = librosa.beat.beat_track(
            y=self.y, sr=self.sr
        )
        
        beat_times = librosa.frames_to_time(beat_frames, sr=self.sr)
        
        return {
            "tempo": float(tempo),
            "beat_count": len(beat_times),
            "beat_times": beat_times.tolist(),
            "beat_interval": 60.0 / float(tempo) if tempo > 0 else 0
        }
    
    def detect_onsets(self) -> Dict:
        """检测音头（ onset ）"""
        onset_env = librosa.onset.onset_strength(y=self.y, sr=self.sr)
        onset_frames = librosa.onset.onset_detect(
            y=self.y, sr=self.sr, onset_envelope=onset_env
        )
        onset_times = librosa.frames_to_time(onset_frames, sr=self.sr)
        
        return {
            "onset_count": len(onset_times),
            "onset_times": onset_times.tolist(),
            "onset_strength": onset_env.tolist()
        }
    
    def get_beat_segments(self, min_length: int = 4) -> List[Dict]:
        """获取节拍段落（有规律的节拍区域）"""
        tempo, beat_frames = librosa.beat.beat_track(y=self.y, sr=self.sr)
        beat_times = librosa.frames_to_time(beat_frames, sr=self.sr)
        
        # 寻找连续节拍段
        segments = []
        if len(beat_times) < min_length:
            return segments
        
        current_segment = [beat_times[0]]
        
        for i in range(1, len(beat_times)):
            interval = beat_times[i] - beat_times[i-1]
            expected_interval = 60.0 / tempo
            
            # 如果间隔偏差超过 30%，认为节拍中断
            if abs(interval - expected_interval) / expected_interval > 0.3:
                if len(current_segment) >= min_length:
                    segments.append({
                        "start_time": current_segment[0],
                        "end_time": current_segment[-1],
                        "beat_count": len(current_segment),
                        "avg_interval": (current_segment[-1] - current_segment[0]) / (len(current_segment) - 1)
                    })
                current_segment = [beat_times[i]]
            else:
                current_segment.append(beat_times[i])
        
        if len(current_segment) >= min_length:
            segments.append({
                "start_time": current_segment[0],
                "end_time": current_segment[-1],
                "beat_count": len(current_segment),
                "avg_interval": (current_segment[-1] - current_segment[0]) / (len(current_segment) - 1)
            })
        
        return segments
```

### 3.3 音乐结构分析

```python
class MusicStructureAnalyzer(AudioAnalyzer):
    """音乐结构分析器"""
    
    def detect_segments(self) -> List[Dict]:
        """检测音乐段落（主歌/副歌等）"""
        # 计算色度特征
        chroma = librosa.feature.chroma_cqt(y=self.y, sr=self.sr)
        
        # 使用相似度矩阵检测段落边界
        recurrence = librosa.segment.recurrence_matrix(chroma)
        
        # 检测边界
        boundaries = librosa.segment.agglomerative(
            chroma, k=8
        )
        boundary_times = librosa.frames_to_time(
            boundaries, sr=self.sr
        )
        
        segments = []
        for i in range(len(boundary_times) - 1):
            start = boundary_times[i]
            end = boundary_times[i + 1]
            duration = end - start
            
            segments.append({
                "index": i,
                "start_time": float(start),
                "end_time": float(end),
                "duration": float(duration)
            })
        
        return segments
    
    def classify_mood(self) -> Dict:
        """音乐情绪分类"""
        features = self.get_basic_features()
        tempo_data = self.detect_tempo()
        
        tempo = tempo_data["tempo"]
        energy = features["rms_mean"]
        valence = features["spectral_centroid_mean"] / 4000.0  # 归一化
        
        # 简单情绪分类
        mood_scores = {
            "happy": min(1.0, (tempo / 140.0) * 0.5 + energy * 2),
            "sad": max(0.0, 1.0 - tempo / 140.0 - energy * 2),
            "energetic": min(1.0, tempo / 160.0 + energy * 1.5),
            "calm": max(0.0, 1.0 - tempo / 100.0 - energy),
            "epic": min(1.0, energy * 3 + valence * 0.5),
            "mysterious": max(0.0, 0.5 - energy + valence * 0.5)
        }
        
        # 找主导情绪
        dominant_mood = max(mood_scores.keys(), key=lambda k: mood_scores[k])
        
        return {
            "mood_scores": mood_scores,
            "dominant_mood": dominant_mood,
            "tempo": tempo,
            "energy": energy,
            "valence": valence
        }
    
    def get_key(self) -> Dict:
        """检测调性"""
        chroma = librosa.feature.chroma_cqt(y=self.y, sr=self.sr)
        chroma_mean = np.mean(chroma, axis=1)
        
        # 简单调性检测
```

```python
class VoiceMusicSeparator(AudioAnalyzer):
    """语音/音乐分离器"""
    
    def estimate_vocals(self) -> Dict:
        """估计人声存在概率"""
        # 使用 HPSS 分离谐波和冲击成分
        harmonic, percussive = librosa.effects.hpss(self.y)
        
        # 计算 MFCC 用于语音检测
        mfcc = librosa.feature.mfcc(y=self.y, sr=self.sr, n_mfcc=13)
        mfcc_delta = librosa.feature.delta(mfcc)
        
        # 简单的语音活动检测（基于能量和过零率）
        rms = librosa.feature.rms(y=self.y)[0]
        zcr = librosa.feature.zero_crossing_rate(self.y)[0]
        
        # 语音特征：中等能量，较高过零率
        frame_times = librosa.times_like(rms, sr=self.sr)
        
        voice_prob = []
        for i in range(len(rms)):
            # 简化的语音概率计算
            energy_score = min(1.0, rms[i] / 0.02)
            zcr_score = min(1.0, zcr[i] / 0.15)
            prob = energy_score * zcr_score * 1.5
            voice_prob.append(min(1.0, prob))
        
        return {
            "voice_probability": voice_prob,
            "frame_times": frame_times.tolist(),
            "harmonic_ratio": float(np.mean(np.abs(harmonic)) / (np.mean(np.abs(self.y)) + 1e-10)),
            "percussive_ratio": float(np.mean(np.abs(percussive)) / (np.mean(np.abs(self.y)) + 1e-10))
        }
    
    def get_speech_segments(self, threshold: float = 0.6,
                           min_duration: float = 0.5) -> List[Dict]:
        """获取语音段落"""
        result = self.estimate_vocals()
        voice_prob = result["voice_probability"]
        times = result["frame_times"]
        
        segments = []
        in_segment = False
        segment_start = 0
        
        for i, prob in enumerate(voice_prob):
            if prob >= threshold and not in_segment:
                in_segment = True
                segment_start = times[i]
            elif prob < threshold and in_segment:
                in_segment = False
                duration = times[i] - segment_start
                if duration >= min_duration:
                    segments.append({
                        "start_time": float(segment_start),
                        "end_time": float(times[i]),
                        "duration": float(duration),
                        "avg_voice_prob": float(np.mean(
                            voice_prob[max(0, i-10):i]
                        ))
                    })
        
        if in_segment:
            duration = times[-1] - segment_start
            if duration >= min_duration:
                segments.append({
                    "start_time": float(segment_start),
                    "end_time": float(times[-1]),
                    "duration": float(duration),
                    "avg_voice_prob": float(np.mean(voice_prob[-10:]))
                })
        
        return segments
```

```python
class SubtitleGenerator(SpeechRecognizer):
    """字幕生成器"""
    
    def generate_srt(self, audio_path: str,
                     output_path: str) -> str:

            start_time = self._format_srt_time(seg["start"])
            end_time = self._format_srt_time(seg["end"])
            text = seg["text"].strip()
            
            srt_content += f"{i}\n"
            srt_content += f"{start_time} --> {end_time}\n"
            srt_content += f"{text}\n\n"
        
        if output_path:
            with open(output_path, 'w', encoding='utf-8') as f:
                f.write(srt_content)
        
        return srt_content
    
    def generate_vtt(self, audio_path: str,
                     output_path: str) -> str:
        """生成 WebVTT 字幕"""
        result = self.transcribe(audio_path)
        segments = result["segments"]
        
        vtt_content = "WEBVTT\n\n"
        for seg in segments:
            start_time = self._format_vtt_time(seg["start"])
            end_time = self._format_vtt_time(seg["end"])
            text = seg["text"].strip()
            
            vtt_content += f"{start_time} --> {end_time}\n"
            vtt_content += f"{text}\n\n"
        
        if output_path:
            with open(output_path, 'w', encoding='utf-8') as f:
                f.write(vtt_content)
        
        return vtt_content
    
    def _format_srt_time(self, seconds: float) -> str:
        """格式化 SRT 时间"""
        hours = int(seconds // 3600)
        minutes = int((seconds % 3600) // 60)
        secs = int(seconds % 60)
        millis = int((seconds % 1) * 1000)
        return f"{hours:02d}:{minutes:02d}:{secs:02d},{millis:03d}"
    
    def _format_vtt_time(self, seconds: float) -> str:
        """格式化 VTT 时间"""
        hours = int(seconds // 3600)
        minutes = int((seconds % 3600) // 60)
        secs = int(seconds % 60)
        millis = int((seconds % 1) * 1000)
        return f"{hours:02d}:{minutes:02d}:{secs:02d}.{millis:03d}"
    
    def get_word_level_timestamps(self, audio_path: str) -> List[Dict]:
        """获取词级时间戳（用于口型同步）"""
        result = self.transcribe(audio_path)
```

```python
class JointAngleCalculator:
    """关节角度计算器"""
    
    @staticmethod
    def calculate_angle(point_a: Dict, point_b: Dict, 
                        point_c: Dict) -> float:
        """计算 ABC 角（B 为顶点）"""
        a = np.array([point_a["x"], point_a["y"]])
        b = np.array([point_b["x"], point_b["y"]])
        c = np.array([point_c["x"], point_c["y"]])
        
        ba = a - b
        bc = c - b
        
        cosine_angle = np.dot(ba, bc) / (np.linalg.norm(ba) * np.linalg.norm(bc))
        angle = np.arccos(np.clip(cosine_angle, -1.0, 1.0))
        
        return np.degrees(angle)
    
    def get_all_joint_angles(self, landmarks: List[Dict]) -> Dict:
        """计算所有关键关节角度"""
        # 索引映射
        lm = {l["name"]: l for l in landmarks}
        
        angles = {}
        
        # 肘部角度
        angles["left_elbow"] = self.calculate_angle(
            lm["left_shoulder"], lm["left_elbow"], lm["left_wrist"]
        )
        angles["right_elbow"] = self.calculate_angle(
            lm["right_shoulder"], lm["right_elbow"], lm["right_wrist"]
        )
        
        # 肩部角度
        angles["left_shoulder"] = self.calculate_angle(
```

```python
class PoseSmoother:
    """姿态平滑器"""
    
    def __init__(self, window_size: int = 5,
                 method: str = "savgol"):
        self.window_size = window_size
        self.method = method
        self.buffer = []
    
    def smooth_landmarks(self, landmarks_list: List[List[Dict]]) -> List[List[Dict]]:
        """平滑多帧关节点数据"""
        if len(landmarks_list) < self.window_size:
            return landmarks_list
        
        num_landmarks = len(landmarks_list[0])
        num_frames = len(landmarks_list)
        
        # 提取坐标数组
        xs = np.zeros((num_frames, num_landmarks))
        ys = np.zeros((num_frames, num_landmarks))
        vis = np.zeros((num_frames, num_landmarks))
        
        for t, frame in enumerate(landmarks_list):
            for i, lm in enumerate(frame):
                xs[t, i] = lm["x"]
                ys[t, i] = lm["y"]
                vis[t, i] = lm.get("visibility", 1.0)
        
        # 平滑处理
        if self.method == "savgol":
            from scipy.signal import savgol_filter
            window = min(self.window_size, num_frames)
            if window % 2 == 0:
                window -= 1
            if window >= 3:
                xs = savgol_filter(xs, window, 2, axis=0)
                ys = savgol_filter(ys, window, 2, axis=0)
        
        elif self.method == "moving_average":
            kernel = np.ones(self.window_size) / self.window_size
            for i in range(num_landmarks):
                xs[:, i] = np.convolve(xs[:, i], kernel, mode='same')
                ys[:, i] = np.convolve(ys[:, i], kernel, mode='same')
        
        elif self.method == "exponential":
            alpha = 0.3
            for t in range(1, num_frames):
                xs[t] = alpha * xs[t] + (1 - alpha) * xs[t-1]
                ys[t] = alpha * ys[t] + (1 - alpha) * ys[t-1]
        
        # 重建数据结构
        smoothed = []
        for t in range(num_frames):
            frame_lms = []
            for i in range(num_landmarks):
                lm = landmarks_list[t][i].copy()
                lm["x"] = float(xs[t, i])
                lm["y"] = float(ys[t, i])
                frame_lms.append(lm)
            smoothed.append(frame_lms)
        
        return smoothed
    
    def fill_missing_frames(self, landmarks_list: List[Optional[List[Dict]]]) -> List[List[Dict]]:
        """填充缺失帧"""
        num_frames = len(landmarks_list)
        num_landmarks = 33  # BlazePose
        
        # 找到有效帧
        valid_indices = [i for i, f in enumerate(landmarks_list) if f is not None]
        
        if not valid_indices:
            return []
        
        # 对每个关节点插值
        xs = np.zeros((num_frames, num_landmarks))
        ys = np.zeros((num_frames, num_landmarks))
        valid_mask = np.zeros((num_frames, num_landmarks), dtype=bool)
        
        for t, frame in enumerate(landmarks_list):
            if frame:
                for i, lm in enumerate(frame):
                    xs[t, i] = lm["x"]
                    ys[t, i] = lm["y"]
                    valid_mask[t, i] = True
        
        # 线性插值填充
        from scipy.interpolate import interp1d
        
        for i in range(num_landmarks):
            valid_t = np.where(valid_mask[:, i])[0]
            if len(valid_t) >= 2:
                f_x = interp1d(valid_t, xs[valid_t, i], 
                              kind='linear', fill_value='extrapolate')
                f_y = interp1d(valid_t, ys[valid_t, i],
                              kind='linear', fill_value='extrapolate')
                all_t = np.arange(num_frames)
                xs[:, i] = f_x(all_t)
                ys[:, i] = f_y(all_t)
        
        # 重建
        result = []
        for t in range(num_frames):
            frame = []
            ref_frame = landmarks_list[valid_indices[0]]
            for i in range(num_landmarks):
                lm = ref_frame[i].copy()
                lm["x"] = float(xs[t, i])
                lm["y"] = float(ys[t, i])
                lm["interpolated"] = not valid_mask[t, i]
                frame.append(lm)
            result.append(frame)
        
        return result
```

```python
class FaceMeshDetector:

            cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        )
        
        if not results.multi_face_landmarks:
            return None
        
        faces = []
        for face_landmarks in results.multi_face_landmarks:
            landmarks = []
            for i, lm in enumerate(face_landmarks.landmark):
                landmarks.append({
                    "index": i,
                    "x": lm.x,
                    "y": lm.y,
                    "z": lm.z
                })
            
            faces.append({
                "landmarks": landmarks,
                "num_landmarks": len(landmarks)
            })
        
        return {
            "num_faces": len(faces),
            "faces": faces
        }
    
    def detect_from_video(self, video_path: str,
                          sample_rate: int = 1) -> List[Dict]:
        """从视频检测"""
        cap = cv2.VideoCapture(video_path)
        fps = cap.get(cv2.CAP_PROP_FPS)
        frame_count = 0
        results_list = []
        
        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                break
            
            if frame_count % sample_rate == 0:
                result = self.detect_from_image(frame)
                results_list.append({
                    "frame": frame_count,
                    "timestamp": frame_count / fps,
                    "faces": result["faces"] if result else []
                })
            
            frame_count += 1
        
        cap.release()
        return results_list
```

```python
class FacialExpressionExtractor:
    """面部表情参数提取器"""
    
    # 关键特征点索引
    LANDMARK_INDICES = {

    }
    
    def extract_expression_params(self, landmarks: List[Dict]) -> Dict:
        """提取表情参数"""
        lm = landmarks
        
        # 归一化参考距离（眼间距）
        eye_distance = self._distance(
            lm[33], lm[263]
        )
        
        # 眼睛开合度（EAR: Eye Aspect Ratio）
        left_ear = self._eye_aspect_ratio(
            lm[159], lm[145], lm[33], lm[133]
        )
        right_ear = self._eye_aspect_ratio(
            lm[386], lm[374], lm[362], lm[263]
        )
        
        # 嘴巴开合度（MAR: Mouth Aspect Ratio）
        mar = self._mouth_aspect_ratio(
            lm[13], lm[14], lm[78], lm[308]
        )
        
        # 眉毛高度
        left_brow_height = self._distance(lm[66], lm[159]) / eye_distance
        right_brow_height = self._distance(lm[296], lm[386]) / eye_distance
        
        # 微笑程度（嘴角上扬）
        smile_level = self._calculate_smile(lm)
        
        # 眨眼检测
        eye_closed = (left_ear + right_ear) / 2 < 0.15
        
        return {
            "eye_openness": {
                "left": float(left_ear),
                "right": float(right_ear),
                "avg": float((left_ear + right_ear) / 2)
            },
            "mouth_openness": float(mar),
            "eyebrow_height": {
                "left": float(left_brow_height),
                "right": float(right_brow_height)
            },
            "smile_level": float(smile_level),
            "eye_closed": bool(eye_closed),
            "reference_eye_distance": float(eye_distance)
        }
    
    def _eye_aspect_ratio(self, top, bottom, left, right) -> float:
        """计算眼睛纵横比"""
        v_dist = self._distance(top, bottom)
        h_dist = self._distance(left, right)
        return v_dist / h_dist if h_dist > 0 else 0
    
    def _mouth_aspect_ratio(self, top, bottom, left, right) -> float:
        """计算嘴巴纵横比"""
        v_dist = self._distance(top, bottom)
        h_dist = self._distance(left, right)
        return v_dist / h_dist if h_dist > 0 else 0
    
    def _calculate_smile(self, landmarks: List[Dict]) -> float:
        """计算微笑程度"""
        # 嘴角与嘴唇中心的相对高度
        mouth_left = landmarks[78]
        mouth_right = landmarks[308]
        mouth_center = landmarks[13]
        
        # 嘴角平均高度相对于上唇中心
        avg_mouth_y = (mouth_left["y"] + mouth_right["y"]) / 2
        smile = (mouth_center["y"] - avg_mouth_y) / self._distance(mouth_left, mouth_right)
        
        return max(0.0, min(1.0, smile * 5 + 0.5))
    
    def _distance(self, p1: Dict, p2: Dict) -> float:
        """两点距离"""
        return np.sqrt((p1["x"] - p2["x"])**2 + (p1["y"] - p2["y"])**2)
    
    def classify_expression(self, params: Dict) -> Dict:
        """分类基本表情"""
        scores = {
            "neutral": 0.0,
            "happy": 0.0,
            "sad": 0.0,
            "angry": 0.0,
            "surprised": 0.0,
            "fear": 0.0
        }
        
        eye_open = params["eye_openness"]["avg"]
        mouth_open = params["mouth_openness"]
        smile = params["smile_level"]
        brow_height = (params["eyebrow_height"]["left"] + 
                       params["eyebrow_height"]["right"]) / 2
        
        # 简单规则分类
        if smile > 0.6 and mouth_open > 0.3:
            scores["happy"] = 0.8
        elif smile > 0.4:
            scores["happy"] = 0.5
        
        if mouth_open > 0.5 and eye_open > 0.4:
            scores["surprised"] = 0.7
        
        if brow_height > 0.3 and eye_open > 0.35:
            scores["fear"] = 0.5
        
        if smile < 0.2 and mouth_open < 0.1:
            scores["sad"] = 0.4
        
        if brow_height < 0.15 and eye_open < 0.25:
            scores["angry"] = 0.4
        
        # 中性表情
        max_score = max(scores.values())
        if max_score < 0.3:
            scores["neutral"] = 0.8
        
        # 归一化
        total = sum(scores.values())
        if total > 0:
            scores = {k: v/total for k, v in scores.items()}
        
        dominant = max(scores.keys(), key=lambda k: scores[k])
        
        return {
            "expression_scores": scores,
            "dominant_expression": dominant
        }
```

```python
class HeadPoseEstimator:
    """头部姿态估计器"""
    
    def __init__(self):
        # 3D 模型点（用于 PnP 求解）
        self.model_points = np.array([
            (0.0, 0.0, 0.0),              # 鼻尖
            (0.0, -330.0, -65.0),         # 下巴
            (-225.0, 170.0, -135.0),      # 左眼角
            (225.0, 170.0, -135.0),       # 右眼角
            (-150.0, -150.0, -125.0),     # 左嘴角
            (150.0, -150.0, -125.0)       # 右嘴角
        ], dtype=np.float64)
    
    def estimate_pose(self, landmarks: List[Dict],
                      image_width: int,
                      image_height: int) -> Dict:
        """估计头部姿态"""
        # 对应的 2D 图像点
        image_points = np.array([
            [landmarks[1]["x"] * image_width, landmarks[1]["y"] * image_height],   # 鼻尖
            [landmarks[152]["x"] * image_width, landmarks[152]["y"] * image_height], # 下巴
            [landmarks[33]["x"] * image_width, landmarks[33]["y"] * image_height],  # 左眼角
            [landmarks[263]["x"] * image_width, landmarks[263]["y"] * image_height], # 右眼角
            [landmarks[78]["x"] * image_width, landmarks[78]["y"] * image_height],  # 左嘴角
            [landmarks[308]["x"] * image_width, landmarks[308]["y"] * image_height] # 右嘴角
        ], dtype=np.float64)
        
        # 相机内参
        focal_length = image_width
        center = (image_width / 2, image_height / 2)
        camera_matrix = np.array([
            [focal_length, 0, center[0]],
            [0, focal_length, center[1]],
            [0, 0, 1]
        ], dtype=np.float64)
        
        dist_coeffs = np.zeros((4, 1))  # 假设无畸变
        
        # 求解 PnP
        success, rotation_vec, translation_vec = cv2.solvePnP(
            self.model_points,
            image_points,
            camera_matrix,
            dist_coeffs
        )
        
        if not success:
            return None
        
        # 旋转矩阵转欧拉角
        rotation_mat, _ = cv2.Rodrigues(rotation_vec)
        euler_angles = self._rotation_matrix_to_euler(rotation_mat)
        
        return {
            "pitch": float(euler_angles[0]),   # 俯仰（上下）
            "yaw": float(euler_angles[1]),     # 偏航（左右）
            "roll": float(euler_angles[2]),    # 翻滚（倾斜）
            "translation": {
                "x": float(translation_vec[0][0]),
                "y": float(translation_vec[1][0]),
                "z": float(translation_vec[2][0])
            }
        }
    
    def _rotation_matrix_to_euler(self, R: np.ndarray) -> np.ndarray:
        """旋转矩阵转欧拉角"""
        sy = np.sqrt(R[0, 0] * R[0, 0] + R[1, 0] * R[1, 0])
        
        singular = sy < 1e-6
        
        if not singular:
            x = np.arctan2(R[2, 1], R[2, 2])
            y = np.arctan2(-R[2, 0], sy)
            z = np.arctan2(R[1, 0], R[0, 0])
        else:
            x = np.arctan2(-R[1, 2], R[1, 1])
            y = np.arctan2(-R[2, 0], sy)
            z = 0
        
        return np.degrees(np.array([x, y, z]))
```

```python
class VideoQualityEvaluator:
    """视频质量评估器"""
    
    def evaluate(self, video_path: str) -> Dict:
        """综合质量评估"""
        results = {}
        
        # 基础技术指标
        results["technical"] = self._evaluate_technical(video_path)
        
        # 噪声水平
        results["noise"] = self._evaluate_noise(video_path)
        
        # 清晰度
        results["sharpness"] = self._evaluate_sharpness(video_path)
        
        # 压缩伪影
        results["compression"] = self._evaluate_compression(video_path)
        
        # 稳定性
        results["stability"] = self._evaluate_stability(video_path)
        
        # 总体评分
        results["overall_score"] = self._calculate_overall_score(results)
        
        return results
    
    def _evaluate_technical(self, video_path: str) -> Dict:
        """技术指标评估"""
        import subprocess
        import json
        
        cmd = [
            "ffprobe", "-v", "quiet",
            "-print_format", "json",
            "-show_format",
            "-show_streams",
            video_path
        ]
        
        result = subprocess.run(cmd, capture_output=True, text=True)
        data = json.loads(result.stdout)
        
        video_stream = next(
            s for s in data["streams"] 
            if s["codec_type"] == "video"
        )
        
        width = int(video_stream["width"])
        height = int(video_stream["height"])
        bitrate = int(data["format"]["bit_rate"])
        duration = float(data["format"]["duration"])
        fps = eval(video_stream["r_frame_rate"])
        
        # 分辨率评分
        resolution_score = min(100, (width * height) / (1920 * 1080) * 100)
        
        # 码率评分
        expected_bitrate = width * height * fps * 0.1  # 经验值
        bitrate_score = min(100, bitrate / expected_bitrate * 100)
        
        # 帧率评分
        fps_score = min(100, fps / 30 * 100)
        
        return {
            "width": width,
            "height": height,
            "resolution": f"{width}x{height}",
            "fps": fps,
            "bitrate": bitrate,
            "duration": duration,
            "resolution_score": resolution_score,
            "bitrate_score": bitrate_score,
            "fps_score": fps_score,
            "technical_score": (resolution_score + bitrate_score + fps_score) / 3
        }
    
    def _evaluate_noise(self, video_path: str) -> Dict:
        """噪声评估"""
        # 采样几帧计算噪声水平
        cap = cv2.VideoCapture(video_path)
        frame_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
        sample_frames = [
            int(frame_count * 0.25),
            int(frame_count * 0.5),
            int(frame_count * 0.75)
        ]
        
        noise_levels = []
        for frame_idx in sample_frames:
            cap.set(cv2.CAP_PROP_POS_FRAMES, frame_idx)
            ret, frame = cap.read()
            if ret:
                # 使用拉普拉斯方差估计噪声
                gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
                laplacian = cv2.Laplacian(gray, cv2.CV_64F)
                noise_level = np.std(laplacian)
                noise_levels.append(noise_level)
        
        cap.release()
        
        avg_noise = np.mean(noise_levels) if noise_levels else 0
        
        # 噪声评分（越小越好）
        noise_score = max(0, min(100, 100 - avg_noise / 2))
        
        return {
            "noise_level": float(avg_noise),
            "noise_score": float(noise_score)
        }
    
    def _evaluate_sharpness(self, video_path: str) -> Dict:
        """清晰度评估"""
        cap = cv2.VideoCapture(video_path)
        frame_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
        
        sharpness_values = []
        for i in range(0, min(frame_count, 30), 3):
            cap.set(cv2.CAP_PROP_POS_FRAMES, i)
            ret, frame = cap.read()
            if ret:
                gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
                # 拉普拉斯方差作为清晰度指标
                laplacian = cv2.Laplacian(gray, cv2.CV_64F)
                sharpness = np.var(laplacian)
                sharpness_values.append(sharpness)
        
        cap.release()
        
        avg_sharpness = np.mean(sharpness_values) if sharpness_values else 0
        
        # 清晰度评分（越高越好，100为基准）
        sharpness_score = min(100, avg_sharpness / 100)
        
        return {
            "sharpness": float(avg_sharpness),
            "sharpness_score": float(sharpness_score)
        }
    
    def _evaluate_compression(self, video_path: str) -> Dict:
        """压缩伪影评估"""
        # 简化版本：基于码率和分辨率估算
        cmd = ["ffprobe", "-v", "quiet", "-print_format", "json",
               "-show_streams", video_path]
        result = subprocess.run(cmd, capture_output=True, text=True)
        data = json.loads(result.stdout)
        
        video_stream = next(
            s for s in data["streams"] 
            if s["codec_type"] == "video"
        )
        
        width = int(video_stream["width"])
        height = int(video_stream["height"])
        bitrate = int(video_stream.get("bit_rate", 0))
        
        # 每像素比特数
        if bitrate > 0:
            bpp = bitrate / (width * height * eval(video_stream["r_frame_rate"]))
            compression_score = min(100, bpp * 1000)  # 越高越好
        else:
            bpp = 0
            compression_score = 50
        
        return {
            "bpp": float(bpp),
            "compression_score": float(compression_score)
        }
    
    def _evaluate_stability(self, video_path: str) -> Dict:
        """视频稳定性评估"""
        cap = cv2.VideoCapture(video_path)
        
        prev_gray = None
        motion_values = []
        
        frame_idx = 0
        while cap.isOpened() and frame_idx < 100:
            ret, frame = cap.read()
            if not ret:
                break
            
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            
            if prev_gray is not None:
                # 计算光流估计运动
                flow = cv2.calcOpticalFlowFarneback(
                    prev_gray, gray, None,
                    0.5, 3, 15, 3, 5, 1.2, 0
                )
                magnitude = np.sqrt(flow[..., 0]**2 + flow[..., 1]**2)
                avg_motion = np.mean(magnitude)
                motion_values.append(avg_motion)
            
            prev_gray = gray
            frame_idx += 1
        
        cap.release()
        
        if motion_values:
            avg_motion = np.mean(motion_values)
            motion_std = np.std(motion_values)
            
            # 稳定性评分（运动越小越稳定）
            stability_score = max(0, min(100, 100 - avg_motion * 10))
        else:
            avg_motion = 0
            motion_std = 0
            stability_score = 50
        
        return {
            "avg_motion": float(avg_motion),
            "motion_std": float(motion_std),
            "stability_score": float(stability_score)
        }
    
    def _calculate_overall_score(self, results: Dict) -> Dict:
        """计算总体评分"""
        weights = {
            "technical": 0.25,
            "noise": 0.20,
            "sharpness": 0.25,
            "compression": 0.15,
            "stability": 0.15
        }
        
        score = (
            results["technical"]["technical_score"] * weights["technical"] +
            results["noise"]["noise_score"] * weights["noise"] +
            results["sharpness"]["sharpness_score"] * weights["sharpness"] +
            results["compression"]["compression_score"] * weights["compression"] +
            results["stability"]["stability_score"] * weights["stability"]
        )
        
        # 质量等级
        if score >= 90:
            grade = "excellent"
        elif score >= 75:
            grade = "good"
        elif score >= 60:
            grade = "fair"
        elif score >= 40:
            grade = "poor"
        else:
            grade = "very_poor"
        
        return {
            "score": float(score),
            "grade": grade,
            "recommendation": self._get_recommendation(score, results)
        }
    
    def _get_recommendation(self, score: float, results: Dict) -> List[str]:
        """获取改进建议"""
        recommendations = []
        
        if results["noise"]["noise_score"] < 60:
            recommendations.append("建议降噪处理，可使用 Topaz Video AI 或 Neat Video")
        
        if results["sharpness"]["sharpness_score"] < 50:
            recommendations.append("建议锐化或超分辨率处理，可使用 Topaz Video AI")
        
        if results["stability"]["stability_score"] < 60:
            recommendations.append("建议防抖处理，可使用 Topaz 或 After Effects 变形稳定器")
        
        if results["compression"]["compression_score"] < 50:
            recommendations.append("建议使用更高码率或无损格式，减少压缩伪影")
        
        if not recommendations:
            recommendations.append("视频质量良好，可直接使用")
        
        return recommendations
```

```python
            "analysis_complete": True,
            "quality_grade": results.get("quality", {}).get("overall_score", {}).get("grade", "unknown"),
            "scene_count": len(results.get("scenes", [])),
            "duration": results.get("audio_features", {}).get("duration", 0)
        }
        
        if "beat" in results:
            summary["tempo"] = results["beat"]["tempo"]
            summary["beat_count"] = results["beat"]["beat_count"]
        
        # 处理建议
        recommendations = []
        quality = results.get("quality", {})
        
        if quality.get("overall_score", {}).get("score", 100) < 70:
            recommendations.append("建议先进行画质增强处理")
            recommendations.append("推荐使用 Topaz Video AI Proteus 模型")
        
        if summary.get("scene_count", 0) > 50:
            recommendations.append("镜头切换频繁，注意风格化一致性")
        
        summary["recommendations"] = recommendations
        
        return summary
```
