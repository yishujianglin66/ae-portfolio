# 效果推演引擎与成片预判

> [!abstract] 文档摘要
> 本文档定义从 MatchPlan → OutcomePrediction 的完整推演规范。包括效果叠加模拟（不真正渲染，预测视觉结果）、视觉签名预测（用既有视觉特征库推断成片特征）、节奏与情绪曲线预判、质量评分四大模块。让引擎在执行前能"看到"成片大致样貌。

> [!tip] 使用指南
> - 推演引擎开发者：第二至五章是核心
> - 质量评估者：第六章是评分体系
> - 调用方：第七章是端到端接口

## 第二章 效果叠加模拟

### 2.1 视觉效果模拟器

不真正渲染，而是用数学模型预测每个效果对画面的影响：

```python
class EffectSimulator:
    """
    模拟 AE 效果对画面的影响
    每个效果有：输入帧 → 输出帧的转换函数
    """
    
    def simulate_glow(self, frame_stats, intensity, radius, threshold):
        """
        模拟 Glow 效果对画面的影响
        frame_stats: 帧统计信息（亮度/对比度/饱和度）
        返回：效果后的帧统计信息
        """
        # Glow 增加亮度
        new_brightness = min(1.0, frame_stats["brightness"] + intensity * 0.2)
        # Glow 增加饱和度（边缘色更鲜艳）
        new_saturation = min(1.0, frame_stats["saturation"] + intensity * 0.1)
        # Glow 略微降低对比度（高光溢出）
        new_contrast = max(0, frame_stats["contrast"] - intensity * 0.05)
        
        return {
            "brightness": new_brightness,
            "saturation": new_saturation,
            "contrast": new_contrast,
            "added_elements": ["glow_halo"]
        }
    
    def simulate_glitch(self, frame_stats, displacement, frequency):
        """
        模拟 Glitch 故障效果
        """
        # 故障降低饱和度
        new_saturation = max(0, frame_stats["saturation"] - 0.1)
        # 故障增加对比度（色块分离）
        new_contrast = min(1.0, frame_stats["contrast"] + 0.15)
        # 添加 RGB 分离特征
        return {
            "brightness": frame_stats["brightness"],
            "saturation": new_saturation,
            "contrast": new_contrast,
            "added_elements": ["rgb_split", "displacement", "scanlines"]
        }
    
    def simulate_blur(self, frame_stats, blurriness):
        """
        模拟模糊效果
        """
        return {
            "brightness": frame_stats["brightness"],
            "saturation": max(0, frame_stats["saturation"] - blurriness * 0.005),
            "contrast": max(0, frame_stats["contrast"] - blurriness * 0.01),
            "added_elements": ["blur"]
        }
    
    # ... 其他效果
```

### 2.2 效果叠加顺序

AE 中效果按从上到下顺序应用，模拟器也按顺序叠加：

```python
def simulate_effect_stack(frame_stats, effect_chain):
    """
    effect_chain: 按应用顺序排列的效果列表
    """
    current_stats = frame_stats.copy()
    added_elements = []
    
    for effect in effect_chain:
        simulator_method = getattr(self, f"simulate_{effect['name']}")
        current_stats = simulator_method(current_stats, **effect["params"])
        added_elements.extend(current_stats.pop("added_elements", []))
    
    current_stats["added_elements"] = added_elements
    return current_stats
```

### 2.3 时间维度模拟

```python
def simulate_timeline(clip_atoms, match_plan, compiler_input):
    """
    模拟整个时间线的视觉效果
    """
    timeline = []
    
    for time_point in range(0, total_duration, 0.5):
        # 1. 找到当前时间点的活跃片段
        active_clip = find_active_clip(time_point, match_plan["alignment"])
        if not active_clip: continue
        
        # 2. 获取片段在该时间点的帧统计
        clip_atom = clip_atoms[active_clip["clip_index"]]
        frame_stats = interpolate_frame_stats(clip_atom, time_point - active_clip["start_time"])
        
        # 3. 获取当前时间点的活跃效果链
        effect_chain = find_active_effects(time_point, match_plan["effect_binding"])
        
        # 4. 模拟效果叠加
        simulated_stats = simulate_effect_stack(frame_stats, effect_chain)
        
        # 5. 记录
        timeline.append({
            "time": time_point,
            "clip": active_clip["clip_index"],
            "frame_stats": simulated_stats,
            "effects": [e["name"] for e in effect_chain]
        })
    
    return timeline
```

## 第三章 视觉签名预测

### 3.1 调用既有视觉特征库

复用 [[AE效果视觉特征库]] 中的视觉签名：

```python
from visual_signatures import VISUAL_SIGNATURE_LIBRARY

def predict_visual_signatures(match_plan, compiler_input):
    """
    预测成片将包含的视觉签名
    """
    predicted_signatures = []
    
    for binding in match_plan["effect_binding"]:
        effect_name = binding["effect"]
        
        # 从视觉特征库查找该效果的签名
        if effect_name in VISUAL_SIGNATURE_LIBRARY:
            signatures = VISUAL_SIGNATURE_LIBRARY[effect_name]
            
            for sig in signatures:
                predicted_signatures.append({
                    "signature_id": sig["id"],
                    "term_name": sig["term_name"],
                    "time_range": binding["time_range"],
                    "intensity": binding["intensity"],
                    "confidence": sig["confidence"] * binding.get("confidence", 1.0)
                })
    
    return predicted_signatures
```

### 3.2 风格匹配度计算

```python
def compute_style_match(predicted_signatures, target_style):
    """
    计算预测签名与目标风格的匹配度
    """
    # 目标风格期望的签名集合
    STYLE_EXPECTED_SIGNATURES = {
        "cyberpunk": ["neon_glow", "glitch", "rgb_split", "scanlines", "chromatic_aberration"],
        "cinematic": ["vignette", "film_grain", "warm_shadows", "cool_highlights", "deep_black"],
        "dream": ["soft_glow", "defocus", "particles", "pastel_colors"],
        "retro_vhs": ["vhs_noise", "scanlines", "color_bleed", "tracking_error"],
        "neon": ["strong_glow", "bloom", "light_rays", "vibrant_colors"]
    }
    
    expected = set(STYLE_EXPECTED_SIGNATURES.get(target_style, []))
    actual = set(sig["signature_id"] for sig in predicted_signatures)
    
    if not expected:
        return 0.5
    
    # Jaccard 相似度
    intersection = expected & actual
    union = expected | actual
    return len(intersection) / len(union)
```

### 3.3 预测色板

```python
def predict_color_palette(match_plan, clip_atoms, target_style):
    """
    预测成片的最终色板（受调色+效果影响）
    """
    # 1. 收集所有片段的主色板
    base_palette = []
    for clip in clip_atoms:
        base_palette.extend(clip["color"]["palette"][:3])
    
    # 2. 应用风格调色
    STYLE_COLOR_ADJUSTMENTS = {
        "cyberpunk": {
            "shift_h": 0.1,    # 色相偏移
            "boost_s": 0.2,   # 饱和度增强
            "target_hues": [180, 320]  # 青色+品红
        },
        "cinematic": {
            "shift_h": 0.0,
            "boost_s": 0.0,
            "shadows_shift": -20,   # 阴影偏蓝
            "highlights_shift": 20  # 高光偏橙
        }
    }
    
    adjustments = STYLE_COLOR_ADJUSTMENTS.get(target_style, {})
    
    # 3. 应用调整
    adjusted_palette = []
    for color in base_palette:
        hsl = hex_to_hsl(color)
        if "shift_h" in adjustments:
            hsl["h"] = (hsl["h"] + adjustments["shift_h"]) % 1
        if "boost_s" in adjustments:
            hsl["s"] = min(1, hsl["s"] + adjustments["boost_s"])
        adjusted_palette.append(hsl_to_hex(hsl))
    
    # 4. 去重+取主色
    unique_palette = list(dict.fromkeys(adjusted_palette))[:5]
    
    return unique_palette
```

## 第四章 节奏曲线预判

### 4.1 节拍对齐度计算

```python
def predict_beat_alignment(match_plan, music_atom):
    """
    预测成片节拍对齐度
    """
    beat_grid = music_atom["rhythm"]["beat_grid"]
    cuts = [a["start_time"] for a in match_plan["alignment"]]
    
    aligned_count = 0
    for cut in cuts:
        # 找最近的节拍
        closest_beat = min(beat_grid, key=lambda b: abs(b - cut))
        if abs(closest_beat - cut) < 0.1:  # 100ms 容差
            aligned_count += 1
    
    return aligned_count / max(1, len(cuts))
```

### 4.2 切镜密度预测

```python
def predict_cut_density(match_plan, total_duration):
    """
    预测切镜密度（每秒切镜数）
    """
    n_cuts = len(match_plan["alignment"])
    return n_cuts / total_duration
```

### 4.3 节奏合理性评分

```python
def evaluate_rhythm_quality(match_plan, music_atom):
    """
    评估节奏合理性 0-10
    """
    scores = []
    
    # 1. 节拍对齐度（0-4分）
    beat_alignment = predict_beat_alignment(match_plan, music_atom)
    scores.append(beat_alignment * 4)
    
    # 2. 段落内切镜一致性（0-3分）
    segment_consistency = compute_segment_consistency(match_plan)
    scores.append(segment_consistency * 3)
    
    # 3. 高潮段密度（0-3分）
    climax_density = compute_climax_density(match_plan, music_atom)
    scores.append(climax_density * 3)
    
    return sum(scores)
```

## 第五章 情绪曲线预判

### 5.1 情绪轨迹预测

```python
def predict_emotional_curve(timeline, match_plan):
    """
    预测成片的情绪轨迹
    """
    curve = []
    
    for time_point in timeline:
        # 1. 视觉情绪（从片段语义）
        clip_mood = time_point["clip_atom"]["semantic"]["mood"]
        
        # 2. 效果情绪（从效果类型）
        effect_mood = infer_effect_mood(time_point["effects"])
        
        # 3. 综合情绪
        combined_mood = combine_moods(clip_mood, effect_mood)
        
        # 4. 转 valence-arousal
        va = mood_to_valence_arousal(combined_mood)
        
        curve.append({
            "time": time_point["time"],
            "valence": va["valence"],
            "arousal": va["arousal"]
        })
    
    return curve
```

## 第六章 质量评分

### 6.1 综合评分体系

```python
def compute_quality_score(prediction, music_atom, match_plan):
    """
    综合质量评分 0-10
    """
    # 1. 节奏分（0-2.5）
    rhythm_score = prediction["rhythm"]["beat_alignment_score"] * 2.5
    
    # 2. 视觉风格分（0-2.5）
    style_score = prediction["visual_style"]["match_score"] * 2.5
    
    # 3. 情绪匹配分（0-2.5）
    expected_emotion = music_atom["emotional"]
    actual_emotion = prediction["emotional_curve"][-1]  # 取最后一点
    emotion_match = 1 - abs(expected_emotion["valence"] - actual_emotion["valence"]) * 0.5 - \
                       abs(expected_emotion["arousal"] - actual_emotion["arousal"]) * 0.5
    emotion_score = emotion_match * 2.5
    
    # 4. 技术分（0-2.5）
    technical_issues = len(prediction.get("potential_issues", []))
    technical_score = max(0, 2.5 - technical_issues * 0.5)
    
    total = rhythm_score + style_score + emotion_score + technical_score
    
    return {
        "overall": total,
        "sub_scores": {
            "rhythm": rhythm_score,
            "style": style_score,
            "emotion": emotion_score,
            "technical": technical_score
        }
    }
```

### 6.2 评分等级

| 分数 | 等级 | 描述 |
|------|------|------|
| 9.0-10 | S | 优秀，可直接执行 |
| 7.5-9.0 | A | 良好，建议微调 |
| 6.0-7.5 | B | 合格，需调整 |
| 4.5-6.0 | C | 一般，需大改 |
| <4.5 | D | 不合格，重新设计 |

### 6.3 潜在问题检测

```python
def detect_potential_issues(match_plan, clip_atoms, music_atom):
    """
    检测潜在问题
    """
    issues = []
    
    # 1. 片段时长问题
    for clip in clip_atoms:
        if clip["metadata"]["duration_seconds"] > 15:
            issues.append({
                "severity": "warning",
                "description": f"片段 {clip['metadata']['file']} 时长 {clip['metadata']['duration_seconds']}s 偏长，建议切分",
                "suggestion": f"在 {clip['metadata']['duration_seconds']/2}s 处切分"
            })
    
    # 2. 切镜密度过高
    avg_density = len(match_plan["alignment"]) / music_atom["metadata"]["duration_seconds"]
    if avg_density > 0.5:
        issues.append({
            "severity": "warning",
            "description": f"切镜密度 {avg_density:.2f}/s 过高",
            "suggestion": "考虑减少切点或延长镜头时长"
        })
    
    # 3. 色彩冲突
    palettes = [c["color"]["palette"] for c in clip_atoms]
    if detect_palette_conflict(palettes):
        issues.append({
            "severity": "info",
            "description": "片段间色板存在冲突",
            "suggestion": "建议统一调色"
        })
    
    # 4. 高潮错位
    music_peaks = music_atom["energy"]["peaks"]
    clip_peaks = [c["energy"]["peak_time"] for c in clip_atoms]
    # 检查片段峰值是否与音乐峰值对齐
    for mp in music_peaks:
        if not any(abs(cp - mp["time"]) < 1.0 for cp in clip_peaks):
            issues.append({
                "severity": "info",
                "description": f"音乐峰值在 {mp['time']}s 附近无视觉爆发",
                "suggestion": "在该时间点添加粒子/闪光效果"
            })
    
    return issues
```

## 第八章 QC 标准

### 8.1 预判准确度目标

| 指标 | 目标 | 验证方法 |
|------|------|---------|
| 视觉风格匹配度 | 与人工评分相关系数 > 0.7 | 渲染后人工评分对比 |
| 节拍对齐度 | 误差 < 5% | 实际渲染后重新计算 |
| 情绪曲线 | 与人工感知相关 > 0.6 | 主观评分对比 |
| 问题检测召回率 | > 80% | 人工核查 |

### 8.2 已知失败模式

| 失败模式 | 现象 | 修复建议 |
|---------|------|---------|
| 效果叠加过强 | 预测饱和度过高 | 加非线性限幅 |
| 风格配方过严 | 匹配度低 | 允许部分签名缺失 |
| 情绪推断错误 | 段落情绪不符 | 加入用户反馈调整 |
