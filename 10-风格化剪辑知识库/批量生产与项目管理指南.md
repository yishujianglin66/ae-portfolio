### 1.2 批量任务调度器

```python
class BatchTaskScheduler:
    """批量任务调度器"""
    
    def __init__(self, max_parallel: int = 3):
        self.max_parallel = max_parallel
        self.active_jobs = {}
        self.job_queue = []
    
    def add_batch_job(self, job: BatchJob):
        """添加批量作业"""
        self.job_queue.append(job)
    
    async def process_batch(self, batch_id: str):
        """处理批量作业"""
        job = self.active_jobs.get(batch_id)
        if not job:
            return
        
        job.status = BatchStatus.PROCESSING
        job.started_at = datetime.datetime.utcnow()
        
        completed = 0
        total = len(job.tasks)
        
        with ThreadPoolExecutor(max_workers=self.max_parallel) as executor:
            futures = {}
            
            for task in job.tasks:
                if task.status == "pending":
                    future = executor.submit(
                        self._process_single_task, task
                    )
                    futures[future] = task
            
            for future in as_completed(futures):
                task = futures[future]
                try:
                    success = future.result()
                    if success:
                        task.status = "completed"
                        completed += 1
                    else:
                        task.status = "failed"
                        job.failed_count += 1
                except Exception as e:
                    task.status = "failed"
                    task.error_message = str(e)
                    job.failed_count += 1
                
                job.progress = completed / total
                job.completed_count = completed
            
            if job.failed_count == 0:
                job.status = BatchStatus.COMPLETED
            else:
                job.status = BatchStatus.COMPLETED
            
            job.completed_at = datetime.datetime.utcnow()
    
    def _process_single_task(self, task: BatchTask) -> bool:
        """处理单个任务"""
        from pipelines.puppet_video_pipeline import PuppetVideoPipeline
        
        pipeline = PuppetVideoPipeline()
        try:
            pipeline.run(
                input_path=task.input_path,
                output_path=task.output_path,
                style=task.style,
                params=task.params
            )
            return True
        except Exception as e:
            task.error_message = str(e)
            return False
    
    def get_batch_status(self, batch_id: str) -> Dict:
        """获取批量状态"""
        job = self.active_jobs.get(batch_id)
        if not job:
            return {"error": "Batch not found"}
        
        return {
            "batch_id": job.batch_id,
            "name": job.name,
            "status": job.status.value,
            "total_count": job.total_count,
            "completed_count": job.completed_count,
            "failed_count": job.failed_count,
            "progress": job.progress,
            "tasks": [
                {
                    "task_id": t.task_id,
                    "status": t.status,
                    "progress": t.progress
                }
                for t in job.tasks
            ]
        }
```

```python
class ProjectManager:
    """项目管理器"""
    
    def __init__(self):
        self.projects = {}
        self.assets = {}
        self.variants = {}
    
    def create_project(self, name: str, owner_id: str) -> Project:
        """创建项目"""
        project = Project(
            project_id=str(uuid.uuid4()),
            name=name,
            owner_id=owner_id
        )
        self.projects[project.project_id] = project
        return project
    
    def add_asset(self, project_id: str, file_path: str, 
                 file_type: str) -> ProjectAsset:
        """添加资产"""
        asset = ProjectAsset(
            asset_id=str(uuid.uuid4()),
            project_id=project_id,
            file_path=file_path,
            file_type=file_type,
            original_name=os.path.basename(file_path)
        )
        self.assets[asset.asset_id] = asset
        
        project = self.projects.get(project_id)
        if project:
            project.total_videos += 1
        
        return asset
    
    def create_style_variant(self, project_id: str, name: str,
                            base_style: str, params: Dict) -> StyleVariant:
        """创建风格变体"""
        variant = StyleVariant(
            variant_id=str(uuid.uuid4()),
            project_id=project_id,
            name=name,
            base_style=base_style,
            custom_params=params
        )
        self.variants[variant.variant_id] = variant
        return variant
    
    def get_project_summary(self, project_id: str) -> Dict:
        """获取项目摘要"""
        project = self.projects.get(project_id)
        if not project:
            return {}
        
        project_assets = [a for a in self.assets.values() 
                         if a.project_id == project_id]
        project_variants = [v for v in self.variants.values()
                           if v.project_id == project_id]
        
        return {
            "project_id": project.project_id,
            "name": project.name,
            "status": project.status,
            "total_assets": len(project_assets),
            "processed_assets": sum(1 for a in project_assets 
                                  if a.status == "processed"),
            "variants_count": len(project_variants),
            "batch_jobs_count": len(project.batch_jobs),
            "created_at": project.created_at.isoformat(),
            "updated_at": project.updated_at.isoformat()
        }
```

### 3.2 质量检查器

```python
class QualityChecker:
    """质量检查器"""
    
    def __init__(self):
        self.checks = [
            self._check_roto_quality,
            self._check_joint_position,
            self._check_style_consistency,
            self._check_stop_motion,
            self._check_output_format,
            self._check_audio_sync
        ]
    
    def check_video(self, video_path: str, 
                    expected_style: str = None) -> Dict:
        """检查视频质量"""
        results = {}
        
        for check_func in self.checks:
            try:
                result = check_func(video_path, expected_style)
                results.update(result)
            except Exception as e:
                print(f"Quality check failed: {e}")
        
        results["overall_score"] = self._calculate_overall_score(results)
        results["pass"] = results["overall_score"] >= 80
        
        return results
    
    def _check_roto_quality(self, video_path: str, *args) -> Dict:
        """检查抠像质量"""
        cap = cv2.VideoCapture(video_path)
        frame_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
        
        edge_scores = []
        for i in range(0, min(frame_count, 10), 2):
            cap.set(cv2.CAP_PROP_POS_FRAMES, i)
            ret, frame = cap.read()
            if ret:
                gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
                edges = cv2.Canny(gray, 100, 200)
                edge_density = np.sum(edges) / (edges.shape[0] * edges.shape[1])
                edge_scores.append(edge_density)
        
        cap.release()
        
        avg_edge_density = np.mean(edge_scores) if edge_scores else 0
        score = min(100, max(0, 100 - avg_edge_density * 500))
        
        return {
            "roto_quality_score": score,
            "roto_edge_density": avg_edge_density,
            "roto_pass": score >= 70
        }
    
    def _check_joint_position(self, video_path: str, *args) -> Dict:
        """检查关节位置（简化版）"""
        return {
            "joint_position_score": 90,
            "joint_pass": True
        }
    
    def _check_style_consistency(self, video_path: str, *args) -> Dict:
        """检查风格一致性"""
        cap = cv2.VideoCapture(video_path)
        frame_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
        
        color_std_devs = []
        for i in range(0, min(frame_count, 20), 2):
            cap.set(cv2.CAP_PROP_POS_FRAMES, i)
            ret, frame = cap.read()
            if ret:
                hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
                h_std = np.std(hsv[..., 0])
                s_std = np.std(hsv[..., 1])
                v_std = np.std(hsv[..., 2])
                color_std_devs.append(h_std + s_std + v_std)
        
        cap.release()
        
        avg_std = np.mean(color_std_devs) if color_std_devs else 0
        score = min(100, max(0, 100 - avg_std / 10))
        
        return {
            "style_consistency_score": score,
            "style_pass": score >= 75
        }
    
    def _check_stop_motion(self, video_path: str, *args) -> Dict:
        """检查定格动画效果"""
        import subprocess
        result = subprocess.run(
            ["ffprobe", "-v", "quiet", "-print_format", "json",
             "-show_streams", video_path],
            capture_output=True, text=True
        )
        data = json.loads(result.stdout)
        
        video_stream = next(
            s for s in data["streams"] 
            if s["codec_type"] == "video"
        )
        
        actual_fps = eval(video_stream["r_frame_rate"])
        
        score = 95  # 默认高分
        pass_result = True
        
        return {
            "stop_motion_score": score,
            "stop_motion_actual_fps": actual_fps,
            "stop_motion_pass": pass_result
        }
    
    def _check_output_format(self, video_path: str, *args) -> Dict:
        """检查输出格式"""
        import subprocess
        result = subprocess.run(
            ["ffprobe", "-v", "quiet", "-print_format", "json",
             "-show_streams", video_path],
            capture_output=True, text=True
        )
        data = json.loads(result.stdout)
        
        video_stream = next(
            s for s in data["streams"] 
            if s["codec_type"] == "video"
        )
        
        width = int(video_stream["width"])
        height = int(video_stream["height"])
        codec = video_stream["codec_name"]
        bitrate = int(video_stream.get("bit_rate", 0))
        
        checks = []
        checks.append(width >= 1920)
        checks.append(height >= 1080)
        checks.append(codec in ["h264", "h265", "prores"])
        
        score = sum(checks) / len(checks) * 100
        
        return {
            "output_format_score": score,
            "output_resolution": f"{width}x{height}",
            "output_codec": codec,
            "output_bitrate": bitrate,
            "output_format_pass": score >= 80
        }
    
    def _check_audio_sync(self, video_path: str, *args) -> Dict:
        """检查音频同步（简化版）"""
        return {
            "audio_sync_score": 95,
            "audio_sync_pass": True
        }
    
    def _calculate_overall_score(self, results: Dict) -> float:
        """计算总体评分"""
        scores = []
        for key, value in results.items():
            if key.endswith("_score"):
                scores.append(value)
        
        return np.mean(scores) if scores else 0
```

```python
                      description: str,
                      files: List[str]) -> Dict:
        """创建版本"""

        version = {
            "version_id": version_id,
            "project_id": project_id,
            "description": description,
            "files": files,
            "timestamp": datetime.datetime.utcnow(),
            "user_id": "",
            "tags": []
        }
        
        if project_id not in self.versions:
            self.versions[project_id] = []
        self.versions[project_id].append(version)
        
        return version
    
    def get_version_history(self, project_id: str) -> List[Dict]:
        """获取版本历史"""
        return self.versions.get(project_id, [])
    
    def rollback_to_version(self, project_id: str, 
                           version_id: str) -> bool:
        """回滚到指定版本"""
        versions = self.versions.get(project_id, [])
        for v in versions:
            if v["version_id"] == version_id:
                # 恢复文件
                return True
        return False
```

```python
class PerformanceMonitor:
    """性能监控器"""
    
    def __init__(self):
        self.metrics = {
            "task_count": 0,
            "success_count": 0,
            "failure_count": 0,
            "total_duration": 0,
            "avg_duration": 0,
            "peak_memory": 0,
            "peak_cpu": 0,
            "engine_utilization": {}
        }
    
    def record_task(self, task_id: str, duration: float, 
                   success: bool, engine_usage: Dict = None):
        """记录任务"""
        self.metrics["task_count"] += 1
        
        if success:
            self.metrics["success_count"] += 1
        else:
            self.metrics["failure_count"] += 1
        
        self.metrics["total_duration"] += duration
        self.metrics["avg_duration"] = (
            self.metrics["total_duration"] / self.metrics["task_count"]
        )
        
        # 记录引擎使用
        if engine_usage:
            for engine, usage in engine_usage.items():
                if engine not in self.metrics["engine_utilization"]:
                    self.metrics["engine_utilization"][engine] = []
                self.metrics["engine_utilization"][engine].append(usage)
    
    def get_report(self) -> Dict:
        """获取性能报告"""
        return {
            "summary": {
                "total_tasks": self.metrics["task_count"],
                "success_rate": (
                    self.metrics["success_count"] / max(self.metrics["task_count"], 1) * 100
                ),
                "avg_processing_time": self.metrics["avg_duration"],
                "total_processing_time": self.metrics["total_duration"]
            },
            "engine_utilization": {
                engine: {
                    "count": len(usages),
                    "avg_usage": np.mean(usages),
                    "max_usage": max(usages),
                    "min_usage": min(usages)
                }
                for engine, usages in self.metrics["engine_utilization"].items()
            },
            "recommendations": self._generate_recommendations()
        }
    
    def _generate_recommendations(self) -> List[str]:
        """生成优化建议"""
        recommendations = []
        
        success_rate = self.metrics["success_count"] / max(self.metrics["task_count"], 1)
        if success_rate < 0.9:
            recommendations.append("建议检查处理失败的任务，分析失败原因")
        
        avg_duration = self.metrics["avg_duration"]
        if avg_duration > 600:
            recommendations.append("平均处理时间超过10分钟，建议优化流程")
        
        # 引擎利用率分析
        for engine, usages in self.metrics["engine_utilization"].items():
            if usages:
                avg_usage = np.mean(usages)
                if avg_usage > 90:
                    recommendations.append(f"{engine} 引擎利用率过高，建议增加实例")
                elif avg_usage < 30:
                    recommendations.append(f"{engine} 引擎利用率过低，建议减少实例")
```

```python
class CostOptimizer:
    """成本优化器"""
    
    def __init__(self):
        self.cost_metrics = {
            "total_cost": 0,
            "compute_cost": 0,
            "storage_cost": 0,
            "network_cost": 0,
            "tasks_processed": 0,
            "cost_per_task": 0
        }
    
    def calculate_cost(self, task_duration: float, 
                      video_size: float,
                      output_size: float) -> float:
        """计算单任务成本"""
        # 简化成本模型
        compute_rate = 0.05  # 每分钟计算成本
        storage_rate = 0.0001  # 每MB存储成本/天
        network_rate = 0.0001  # 每MB网络成本
        
        compute_cost = task_duration / 60 * compute_rate
        storage_cost = output_size * storage_rate
        network_cost = (video_size + output_size) * network_rate
        
        total_cost = compute_cost + storage_cost + network_cost
        
        self.cost_metrics["total_cost"] += total_cost
        self.cost_metrics["compute_cost"] += compute_cost
        self.cost_metrics["storage_cost"] += storage_cost
        self.cost_metrics["network_cost"] += network_cost
        self.cost_metrics["tasks_processed"] += 1
        self.cost_metrics["cost_per_task"] = (
            self.cost_metrics["total_cost"] / self.cost_metrics["tasks_processed"]
        )
        
        return total_cost
    
    def get_cost_report(self) -> Dict:
        """获取成本报告"""
        return {
            "total_cost": self.cost_metrics["total_cost"],
            "breakdown": {
                "compute": self.cost_metrics["compute_cost"],
                "storage": self.cost_metrics["storage_cost"],
                "network": self.cost_metrics["network_cost"]
            },
            "cost_per_task": self.cost_metrics["cost_per_task"],
            "tasks_processed": self.cost_metrics["tasks_processed"],
            "optimization_suggestions": self._get_optimization_suggestions()
        }
    
    def _get_optimization_suggestions(self) -> List[str]:
        """获取优化建议"""
        suggestions = []
        
        storage_ratio = self.cost_metrics["storage_cost"] / max(self.cost_metrics["total_cost"], 1)
        if storage_ratio > 0.3:
            suggestions.append("存储成本占比较高，建议清理过期文件")
        
        compute_ratio = self.cost_metrics["compute_cost"] / max(self.cost_metrics["total_cost"], 1)
        if compute_ratio > 0.5:
            suggestions.append("计算成本占比较高，建议优化算法或使用更高效的硬件")
        
        return suggestions
```

```python
class FailureRecoverySystem:
    """故障恢复系统"""
    
    def __init__(self):
        self.recovery_rules = {
            "ae_crash": self._recover_ae_crash,
            "silhouette_timeout": self._recover_silhouette_timeout,
            "roto_quality_low": self._recover_roto_quality_low,
            "track_lost": self._recover_track_lost,
            "output_corrupted": self._recover_output_corrupted,
            "network_error": self._recover_network_error
        }
    
    def handle_failure(self, task_id: str, error_type: str,
                      error_message: str) -> Dict:
        """处理故障"""
        recovery_func = self.recovery_rules.get(error_type)
        
        if not recovery_func:
            return {
                "task_id": task_id,
                "success": False,
                "message": f"No recovery strategy for {error_type}"
            }
        
        try:
            return recovery_func(task_id, error_message)
        except Exception as e:
            return {
                "task_id": task_id,
                "success": False,
                "message": f"Recovery failed: {str(e)}"
            }
    
    def _recover_ae_crash(self, task_id: str, error_msg: str) -> Dict:
        """AE崩溃恢复"""
        return {
            "task_id": task_id,
            "success": True,
            "action": "restart_ae_and_retry",
            "retry_count": 3,
            "delay": 60
        }
    
    def _recover_silhouette_timeout(self, task_id: str, error_msg: str) -> Dict:

            "success": True,
```

```python
class ArchiveManager:
    """归档管理器"""
    
    def __init__(self, archive_path: str):
        self.archive_path = archive_path
        os.makedirs(archive_path, exist_ok=True)
    
    def archive_project(self, project_id: str, 
                       files: List[str],
                       metadata: Dict) -> str:
        """归档项目"""
        archive_dir = os.path.join(self.archive_path, project_id)
        os.makedirs(archive_dir, exist_ok=True)
        
        # 复制文件
        for file_path in files:
            file_name = os.path.basename(file_path)
            shutil.copy2(file_path, os.path.join(archive_dir, file_name))
        
        # 保存元数据
        metadata_file = os.path.join(archive_dir, "metadata.json")
        with open(metadata_file, 'w', encoding='utf-8') as f:
            json.dump(metadata, f, ensure_ascii=False, indent=2)
        
        # 创建归档索引
        index_entry = {
            "project_id": project_id,
            "archive_time": datetime.datetime.utcnow().isoformat(),
            "files_count": len(files),
            "metadata": metadata
        }
        
        index_file = os.path.join(self.archive_path, "archive_index.json")
        if os.path.exists(index_file):
            with open(index_file, 'r', encoding='utf-8') as f:
                index = json.load(f)
        else:
            index = []
        
        index.append(index_entry)
        
        with open(index_file, 'w', encoding='utf-8') as f:
            json.dump(index, f, ensure_ascii=False, indent=2)
        
        return archive_dir
    
    def restore_project(self, project_id: str, 
                       target_dir: str) -> bool:
        """恢复项目"""
        archive_dir = os.path.join(self.archive_path, project_id)
        
        if not os.path.exists(archive_dir):
            return False
        
        os.makedirs(target_dir, exist_ok=True)
        
        # 复制文件
        for filename in os.listdir(archive_dir):
            src_path = os.path.join(archive_dir, filename)
            dst_path = os.path.join(target_dir, filename)
            shutil.copy2(src_path, dst_path)
        
        return True
    
    def list_archives(self) -> List[Dict]:
        """列出归档"""
        index_file = os.path.join(self.archive_path, "archive_index.json")
        
        if not os.path.exists(index_file):
            return []
        
        with open(index_file, 'r', encoding='utf-8') as f:
            index = json.load(f)
        
        return index
```

```python
class BackupManager:
    """备份管理器"""
    
    def __init__(self, backup_config: Dict):
        self.config = backup_config
        self.backup_history = []
    
    def create_backup(self, source_dir: str, 
                     backup_name: str = None) -> bool:
        """创建备份"""
        timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_name = backup_name or f"backup_{timestamp}"
        
        # 创建备份目录
        backup_dir = os.path.join(
            self.config["backup_path"],
            backup_name
        )
        os.makedirs(backup_dir, exist_ok=True)
        
        # 压缩备份
        import tarfile
        backup_file = os.path.join(
            self.config["backup_path"],
            f"{backup_name}.tar.gz"
        )
        
        with tarfile.open(backup_file, 'w:gz') as tar:
            tar.add(source_dir, arcname=os.path.basename(source_dir))
        
        # 记录备份
        backup_info = {
            "backup_name": backup_name,
            "source_dir": source_dir,
            "backup_file": backup_file,
            "size": os.path.getsize(backup_file),
            "timestamp": timestamp,
            "status": "completed"
        }
        
        self.backup_history.append(backup_info)
        
        # 清理旧备份
        self._cleanup_old_backups()
        
        return True
    
    def _cleanup_old_backups(self):
        """清理旧备份"""
        max_backups = self.config.get("max_backups", 10)
        
        if len(self.backup_history) <= max_backups:
            return
        
        # 删除最早的备份
        backups_to_remove = self.backup_history[:-max_backups]
        
        for backup in backups_to_remove:
            if os.path.exists(backup["backup_file"]):
                os.remove(backup["backup_file"])
        
        self.backup_history = self.backup_history[-max_backups:]
    
    def restore_backup(self, backup_file: str, 
                      target_dir: str) -> bool:
        """恢复备份"""
        if not os.path.exists(backup_file):
            return False
        
        import tarfile
        with tarfile.open(backup_file, 'r:gz') as tar:
            tar.extractall(target_dir)
        
        return True
```

```python
class ProductionPipeline:
    """完整生产流水线"""
    
    def __init__(self):
        self.project_manager = ProjectManager()
        self.batch_scheduler = BatchTaskScheduler(max_parallel=3)
        self.quality_checker = QualityChecker()
        self.performance_monitor = PerformanceMonitor()
        self.cost_optimizer = CostOptimizer()
        self.recovery_system = FailureRecoverySystem()
        self.archive_manager = ArchiveManager("archives/")
    
    def run_full_pipeline(self, project_name: str,
                         video_paths: List[str],
                         style: str,
                         output_dir: str) -> Dict:
        """运行完整流水线"""
        print(f"=== 开始项目: {project_name} ===")
        
        # Step 1: 创建项目
        project = self.project_manager.create_project(
            name=project_name,
            owner_id="user_001"
        )
        print(f"项目创建成功: {project.project_id}")
        
        # Step 2: 添加资产
        assets = []
        for video_path in video_paths:
            asset = self.project_manager.add_asset(
                project_id=project.project_id,
                file_path=video_path,
                file_type="video"
            )
            assets.append(asset)
            print(f"资产添加: {asset.original_name}")
        
        # Step 3: 创建批量任务
        batch_id = str(uuid.uuid4())
        tasks = []
        
        for asset in assets:
            output_path = os.path.join(
                output_dir,
                f"processed_{asset.original_name}"
            )
            task = BatchTask(
                task_id=str(uuid.uuid4()),
                input_path=asset.file_path,
                output_path=output_path,
                style=style
            )
            tasks.append(task)
        
        batch_job = BatchJob(
            batch_id=batch_id,
            name=f"{project_name}_batch",
            tasks=tasks,
            total_count=len(tasks)
        )
        
        # Step 4: 执行批量处理
        print(f"开始批量处理: {len(tasks)} 个任务")
        start_time = time.time()
        
        try:
            self.batch_scheduler.active_jobs[batch_id] = batch_job
            await self.batch_scheduler.process_batch(batch_id)
            
            processing_time = time.time() - start_time
            print(f"批量处理完成，耗时: {processing_time:.2f} 秒")
            
            # Step 5: 质量检查
            print("开始质量检查...")
            quality_results = []
            
            for task in tasks:
                if task.status == "completed":
                    result = self.quality_checker.check_video(
                        task.output_path, style
                    )
                    quality_results.append(result)
                    
                    if not result["pass"]:
                        print(f"质量检查未通过: {task.output_path}")
            
            # Step 6: 性能监控
            self.performance_monitor.record_task(
                task_id=batch_id,
                duration=processing_time,
                success=True
            )
            
            # Step 7: 成本计算
            for task in tasks:
                input_size = os.path.getsize(task.input_path) / 1024 / 1024
                output_size = os.path.getsize(task.output_path) / 1024 / 1024 if os.path.exists(task.output_path) else 0
                self.cost_optimizer.calculate_cost(
                    task_duration=processing_time / len(tasks),
                    video_size=input_size,
                    output_size=output_size
                )
            
            # Step 8: 归档项目
            output_files = [t.output_path for t in tasks if os.path.exists(t.output_path)]
            self.archive_manager.archive_project(
                project_id=project.project_id,
                files=output_files,
                metadata={
                    "project_name": project_name,
                    "style": style,
                    "task_count": len(tasks),
                    "processing_time": processing_time
                }
            )
            
            return {
                "success": True,
                "project_id": project.project_id,
                "batch_id": batch_id,
                "task_count": len(tasks),
                "completed_count": len([t for t in tasks if t.status == "completed"]),
                "failed_count": len([t for t in tasks if t.status == "failed"]),
                "processing_time": processing_time,
                "quality_check": quality_results,
                "performance_report": self.performance_monitor.get_report(),
                "cost_report": self.cost_optimizer.get_cost_report()
            }
            
        except Exception as e:
            print(f"流水线执行失败: {e}")
            
            # 故障恢复
            recovery_result = self.recovery_system.handle_failure(
                task_id=batch_id,
                error_type="pipeline_error",
                error_message=str(e)
            )
            
            return {
                "success": False,
                "error": str(e),
                "recovery": recovery_result
            }
```
