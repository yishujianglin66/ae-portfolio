## 第二章 AE 图层操作详解

### 2.1 图层修剪（Trim）

#### 修剪类型

| 类型 | 描述 | AE 操作 | ExtendScript 方法 |
|------|------|---------|------------------|
| in_point 修剪 | 修改入点 | `[ 拖动图层左端 ]` | `layer.inPoint = time` |
| out_point 修剪 | 修改出点 | `[ 拖动图层右端 ]` | `layer.outPoint = time` |
| 滑动修剪 | 保持时长不变，移动内容 | `[ Ctrl+拖动 ]` | 用 `sourceTime` 表达式 |
| 分割 | 一分为二 | `Ctrl+Shift+D` | `layer.split()` |

#### ExtendScript 示例

```javascript
// 1. 设置入点和出点
var layer = app.project.activeItem.layer(1);
layer.inPoint = 2.0;    // 从第2秒开始
layer.outPoint = 8.0;   // 到第8秒结束

// 2. 分割图层（在当前时间分割）
var newLayer = layer.split(app.project.activeItem.time);

// 3. 滑动内容（保持图层时长，移动源时间）
// 用时间重映射实现
layer.timeRemapEnabled = true;
layer.timeRemap.setValueAtTime(0, 2);  // 图层开始时播放源2秒处
layer.timeRemap.setValueAtTime(6, 8); // 图层6秒时播放源8秒处
```

## 第三章 FFmpeg 视频处理工具链

### 3.1 基本信息读取

```bash
# 读取视频元数据
ffprobe -v quiet -print_format json -show_format -show_streams input.mp4
```

### 3.2 视频修剪

```bash
# 按时间修剪（重新编码，最可靠）
ffmpeg -i input.mp4 -ss 00:01:30 -t 00:00:30 -c:v libx264 -crf 18 -c:a aac output.mp4

# 按时间修剪（快速复制，可能不准确）
ffmpeg -i input.mp4 -ss 00:01:30 -to 00:02:00 -c copy output.mp4

# 按帧修剪（精确到帧）
ffmpeg -i input.mp4 -vf "select=between(n\,100\,300)" -c:v libx264 -crf 18 output.mp4
```

### 3.3 视频拼接

```bash
# 方法1：拼接列表文件（推荐，重编码确保一致性）
# 创建 filelist.txt：
# file 'part1.mp4'
# file 'part2.mp4'
ffmpeg -f concat -safe 0 -i filelist.txt -c:v libx264 -crf 18 -c:a aac -movflags +faststart output.mp4

# 方法2：filter_complex 拼接（最可靠）
ffmpeg -i part1.mp4 -i part2.mp4 -filter_complex "[0:v][0:a][1:v][1:a]concat=n=2:v=1:a=1[v][a]" -map "[v]" -map "[a]" -c:v libx264 -crf 18 output.mp4

# 注意：-c copy 拼接可能因 moov atom 损坏失败，重要项目必须重编码
```

### 3.4 格式转换/转码

```bash
# H.264 高质量
ffmpeg -i input.mov -c:v libx264 -crf 18 -preset slow -c:a aac -b:a 320k -movflags +faststart output.mp4

# H.265 / HEVC
ffmpeg -i input.mp4 -c:v libx265 -crf 20 -preset slow -c:a aac output_hevc.mp4

# ProRes（苹果高质量）
ffmpeg -i input.mp4 -c:v prores_ks -profile:v 3 -c:a pcm_s16le output.mov

# 调整分辨率
ffmpeg -i input.mp4 -vf "scale=1920:1080" -c:v libx264 -crf 18 output_1080p.mp4

# 调整帧率
ffmpeg -i input.mp4 -r 30 -c:v libx264 -crf 18 output_30fps.mp4
```

### 3.5 音频提取/处理

```bash
# 提取音频
```

### 3.7 视频信息批量分析脚本

```python
import subprocess
import json

def get_video_info(filepath):
    """获取视频元数据"""
    cmd = [
        'ffprobe', '-v', 'quiet',
        '-print_format', 'json',
        '-show_format', '-show_streams',
        filepath
    ]
    result = subprocess.run(cmd, capture_output=True, text=True, encoding='utf-8', errors='replace')
    return json.loads(result.stdout)

def analyze_folder(folder):
    """批量分析文件夹中视频"""
    import os
    results = []
    for f in os.listdir(folder):
        if f.endswith(('.mp4', '.mov', '.avi', '.mkv')):
            info = get_video_info(os.path.join(folder, f))
            results.append({
                "file": f,
                "duration": float(info['format']['duration']),
                "resolution": f"{info['streams'][0]['width']}x{info['streams'][0]['height']}",
                "codec": info['streams'][0]['codec_name']
            })
    return results
```

```python
def auto_rough_cut(video_path, target_duration, music_beats=None):
    """
    自动粗剪：从视频中提取最佳片段组合
    """
    # 1. 场景检测
    scenes = detect_scenes(video_path)
    
    # 2. 评估每个场景的质量（运动/色彩/构图）
    scene_scores = []
    for scene in scenes:
        score = evaluate_scene_quality(scene)
        scene_scores.append(score)
    
    # 3. 选择最佳场景组合，总时长接近目标
    selected = select_best_scenes(scenes, scene_scores, target_duration)
    
    # 4. 如果有音乐节拍，对齐到节拍
    if music_beats:
        selected = align_to_beats(selected, music_beats)
    
    # 5. 输出粗剪
    output_rough_cut(video_path, selected, "rough_cut.mp4")
    
    return selected
```

### 4.3 场景质量评估

```python
def evaluate_scene_quality(scene_frames):
    """评估一个场景的质量"""
    quality_scores = {}
    
    # 1. 运动评分（中等运动最好，过静/过动都减分）
    motion = calculate_motion(scene_frames)
    motion_score = -abs(motion - 0.5) + 0.5  # 0.5最佳
    quality_scores["motion"] = motion_score
    
    # 2. 色彩评分（饱和度适中+对比度高）
    color = analyze_color(scene_frames)
    color_score = color["contrast"] * 0.5 + color["saturation"] * 0.5
    quality_scores["color"] = color_score
    
    # 3. 构图评分
    composition = analyze_composition(scene_frames)
    quality_scores["composition"] = composition["rule_of_thirds"]
    
    # 4. 综合
    total = sum(quality_scores.values()) / len(quality_scores)
    return total
```

### 4.4 场景选择算法

```python
def select_best_scenes(scenes, scores, target_duration):
    """
    贪心算法选择最佳场景，总时长接近目标
    """
    # 按质量分数排序
    ranked = sorted(zip(scores, scenes), key=lambda x: x[0], reverse=True)
    
    selected = []
    total_duration = 0
    
    for score, scene in ranked:
        if total_duration + scene["duration"] <= target_duration * 1.2:  # 允许20%冗余
            selected.append(scene)
            total_duration += scene["duration"]
        if total_duration >= target_duration:
            break
    
    # 按时序排序
    selected.sort(key=lambda s: s["start_time"])
    
    return selected
```

## 第五章 自动化脚本与 AE 集成

### 5.1 素材预处理流水线

```bash
#!/bin/bash
# preprocess_pipeline.sh
# 素材预处理流水线：格式统一+场景检测+代理生成

INPUT_FOLDER="02-素材/原始素材"
OUTPUT_FOLDER="02-素材/剪辑素材"
PROXY_FOLDER="02-素材/代理素材"

# 1. 格式统一
for f in "$INPUT_FOLDER"/*.{mp4,mov,avi}; do
    if [ -f "$f" ]; then
        filename=$(basename "$f")
        echo "Processing: $filename"
        ffmpeg -i "$f" -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 320k -movflags +faststart "$OUTPUT_FOLDER/${filename%.*}.mp4" -y
    fi
done

# 2. 场景检测
for f in "$OUTPUT_FOLDER"/*.mp4; do
    if [ -f "$f" ]; then
        echo "Detecting scenes: $(basename $f)"
        scenedetect -i "$f" detect-content list-scenes -o "$OUTPUT_FOLDER/scenes/"
    fi
done

# 3. 代理生成
for f in "$OUTPUT_FOLDER"/*.mp4; do
    if [ -f "$f" ]; then
        filename=$(basename "$f")
        echo "Creating proxy: $filename"
        ffmpeg -i "$f" -vf "scale=960:540" -c:v libx264 -crf 23 -preset fast -c:a aac -b:a 128k "$PROXY_FOLDER/${filename%.*}_proxy.mp4" -y
    fi
done

echo "=== Preprocessing complete ==="
```

### 5.2 Python 自动化模块

```python
"""
video_pipeline.py
视频处理流水线自动化模块
"""
import subprocess
import json
import os

class VideoProcessor:
    """视频处理器"""
    
    def __init__(self, ffmpeg_path="ffmpeg", ffprobe_path="ffprobe"):
        self.ffmpeg = ffmpeg_path
        self.ffprobe = ffprobe_path
    
    def get_info(self, filepath):
        """获取视频信息"""
        cmd = [
            self.ffprobe, '-v', 'quiet',
            '-print_format', 'json',
            '-show_format', '-show_streams',
            filepath
        ]
        result = subprocess.run(cmd, capture_output=True, text=True, encoding='utf-8', errors='replace')
        return json.loads(result.stdout)
    
    def trim(self, input_path, output_path, start_time, duration):
        """修剪视频（重编码，确保质量）"""
        cmd = [
            self.ffmpeg, '-i', input_path,
            '-ss', str(start_time),
            '-t', str(duration),
            '-c:v', 'libx264', '-crf', '18',
            '-c:a', 'aac', '-b:a', '320k',
            '-movflags', '+faststart',
            '-y', output_path
        ]
        return subprocess.run(cmd, capture_output=True, encoding='utf-8', errors='replace')
    
    def concat(self, input_files, output_path):
        """拼接视频（重编码确保一致）"""
        # 创建临时文件列表
        list_file = "concat_list.txt"
        with open(list_file, 'w') as f:
            for file in input_files:
                f.write(f"file '{os.path.abspath(file)}'\n")
        
        cmd = [
            self.ffmpeg, '-f', 'concat', '-safe', '0',
            '-i', list_file,
            '-c:v', 'libx264', '-crf', '18',
            '-c:a', 'aac', '-b:a', '320k',
            '-movflags', '+faststart',
            '-y', output_path
        ]
        result = subprocess.run(cmd, capture_output=True, encoding='utf-8', errors='replace')
        os.remove(list_file)
        return result
    
    def add_effect(self, input_path, output_path, filter_string):
        """添加视频效果"""
        cmd = [
            self.ffmpeg, '-i', input_path,
            '-vf', filter_string,
            '-c:v', 'libx264', '-crf', '18',
            '-c:a', 'copy',
            '-y', output_path
        ]
        return subprocess.run(cmd, capture_output=True, encoding='utf-8', errors='replace')
    
    def get_duration(self, filepath):
        """获取视频时长（秒）"""
        info = self.get_info(filepath)
        return float(info['format']['duration'])
```

### 6.2 FFmpeg 使用注意

> [!important] Windows/PowerShell 环境注意事项
> 1. 子进程调用时指定 `encoding='utf-8', errors='replace'` 避免 GBK 解码错误
> 2. 拼接视频优先用 `filter_complex concat` 重编码，避免 `-c copy` 因 moov atom 损坏失败
> 3. 中间片段统一加 `-movflags +faststart` 确保可流式播放
> 4. PowerShell 中不要用 cmd 语法（`&&`、`for %f` 等），用 PowerShell 原生命令
