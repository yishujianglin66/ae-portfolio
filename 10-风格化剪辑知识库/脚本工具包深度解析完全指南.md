```javascript
var AEScriptTools = {
    version: '1.0.0',
    
    utils: {
        getActiveComp: function() {
            if (app.project.activeItem instanceof CompItem) {
                return app.project.activeItem;
            }
            return null;
        },
        
        getSelectedLayers: function(comp) {
            var comp = comp || this.getActiveComp();
            if (!comp) return [];
            
            var selected = [];
            for (var i = 1; i <= comp.numLayers; i++) {
                if (comp.layer(i).selected) {
                    selected.push(comp.layer(i));
                }
            }
            return selected;
        },
        
        createComp: function(name, width, height, frameRate, duration) {
            return app.project.items.addComp(
                name, 
                width, 
                height, 
                1, 
                duration, 
                frameRate
            );
        },
        
        importFootage: function(filePath) {
            var file = new File(filePath);
            if (file.exists) {
                return app.project.importFile(file);
            }
            return null;
        },
        
        addTextLayer: function(comp, text, options) {
            var textLayer = comp.layers.addText(text);
            
            if (options) {
                if (options.position) {
                    textLayer.position.setValue(options.position);
                }
                if (options.fontSize) {
                    textLayer.sourceText.style.fontSize = options.fontSize;
                }
                if (options.fontFamily) {
                    textLayer.sourceText.style.fontFamily = options.fontFamily;
                }
                if (options.fillColor) {
                    textLayer.sourceText.style.fillColor = options.fillColor;
                }
            }
            
            return textLayer;
        },
        
        addShapeLayer: function(comp, shapeType, options) {
            var shapeLayer = comp.layers.addShape();
            var group = shapeLayer.property("Contents").addProperty("ADBE Vector Group");
            
            if (shapeType === 'rectangle') {
                var rect = group.property("Contents").addProperty("ADBE Vector Shape - Rect");
                if (options && options.size) {
                    rect.property("ADBE Rect Size").setValue(options.size);
                }
            } else if (shapeType === 'ellipse') {
                var ellipse = group.property("Contents").addProperty("ADBE Vector Shape - Ellipse");
                if (options && options.size) {
                    ellipse.property("ADBE Ellipse Size").setValue(options.size);
                }
            } else if (shapeType === 'path') {
                var path = group.property("Contents").addProperty("ADBE Vector Shape - Path");
                if (options && options.pathPoints) {
                    path.property("ADBE Vector Path").setValue(options.pathPoints);
                }
            }
            
            if (options && options.fillColor) {
                var fill = group.property("Contents").addProperty("ADBE Vector Graphic - Fill");
                fill.property("ADBE Fill Color").setValue(options.fillColor);
            }
            
            return shapeLayer;
        },
        
        addEffect: function(layer, effectName, properties) {
            var effect = layer.effects.add(effectName);
            
            if (properties) {
                for (var propName in properties) {
                    if (effect.property(propName)) {
                        effect.property(propName).setValue(properties[propName]);
                    }
                }
            }
            
            return effect;
        },
        
        setKeyframe: function(property, time, value) {
            property.setValueAtTime(time, value);
            property.setInterpolationTypeAtKey(time, KeyframeInterpolationType.BEZIER);
        },
        
        copyKeyframes: function(sourceProp, targetProp) {
            sourceProp.copyKeyframes();
            targetProp.pasteKeyframes();
        },
        
        createNullObject: function(comp, name) {
            return comp.layers.addNull(name);
        },
        
        parentLayer: function(childLayer, parentLayer) {
            childLayer.parent = parentLayer;
        },
        
        createPrecomp: function(comp, layers, name) {
            return comp.layers.precompose(layers, name, true);
        },
        
        renderComp: function(comp, outputPath, presetName) {
            var queue = app.project.renderQueue.items.add(comp);
            var renderItem = queue.item(queue.numItems);
            
            renderItem.outputModule(1).applyTemplate(presetName);
            renderItem.outputModule(1).file = new File(outputPath);
            
            queue.render();
        },
        
        exportFrame: function(comp, time, outputPath) {
            comp.saveFrameToFile(time, new File(outputPath));
        },
        
        batchProcess: function(items, processor) {
            items.forEach(function(item) {
                try {
                    processor(item);
                } catch (e) {
                    $.writeln('Error processing ' + item.name + ': ' + e.message);
                }
            });
        },
        
        log: function(message, level) {
            var level = level || 'info';
            $.writeln('[' + level.toUpperCase() + '] ' + message);
        },
        
        error: function(message) {
            this.log(message, 'error');
            throw new Error(message);
        }
    },
    
    animation: {
        createKeyframeAnimation: function(property, keyframes) {
            keyframes.forEach(function(keyframe) {
                AEScriptTools.utils.setKeyframe(
                    property, 
                    keyframe.time, 
                    keyframe.value
                );
            });
        },
        
        createLoopAnimation: function(property, duration, startValue, endValue) {
            var keyframes = [
                { time: 0, value: startValue },
                { time: duration, value: endValue }
            ];
            this.createKeyframeAnimation(property, keyframes);
            
            property.setLoopOutType(LoopOutType.CYCLE);
        },
        
        createPulseAnimation: function(property, baseValue, amplitude, frequency) {
            var expression = baseValue + ' + ' + amplitude + ' * Math.sin(time * ' + frequency + ')';
            property.expression = expression;
        },
        
        createWiggleAnimation: function(property, frequency, amplitude) {
            property.expression = 'wiggle(' + frequency + ', ' + amplitude + ')';
        },
        
        createMotionPath: function(layer, points, duration) {
            var keyframes = [];
            var segmentDuration = duration / (points.length - 1);
            
            points.forEach(function(point, index) {
                keyframes.push({
                    time: index * segmentDuration,
                    value: point
                });
            });
            
            this.createKeyframeAnimation(layer.position, keyframes);
        },
        
        createScaleAnimation: function(layer, startScale, endScale, duration) {
            var keyframes = [
                { time: 0, value: startScale },
                { time: duration, value: endScale }
            ];
            this.createKeyframeAnimation(layer.scale, keyframes);
        },
        
        createOpacityAnimation: function(layer, startOpacity, endOpacity, duration) {
            var keyframes = [
                { time: 0, value: startOpacity },
                { time: duration, value: endOpacity }
            ];
            this.createKeyframeAnimation(layer.opacity, keyframes);
        },
        
        createRotationAnimation: function(layer, startAngle, endAngle, duration) {
            var keyframes = [
                { time: 0, value: startAngle },
                { time: duration, value: endAngle }
            ];
            this.createKeyframeAnimation(layer.rotation, keyframes);
        }
    },
    
    effects: {
        applyColorCorrection: function(layer, settings) {
            var effect = AEScriptTools.utils.addEffect(layer, 'Lumetri Color', settings);
            return effect;
        },
        
        applyBlur: function(layer, blurAmount) {
            return AEScriptTools.utils.addEffect(layer, 'Gaussian Blur', {
                'Blurriness': blurAmount
            });
        },
        
        applyGlow: function(layer, glowColor, glowIntensity) {
            return AEScriptTools.utils.addEffect(layer, 'Glow', {
                'Glow Color': glowColor,
                'Glow Intensity': glowIntensity
            });
        },
        
        applyShadow: function(layer, shadowColor, shadowDistance) {
            return AEScriptTools.utils.addEffect(layer, 'Drop Shadow', {
                'Shadow Color': shadowColor,
                'Distance': shadowDistance
            });
        },
        
        applyNoise: function(layer, noiseAmount) {
            return AEScriptTools.utils.addEffect(layer, 'Noise', {
                'Amount of Noise': noiseAmount
            });
        },
        
        applySharpen: function(layer, sharpenAmount) {
            return AEScriptTools.utils.addEffect(layer, 'Unsharp Mask', {
                'Amount': sharpenAmount
            });
        },
        
        applyDistort: function(layer, effectName, settings) {
            return AEScriptTools.utils.addEffect(layer, effectName, settings);
        },
        
        applyTextAnimation: function(textLayer, animationType, settings) {
            switch (animationType) {
                case 'fadeIn':
                    this.createFadeIn(textLayer, settings.duration);
                    break;
                case 'slideIn':
                    this.createSlideIn(textLayer, settings.direction, settings.duration);
                    break;
                case 'scaleIn':
                    this.createScaleIn(textLayer, settings.duration);
                    break;
                case 'rotateIn':
                    this.createRotateIn(textLayer, settings.duration);
                    break;
            }
        },
        
        createFadeIn: function(layer, duration) {
            AEScriptTools.animation.createOpacityAnimation(layer, 0, 100, duration);
        },
        
        createSlideIn: function(layer, direction, duration) {
            var startPos, endPos;
            var comp = layer.parentComp;
            
            switch (direction) {
                case 'left':
                    startPos = [-comp.width / 2, layer.position.value[1]];
                    endPos = [layer.position.value[0], layer.position.value[1]];
                    break;
                case 'right':
                    startPos = [comp.width / 2, layer.position.value[1]];
                    endPos = [layer.position.value[0], layer.position.value[1]];
                    break;
                case 'top':
                    startPos = [layer.position.value[0], -comp.height / 2];
                    endPos = [layer.position.value[0], layer.position.value[1]];
                    break;
                case 'bottom':
                    startPos = [layer.position.value[0], comp.height / 2];
                    endPos = [layer.position.value[0], layer.position.value[1]];
                    break;
            }
            
            layer.position.setValueAtTime(0, startPos);
            layer.position.setValueAtTime(duration, endPos);
        },
        
        createScaleIn: function(layer, duration) {
            AEScriptTools.animation.createScaleAnimation(layer, [0, 0], [100, 100], duration);
        },
        
        createRotateIn: function(layer, duration) {
            AEScriptTools.animation.createRotationAnimation(layer, 360, 0, duration);
        }
    },
    
    puppeteering: {
        createPuppetPin: function(layer, position) {
            var puppet = AEScriptTools.utils.addEffect(layer, 'Puppet');
            var pin = puppet.property('Pins').addProperty('ADBE Puppet Pin');
            
            pin.property('Position').setValue(position);
            
            return pin;
        },
        
        addPuppetPin: function(puppetEffect, position) {
            var pin = puppetEffect.property('Pins').addProperty('ADBE Puppet Pin');
            pin.property('Position').setValue(position);
            return pin;
        },
        
        setPuppetKeyframe: function(pin, time, position) {
            pin.property('Position').setValueAtTime(time, position);
        },
        
        createPuppetAnimation: function(layer, keyframes) {
            var puppet = AEScriptTools.utils.addEffect(layer, 'Puppet');
            
            keyframes.forEach(function(keyframe) {
                keyframe.pins.forEach(function(pinData) {
                    var pin = puppet.property('Pins').property(pinData.index);
                    pin.property('Position').setValueAtTime(keyframe.time, pinData.position);
                });
            });
            
            return puppet;
        },
        
        createCharacterRig: function(layer, rigPoints) {
            var puppet = AEScriptTools.utils.addEffect(layer, 'Puppet');
            
            var pins = [];
            rigPoints.forEach(function(point) {
                var pin = puppet.property('Pins').addProperty('ADBE Puppet Pin');
                pin.property('Position').setValue(point.position);
                pin.name = point.name;
                pins.push(pin);
            });
            
            return pins;
        }
    }
};
```

```javascript
function batchApplyEffectToSelectedLayers(effectName, properties) {
    var comp = AEScriptTools.utils.getActiveComp();
    if (!comp) {
        AEScriptTools.utils.error('请选择一个合成');
        return;
    }
    
    var selectedLayers = AEScriptTools.utils.getSelectedLayers(comp);
    
    if (selectedLayers.length === 0) {
        AEScriptTools.utils.error('请选择至少一个图层');
        return;
    }
    
    selectedLayers.forEach(function(layer) {
        AEScriptTools.utils.addEffect(layer, effectName, properties);
        AEScriptTools.utils.log('Applied ' + effectName + ' to ' + layer.name);
    });
    
    AEScriptTools.utils.log('Batch completed: ' + selectedLayers.length + ' layers processed');
}

function batchRenderComps(outputFolder, presetName) {
```

```python
class ResolveScriptTools:
    VERSION = '1.0.0'
    
    def __init__(self, resolve):
        self.resolve = resolve
        self.project_manager = resolve.GetProjectManager()
        self.project = None
        self.timeline = None
        
    def get_current_project(self):
        self.project = self.project_manager.GetCurrentProject()
        return self.project
    
    def get_current_timeline(self):
        if not self.project:
            self.get_current_project()
        self.timeline = self.project.GetTimelineByIndex(self.project.GetTimelineCount())
        return self.timeline
    
    def create_project(self, name):
        self.project_manager.CreateProject(name)
        self.project = self.project_manager.GetCurrentProject()
        return self.project
    
    def import_media(self, file_paths):
        if not self.project:
            self.get_current_project()
            
        media_pool = self.project.GetMediaPool()
        imported = media_pool.ImportMedia(file_paths)
        return imported
    
    def create_timeline(self, name, video_track_count=1, audio_track_count=1):
        if not self.project:
            self.get_current_project()
            
        media_pool = self.project.GetMediaPool()
        timeline = media_pool.CreateTimelineFromClips(name, [])
        
        for i in range(video_track_count - 1):
            timeline.InsertTrack('video', i + 1)
        for i in range(audio_track_count - 1):
            timeline.InsertTrack('audio', i + 1)
            
        self.timeline = timeline
        return timeline
    
    def add_clip_to_timeline(self, media_item, track_index=1, start_frame=0):
        if not self.timeline:
            self.get_current_timeline()
            
        timeline = self.timeline
        timeline.InsertClips([media_item], 'video', track_index, start_frame)
        
        return timeline
    
    def apply_color_grade(self, clip_index, track_index=1, settings=None):
        if not self.timeline:
            self.get_current_timeline()
            
        clip = self.timeline.GetItemListInTrack('video', track_index)[clip_index - 1]
        
        if settings:
            for setting_name, setting_value in settings.items():
                clip.SetClipColorCorrectorParam(setting_name, setting_value)
        
        return clip
    
    def apply_lut(self, clip_index, track_index=1, lut_path=None):
        if not self.timeline:
            self.get_current_timeline()
            
        clip = self.timeline.GetItemListInTrack('video', track_index)[clip_index - 1]
        
        if lut_path:
            clip.SetClipLUT(lut_path)
        
        return clip
    
    def apply_effect(self, clip_index, track_index=1, effect_name=None, parameters=None):
        if not self.timeline:
            self.get_current_timeline()
            
        clip = self.timeline.GetItemListInTrack('video', track_index)[clip_index - 1]
        
        if effect_name:
            clip.AddEffect(effect_name)
            
            if parameters:
                for param_name, param_value in parameters.items():
                    clip.SetEffectParam(effect_name, param_name, param_value)
        
        return clip
    
    def set_clip_speed(self, clip_index, track_index=1, speed=100):
        if not self.timeline:
            self.get_current_timeline()
            
        clip = self.timeline.GetItemListInTrack('video', track_index)[clip_index - 1]
        clip.SetClipSpeed(speed)
        
        return clip
    
    def set_in_out_points(self, clip_index, track_index=1, in_point=0, out_point=100):
        if not self.timeline:
            self.get_current_timeline()
            
        clip = self.timeline.GetItemListInTrack('video', track_index)[clip_index - 1]
        clip.SetClipInPoint(in_point)
        clip.SetClipOutPoint(out_point)
        
        return clip
    
    def add_marker(self, timecode, name=None, color=None):
        if not self.timeline:
            self.get_current_timeline()
            
        self.timeline.AddMarker(timecode, name, color)
        
        return True
    
    def export_timeline(self, output_path, preset_name=None):
        if not self.project:
            self.get_current_project()
            
        render_settings = self.project.GetRenderSettings()
        
        if preset_name:
            render_settings.LoadPreset(preset_name)
        
        render_settings.SetOutputPath(output_path)
        
        self.project.StartRendering()
        
        return True
    
    def create_fusion_comp(self, name):
        if not self.timeline:
            self.get_current_timeline()
            
        fusion_comp = self.timeline.AddFusionComp()
        fusion_comp.SetName(name)
        
        return fusion_comp
    
    def add_fusion_node(self, comp, node_type):
        return comp.AddTool(node_type)
    
    def connect_fusion_nodes(self, from_node, from_output, to_node, to_input):
        from_node.ConnectOutput(from_output, to_node, to_input)
        return True
    
    def batch_process_clips(self, processor, track_index=1):
        if not self.timeline:
            self.get_current_timeline()
            
        clips = self.timeline.GetItemListInTrack('video', track_index)
        
        for clip in clips:
            try:
                processor(clip)
            except Exception as e:
                print(f"Error processing {clip.GetName()}: {str(e)}")
                
        return len(clips)
    
    def log(self, message, level='info'):
        print(f"[{level.upper()}] {message}")
        
    def error(self, message):
        self.log(message, 'error')
        raise Exception(message)
```

```javascript
var PPScriptTools = {
    version: '1.0.0',
    
    utils: {
        getActiveSequence: function() {
            return PP.project.activeSequence;
        },
        
        getSelectedClips: function(sequence) {
            var sequence = sequence || this.getActiveSequence();
            if (!sequence) return [];
            
            var selected = [];
            for (var i = 0; i < sequence.videoTracks.length; i++) {
                var track = sequence.videoTracks[i];
                for (var j = 0; j < track.clips.length; j++) {
                    if (track.clips[j].selected) {
                        selected.push(track.clips[j]);
                    }
                }
            }
            return selected;
        },
        
        createSequence: function(name, settings) {
            return PP.project.sequences.createNewSequence(name, settings);
        },
        
        importMedia: function(filePath) {
            var file = new File(filePath);
            if (file.exists) {
                return PP.project.rootItem.children.add(file);
            }
            return null;
        },
        
        addClipToTrack: function(sequence, footage, trackIndex, startTime) {
            var track = sequence.videoTracks[trackIndex] || sequence.videoTracks.add();
            var clip = track.clips.add(footage);
            
            if (startTime !== undefined) {
                clip.startTime = startTime;
            }
            
            return clip;
        },
        
        applyEffect: function(clip, effectName, properties) {
            var effect = clip.effects.add(effectName);
            
            if (properties) {
                for (var propName in properties) {
                    if (effect.properties[propName]) {
                        effect.properties[propName].setValue(properties[propName]);
                    }
                }
            }
            
            return effect;
        },
        
        addTransition: function(clip1, clip2, transitionType, duration) {
            var transition = clip1.videoTransitions.add(clip2);
            transition.transitionType = transitionType;
            
            if (duration) {
                transition.duration = duration;
            }
            
            return transition;
        },
        
        setClipSpeed: function(clip, speed) {
            clip.speed = speed;
        },
        
        trimClip: function(clip, inPoint, outPoint) {
            clip.inPoint = inPoint;
            clip.outPoint = outPoint;
        },
        
        addMarker: function(sequence, time, name, color) {
            var marker = sequence.markers.add(time);
            marker.name = name || '';
            marker.color = color || 'Red';
            return marker;
        },
        
        exportSequence: function(sequence, outputPath, presetName) {
            var desc = new ActionDescriptor();
            var targetRef = new ActionReference();
            targetRef.putEnumerated(charIDToTypeID('Seq '), charIDToTypeID('Ordn'), charIDToTypeID('Trgt'));
            desc.putReference(charIDToTypeID('null'), targetRef);
            
            var pathDesc = new ActionDescriptor();
            pathDesc.putPath(charIDToTypeID('null'), new File(outputPath));
            desc.putObject(charIDToTypeID('T   '), charIDToTypeID('Pth '), pathDesc);
            
            var presetRef = new ActionReference();
            presetRef.putEnumerated(charIDToTypeID('Prst'), charIDToTypeID('Prst'), charIDToTypeID(presetName));
            desc.putReference(charIDToTypeID('Prst'), presetRef);
            
            executeAction(charIDToTypeID('Expt'), desc, DialogModes.NO);
```
