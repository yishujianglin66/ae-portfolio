```python
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from contextlib import asynccontextmanager

from app.api.v1.api import api_router
from app.core.config import settings
from app.db.session import engine
from app.db.base import Base

@asynccontextmanager
async def lifespan(app: FastAPI):
    # 启动时
    # 创建数据库表
    Base.metadata.create_all(bind=engine)
    
    # 初始化引擎池
    from app.engines.pool import engine_pool
    engine_pool.initialize()
    
    yield
    
    # 关闭时
    # 清理资源
    engine_pool.shutdown()

app = FastAPI(

    version="2.0.0",
    lifespan=lifespan,

    allow_methods=["*"],

)

# API路由
```

### 3.2 核心数据模型

```python
# app/db/models/task.py
from sqlalchemy import Column, Integer, String, Float, DateTime, JSON, Enum
from sqlalchemy.sql import func
from app.db.base import Base
import enum

class TaskStatus(str, enum.Enum):
    PENDING = "pending"
    QUEUED = "queued"
    PROCESSING = "processing"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"

class TaskPhase(str, enum.Enum):
    PREPROCESS = "preprocess"
    ROTO_TRACK = "roto_track"
    STYLIZE = "stylize"
    RENDER = "render"
    QUALITY_CHECK = "quality_check"

class Task(Base):
    """任务表"""
    __tablename__ = "tasks"
    
    id = Column(Integer, primary_key=True, index=True)
    task_id = Column(String, unique=True, index=True)
    name = Column(String, index=True)
    status = Column(Enum(TaskStatus), default=TaskStatus.PENDING)
    current_phase = Column(Enum(TaskPhase), default=TaskPhase.PREPROCESS)
    priority = Column(Integer, default=5)
    
    # 输入输出
    input_video_path = Column(String)
    output_path = Column(String)
    
    # 配置参数
    style_config = Column(JSON)  # 风格化配置
    effects_config = Column(JSON)  # 效果配置
    
    # 进度
    progress = Column(Float, default=0.0)
    progress_message = Column(String, default="")
    
    # 时间
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    started_at = Column(DateTime(timezone=True))
    completed_at = Column(DateTime(timezone=True))
    estimated_duration = Column(Float)  # 预计时长（秒）
    
    # 质量报告
    quality_report = Column(JSON)
    
    # 错误信息
    error_message = Column(String)
    error_trace = Column(String)
    
    # 用户
    user_id = Column(String, index=True)
    
    # 元数据
    metadata = Column(JSON, default={})
```

### 3.3 Celery 任务定义

```python
# app/worker.py
from celery import Celery
from app.core.config import settings

celery_app = Celery(
    "puppet_worker",
    broker=settings.REDIS_URL,
    backend=settings.REDIS_URL,
)

# 路由配置
celery_app.conf.task_routes = {
    "app.tasks.preprocess.*": {"queue": "preprocess"},
    "app.tasks.roto.*": {"queue": "roto"},
    "app.tasks.stylize.*": {"queue": "stylize"},
    "app.tasks.render.*": {"queue": "render"},
}

# 自动发现任务
celery_app.autodiscover_tasks(["app.tasks"])

# app/tasks/preprocess.py
from app.worker import celery_app
from app.services.preprocess import PreprocessService

@celery_app.task(bind=True, name="app.tasks.preprocess.process_video")
def preprocess_video(self, task_id: str, video_path: str, config: dict):
    """预处理任务"""
    service = PreprocessService()
    
    try:
        # 更新状态
        service.update_task_status(task_id, "processing", phase="preprocess")
        
        # 执行预处理
        result = service.process(task_id, video_path, config)
        
        # 更新状态
        service.update_task_status(task_id, "completed", progress=100)
        
        # 触发下一阶段
        from app.tasks.roto import start_roto_track
        start_roto_track.delay(task_id, result)
        
        return result
        
    except Exception as e:
        service.handle_error(task_id, str(e))
        self.retry(exc=e, countdown=60, max_retries=3)
```

### 5.2 告警规则

| 告警级别 | 触发条件 | 通知方式 | 响应时间 |
|---------|---------|---------|---------|
| **P0 紧急** | 核心服务宕机 / 数据丢失 | 电话 + 短信 + 企业微信 | 15分钟内 |
| **P1 严重** | 任务失败率 > 5% / 队列积压 > 100 | 企业微信 + 短信 | 1小时内 |
| **P2 警告** | 磁盘 > 80% / 响应时间 > 2s | 企业微信 | 4小时内 |
| **P3 提示** | 非核心指标异常 | 邮件 / 日报 | 次日处理 |

```
          │
          │ 共享存储
    ┌─────▼───────────────────────────────────────┐
    │  Storage Class                              │
    │  - Rook/Ceph (块存储)                        │
    │  - MinIO (对象存储)                          │
    │  - NFS/EFS (共享文件)                        │
    └─────────────────────────────────────────────┘
```

## 八、CI/CD 流水线

### 8.1 GitHub Actions 工作流

```yaml
# .github/workflows/ci.yml
name: CI/CD Pipeline

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

jobs:
  # 代码检查
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.10'
      - run: pip install -r backend/requirements-dev.txt
      - run: flake8 backend/
      - run: mypy backend/
      - run: black --check backend/

  # 单元测试
  test:
    runs-on: ubuntu-latest
    needs: lint
    services:
      postgres:
        image: postgres:15
        env:
          POSTGRES_USER: test
          POSTGRES_PASSWORD: test
          POSTGRES_DB: test_db
        ports:
          - 5432:5432
      redis:
        image: redis
        ports:
          - 6379:6379
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.10'
      - run: pip install -r backend/requirements.txt
      - run: pytest backend/tests/ -v --cov=backend/app

  # 构建镜像
  build:
    runs-on: ubuntu-latest
    needs: test
    if: github.ref == 'refs/heads/main'
    steps:
      - uses: actions/checkout@v4
      - uses: docker/setup-buildx-action@v3
      - uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}
      - uses: docker/build-push-action@v5
        with:
          context: ./backend
          push: true
          tags: ghcr.io/${{ github.repository }}/backend:latest
          cache-from: type=gha
          cache-to: type=gha,mode=max

  # 部署
  deploy:
    runs-on: ubuntu-latest
    needs: build
    if: github.ref == 'refs/heads/main'
    environment: production
    steps:
      - name: Deploy to K8s
        run: |
          kubectl set image deployment/backend \
            backend=ghcr.io/${{ github.repository }}/backend:latest
          kubectl rollout status deployment/backend
```

## 九、容量规划与扩展

### 9.1 容量预估

| 指标 | 初期 | 中期(6个月) | 长期(12个月) |
|------|------|------------|-------------|
| **日任务量** | 10-50 | 100-500 | 1000+ |
| **并发任务** | 5-10 | 30-50 | 100+ |
| **存储需求** | 1TB | 10TB | 100TB+ |
| **GPU节点** | 1-2 | 4-8 | 20+ |
| **渲染节点** | 1-2 | 4-6 | 15+ |
| **用户数** | 5-10 | 50-100 | 500+ |

### 9.2 扩展策略

| 扩展方式 | 触发条件 | 操作 |
|---------|---------|------|
| **水平扩展** | 队列积压 > 阈值 | 增加Worker / 渲染节点 |
| **垂直扩展** | 单任务资源不足 | 升级节点配置 |
| **存储扩展** | 使用率 > 80% | 增加存储节点 / 扩容 |
| **数据库扩展** | 慢查询增加 | 读写分离 / 分库分表 |
