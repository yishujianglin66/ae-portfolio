## 第二章 运动原子分析

### 2.1 整体运动强度

**算法**：稠密光流（Farneback）

```python
import cv2
import numpy as np

cap = cv2.VideoCapture("clip.mp4")
prev_frame = None
motion_curve = []

while True:
    ret, frame = cap.read()
    if not ret: break
    
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    if prev_frame is not None:
        # 计算光流
        flow = cv2.calcOpticalFlowFarneback(prev_frame, gray, None, 0.5, 3, 15, 3, 5, 1.2, 0)
        magnitude = np.sqrt(flow[..., 0]**2 + flow[..., 1]**2)
        avg_magnitude = magnitude.mean()
        motion_curve.append(avg_magnitude)
    
    prev_frame = gray

# 归一化到 0-1
motion_intensity = np.array(motion_curve)
motion_intensity = (motion_intensity - motion_intensity.min()) / (motion_intensity.max() - motion_intensity.min() + 1e-6)

overall_motion = float(motion_intensity.mean())
```

| 0-0.2 | 静止 | 几乎无运动 | 前奏/收尾 |
| 0.2-0.4 | 低运动 | 缓慢运动 | 主歌铺垫 |

### 2.4 相机运动分类

**算法**：基于特征点匹配的全局运动估计

```python
def classify_camera_motion(flow):
    # 计算全局运动向量
    avg_dx = flow[..., 0].mean()
    avg_dy = flow[..., 1].mean()
    
    # 检测缩放
    center = (flow.shape[1] // 2, flow.shape[0] // 2)
    radial_flow = np.zeros_like(flow)
    for y in range(0, flow.shape[0], 10):
        for x in range(0, flow.shape[1], 10):
            dx = flow[y, x, 0]
            dy = flow[y, x, 1]
            radial_x = x - center[0]
            radial_y = y - center[1]
            # 投影到径向方向
            norm = np.sqrt(radial_x**2 + radial_y**2) + 1e-6
            radial_flow[y, x] = (dx * radial_x + dy * radial_y) / norm
    
    zoom_factor = radial_flow.mean()
    
    if abs(zoom_factor) > 0.5:
        return "zoom_in" if zoom_factor > 0 else "zoom_out"
    elif abs(avg_dx) > abs(avg_dy) and abs(avg_dx) > 0.5:
        return "pan_right" if avg_dx > 0 else "pan_left"
    elif abs(avg_dy) > 0.5:
        return "tilt_up" if avg_dy < 0 else "tilt_down"
    else:
        return "static"
```

| 相机运动 | 推荐效果叠加 |
|---------|------------|
| static | 静态画面适合加文字/图形 |
| pan_left/right | 适合拉镜转场 |
| tilt_up/down | 适合垂直滑入转场 |
| zoom_in | 适合爆发性效果 |
| zoom_out | 适合收尾效果 |
| handheld | 适合稳定器+故障 |

## 第三章 色彩原子分析

### 3.1 主色板提取

**算法**：K-Means 聚类

```python
from sklearn.cluster import KMeans

def extract_palette(video_path, n_colors=5, sample_rate=30):
    cap = cv2.VideoCapture(video_path)
    pixels = []
    frame_count = 0
    
    while True:
        ret, frame = cap.read()
        if not ret: break
        frame_count += 1
        if frame_count % sample_rate != 0: continue
        
        # 转换到 Lab 颜色空间（更符合人感）
        lab = cv2.cvtColor(frame, cv2.COLOR_BGR2LAB)
        pixels.extend(lab.reshape(-1, 3)[::100])  # 采样
    
    pixels = np.array(pixels)
    kmeans = KMeans(n_clusters=n_colors, random_state=42)
    kmeans.fit(pixels)
    
    # 提取主色
    palette = []
    for center in kmeans.cluster_centers_:
        # Lab → RGB → HEX
        rgb = cv2.cvtColor(np.uint8([[center]]), cv2.COLOR_LAB2RGB)[0][0]
        hex_color = '#{:02x}{:02x}{:02x}'.format(*rgb)
        palette.append(hex_color)
    
    return palette
```

### 3.2 冷暖调判断

```python
def color_temperature(palette_rgb):
    # 计算 R-B 平均差
    r_avg = np.mean([c[0] for c in palette_rgb])
    b_avg = np.mean([c[2] for c in palette_rgb])
    
    if r_avg - b_avg > 30:
        return "warm"
    elif b_avg - r_avg > 30:
        return "cool"
    else:
        return "neutral"
```

| 色温 | RGB 特征 | 适用风格 |
|------|---------|---------|
| warm | R > B+30 | 复古/暖色电影感 |
| cool | B > R+30 | 赛博朋克/冷色梦幻 |
| neutral | |B-R|<30 | 纪实/中性 |

### 3.3 饱和度/亮度/对比度

```python
def compute_color_stats(video_path, sample_rate=30):
    saturations, brightnesses = [], []
    cap = cv2.VideoCapture(video_path)
    frame_count = 0
    
    while True:
        ret, frame = cap.read()
        if not ret: break
        frame_count += 1
        if frame_count % sample_rate != 0: continue
        
        hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
        saturations.append(hsv[..., 1].mean())
        brightnesses.append(hsv[..., 2].mean())
    
    return {
        "saturation": float(np.mean(saturations) / 255),
        "brightness": float(np.mean(brightnesses) / 255),
        "contrast": float(np.std(brightnesses) / 255)
    }
```

## 第四章 构图原子分析

### 4.1 三分法符合度

```python
def rule_of_thirds_score(frame):
    h, w = frame.shape[:2]
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    # 计算显著图（简化版：用边缘强度代替）
    edges = cv2.Canny(gray, 50, 150)
    
    # 三分线位置
    thirds_x = [w // 3, 2 * w // 3]
    thirds_y = [h // 3, 2 * h // 3]
    
    # 计算三分线附近的能量占比
    band_width = min(w, h) // 20
    third_energy = 0
    for x in thirds_x:
        third_energy += edges[:, max(0, x-band_width):x+band_width].sum()
    for y in thirds_y:
        third_energy += edges[max(0, y-band_width):y+band_width, :].sum()
    
    total_energy = edges.sum() + 1e-6
    score = third_energy / total_energy
    return float(score)
```

| 评分 | 描述 |
|------|------|
| >0.6 | 主体位于三分点，构图良好 |
| 0.4-0.6 | 构图中等 |
| <0.4 | 主体偏离，构图差 |

### 4.2 对称性

```python
def symmetry_score(frame):
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    h, w = gray.shape
    
    # 水平对称
    left = gray[:, :w//2]
    right = cv2.flip(gray[:, w//2:], 1)
    h_diff = np.abs(left.astype(float) - right.astype(float)).mean()
    h_score = 1 - h_diff / 255
    
    # 垂直对称
    top = gray[:h//2, :]
    bottom = cv2.flip(gray[h//2:, :], 0)
    v_diff = np.abs(top.astype(float) - bottom.astype(float)).mean()
    v_score = 1 - v_diff / 255
    
    return float(max(h_score, v_score))
```

### 4.3 景深判断

```python
def depth_of_field(frame):
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    # 计算图像不同区域的清晰度（拉普拉斯方差）
    h, w = gray.shape
    center = gray[h//4:3*h//4, w//4:3*w//4]
    edges_region = np.concatenate([
        gray[:h//4, :].ravel(),
        gray[3*h//4:, :].ravel(),
        gray[:, :w//4].ravel(),
        gray[:, 3*w//4:].ravel()
    ])
    
    # 中心清晰度
    center_blur = cv2.Laplacian(center, cv2.CV_64F).var()
    # 边缘清晰度
    edge_blur = cv2.Laplacian(edges_region.reshape(1, -1), cv2.CV_64F).var()
    
    if center_blur > edge_blur * 2:
        return "shallow"  # 浅景深
    elif center_blur > edge_blur:
        return "medium"
    else:
        return "deep"      # 深景深
```

### 4.4 主体位置

```python
def detect_subject_position(frame):
    # 用显著性检测或简单的中心检测
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    # 用最大轮廓的中心
    _, thresh = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    contours, _ = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    
    if not contours:
        return [0.5, 0.5]  # 默认中心
    
    largest = max(contours, key=cv2.contourArea)
    M = cv2.moments(largest)
    if M["m00"] == 0:
        return [0.5, 0.5]
    
    cx = M["m10"] / M["m00"] / frame.shape[1]
    cy = M["m01"] / M["m00"] / frame.shape[0]
    return [float(cx), float(cy)]
```

## 第五章 能量原子分析

### 5.1 综合能量值

片段能量是运动+色彩+对比度的综合：

```python
def compute_clip_energy(motion_intensity, color_stats):
    # 归一化各分量到 0-1
    motion = motion_intensity
    saturation = color_stats["saturation"]
    contrast = color_stats["contrast"]
    
    # 加权求和
    energy = motion * 0.5 + saturation * 0.25 + contrast * 0.25
    return float(np.clip(energy, 0, 1))
```

### 5.2 能量曲线

```python
def compute_energy_curve(motion_curve, brightness_curve):
    # 运动归一化
    motion = np.array(motion_curve)
    motion = (motion - motion.min()) / (motion.max() - motion.min() + 1e-6)
    
    # 亮度方差作为对比度指标
    brightness = np.array(brightness_curve)
    brightness_var = np.abs(np.diff(brightness))
    
    # 综合曲线
    min_len = min(len(motion), len(brightness_var))
    energy = motion[:min_len] * 0.7 + brightness_var[:min_len] / 255 * 0.3
    return energy.tolist()
```

### 5.3 峰值时刻

```python
from scipy.signal import find_peaks

def find_energy_peaks(energy_curve, sample_rate=30):
    peaks, _ = find_peaks(energy_curve, height=0.6, prominence=0.2, distance=15)
    
    return [{
        "time": float(p / sample_rate),
        "intensity": float(energy_curve[p])
    } for p in peaks]
```

### 6.3 情绪标签

基于色彩+运动+构图的组合推断：

```python
def infer_mood(clip_atom):
    mood = []
    
    # 冷调 + 低运动 + 浅景深 → 忧郁/沉思
    if clip_atom.color.temperature == "cool" and \
       clip_atom.motion.intensity < 0.3 and \
       clip_atom.composition.depth == "shallow":
        mood.extend(["melancholic", "contemplative"])
    
    # 暖调 + 中运动 + 高对比 → 温暖/活力
    elif clip_atom.color.temperature == "warm" and \
         0.3 <= clip_atom.motion.intensity <= 0.6:
        mood.extend(["warm", "energetic"])
    
    # 高运动 + 高饱和 → 激烈/兴奋
    elif clip_atom.motion.intensity > 0.7 and \
         clip_atom.color.saturation > 0.7:
        mood.extend(["intense", "exciting"])
    
    # 低对比 + 低饱和 + 静止 → 平静/空灵
    elif clip_atom.color.contrast < 0.4 and \
         clip_atom.color.saturation < 0.4 and \
         clip_atom.motion.intensity < 0.2:
        mood.extend(["calm", "ethereal"])
    
    else:
        mood.append("neutral")
    
    return mood
```
