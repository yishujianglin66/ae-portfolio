// AE 2026 兼容性检查
function checkAEVersion() {
    var version = app.version;
    if (version >= "26.0") {
        return "AE 2026 or later - Full support";
    } else if (version >= "25.0") {
        return "AE 2025 - Partial support";
    } else {
        return "Older version - Limited support";
    }
}

// 创建合成（AE 2026优化）
function createComp2026(name, width, height, duration, fps) {
    var comp = app.project.items.addComp(name, width, height, 1, duration, fps);
    // AE 2026新特性
    comp.displayStartTime = 0;
    comp.motionBlur = true;
    comp.frameBlending = true;
    return comp;
}

### 三、Null Controller 创建脚本（AE 2026适配）

```jsx
// AE 2026 Null Controller 创建脚本
app.beginUndoGroup("Create Null Controller 2026");

// 创建合成
var comp = createComp2026("Null_Controller_Training", 1920, 1080, 10, 30);

// 创建 Null 控制器
var nullLayer = comp.layers.addNull();
nullLayer.name = "Null_Controller";

// 添加控制效果（AE 2026新方式）
var effects = [
    {name: "缩放", type: "ADBE Slider Control", value: 100},
    {name: "速度", type: "ADBE Slider Control", value: 1},
    {name: "振幅", type: "ADBE Slider Control", value: 10},
    {name: "旋转", type: "ADBE Angle Control", value: 0},
    {name: "主色", type: "ADBE Color Control", value: [1, 0.5, 0.2]},
    {name: "副色", type: "ADBE Color Control", value: [0.2, 0.5, 1]},
    {name: "位置", type: "ADBE Point Control", value: [960, 540]},
    {name: "开关", type: "ADBE Checkbox Control", value: 1}
];

for (var i = 0; i < effects.length; i++) {
    var effect = nullLayer.Effects.addProperty(effects[i].type);
    effect.name = effects[i].name;
    effect.property(1).setValue(effects[i].value);
}

app.endUndoGroup();
```

### 四、一拳超人效果创建脚本（AE 2026适配）

```jsx
// AE 2026 一拳超人效果创建脚本
app.beginUndoGroup("Create One Punch Man Effect 2026");

// 创建合成
var comp = createComp2026("一拳超人_埼玉_重构版_v26", 1920, 1080, 17.6, 30);

// 导入视频
var videoFile = new File("D:\\AE-Work\\视频素材库\\抖音_一拳超人_埼玉.mp4");
if (videoFile.exists) {
    var footage = app.project.importFile(videoFile);
    var videoLayer = comp.layers.add(footage);
}

// 添加景深效果（AE 2026优化）
var cameraLayer = comp.layers.addCamera("Camera", [960, 540]);
cameraLayer.property("Zoom").setValue(2000);
cameraLayer.property("Depth of Field").setValue(true);
cameraLayer.property("Aperture").setValue(50);
cameraLayer.property("Focus Distance").setValue(1000);

// 添加发光效果
var glowEffect = videoLayer.Effects.addProperty("ADBE Glo2");
glowEffect.property("Glow Radius").setValue(20);
glowEffect.property("Glow Intensity").setValue(1.5);

// 添加暗角效果
var vignetteEffect = videoLayer.Effects.addProperty("ADBE Vignette");
vignetteEffect.property("Amount").setValue(30);

// 添加粒子效果（AE 2026增强）
var particleEffect = videoLayer.Effects.addProperty("CC Particle World");

// 添加调色效果
var colorEffect = videoLayer.Effects.addProperty("ADBE Color Balance");
colorEffect.property("Shadows Red Balance").setValue(-10);
colorEffect.property("Midtones Green Balance").setValue(5);

app.endUndoGroup();
```
