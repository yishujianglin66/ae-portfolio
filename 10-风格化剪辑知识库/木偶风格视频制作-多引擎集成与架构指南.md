# AE MCP 客户端架构
class AEMCPClient:
    """AE MCP 客户端 - 核心执行引擎"""
    
    def __init__(self, host="localhost", port=3000):
        self.host = host
        self.port = port
        self.atom_compiler = AtomCompiler()
        self.connection_pool = ConnectionPool(max_connections=3)
    
    def execute_atom_script(self, script_content: str) -> ExecuteResult:
        """执行原子脚本"""
        with self.connection_pool.get_connection() as conn:
            return conn.send_script(script_content)
    
    def apply_effect(self, comp_name: str, layer_name: str, 
                     effect_name: str, params: dict) -> bool:
        """应用效果到指定图层"""
        atom_script = self.atom_compiler.compile_effect(
            comp_name, layer_name, effect_name, params
        )
        result = self.execute_atom_script(atom_script)
        return result.success
    
    def batch_apply_effects(self, tasks: List[EffectTask]) -> BatchResult:
        """批量应用效果"""
        scripts = []
        for task in tasks:
            script = self.atom_compiler.compile_effect(
                task.comp_name, task.layer_name,
                task.effect_name, task.params
            )
            scripts.append(script)
        
        combined_script = "\n".join(scripts)
        return self.execute_atom_script(combined_script)

# Silhouette 执行器
class SilhouetteExecutor:
    """Silhouette 执行器 - 抠像与跟踪核心"""
    
    def __init__(self, silhouette_path: str = None):
        self.silhouette_path = silhouette_path or self._detect_silhouette()
        self.session = None
        self.node_graph = {}
    
    def create_roto_session(self, video_path: str, 
                            output_path: str) -> Session:
        """创建 Roto 抠像会话"""
        session = createObject("Session")
        session.setMediaPath(video_path)
        
        # 创建节点图
        source_node = createObject("SourceNode")
        source_node.property("mediaPath").setValue(video_path, 0)
        
        roto_node = createObject("RotoNode")
        roto_node.property("mode").setValue("foreground", 0)
        
        output_node = createObject("OutputNode")
        output_node.property("format").setValue("OpenEXR", 0)
        output_node.property("outputPath").setValue(output_path, 0)
        
        # 连接节点
        source_node.outputs[0].connect(roto_node.inputs[1])  # foreground
        roto_node.outputs[0].connect(output_node.inputs[0])
        
        self.session = session
        self.node_graph = {
            "source": source_node,
            "roto": roto_node,
            "output": output_node
        }
        
        return session
    
    def run_tracker(self, tracker_type: str = "point",
                    points: List[dict] = None) -> TrackResult:
        """运行跟踪器"""
        tracker = createObject("TrackerNode")
        tracker.property("trackerType").setValue(tracker_type, 0)
        
        for i, point in enumerate(points):
            tracker.property("points").addProperty("ADBE Point Tracker")
            tracker.property("points")[i].property("featureCenter").setValue(
                [point["x"], point["y"]], 0
            )
        
        # 执行跟踪
        tracker.property("trackForward").setValue(True, 0)
        
        return TrackResult(
            success=True,
            track_data=tracker.property("trackData").value
        )

# Topaz CLI 封装
class TopazEnhancer:
    """Topaz Video AI 画质增强器"""
    
    def __init__(self, topaz_cli_path: str = None):
        self.cli_path = topaz_cli_path or self._detect_topaz()
        self.models = {
            "proteus": "Proteus",
            "artemis": "Artemis",
            "gaia": "Gaia",
            "theia": "Theia"
        }
    
    def enhance_video(self, input_path: str, output_path: str,
                      model: str = "proteus",
                      scale: float = 2.0,
                      fps: int = None,
                      denoise: int = 50,
                      deblur: int = 30) -> EnhanceResult:
        """增强视频"""
        cmd = [
            self.cli_path,
            "-i", input_path,
            "-o", output_path,
            "-m", self.models[model],
            "-s", str(scale)
        ]
        
        if fps:
            cmd.extend(["-f", str(fps)])
        
        # 高级参数
        cmd.extend([
            "--denoise", str(denoise),
            "--deblur", str(deblur),
            "--auto", "true"
        ])
        
        # 执行命令
        process = subprocess.Popen(
            cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE
        )
        stdout, stderr = process.communicate()
        
        return EnhanceResult(
            success=process.returncode == 0,
            output_path=output_path,
            model=model,
            scale=scale
        )
    
    def batch_enhance(self, tasks: List[EnhanceTask]) -> List[EnhanceResult]:
        """批量增强"""
        results = []
        with ThreadPoolExecutor(max_workers=2) as executor:
            futures = {
                executor.submit(self.enhance_video, 
                              task.input_path, task.output_path,
                              task.model, task.scale, task.fps): task
                for task in tasks
            }
            for future in as_completed(futures):
                results.append(future.result())
        return results
```

# ffmpeg 工具类
class FFmpegToolkit:
    """ffmpeg 工具集"""
    
    def __init__(self, ffmpeg_path: str = "ffmpeg"):
        self.ffmpeg_path = ffmpeg_path
        self.ffprobe_path = ffmpeg_path.replace("ffmpeg", "ffprobe")
    
    def get_video_info(self, video_path: str) -> VideoInfo:
        """获取视频信息"""
        cmd = [
            self.ffprobe_path,
            "-v", "quiet",
            "-print_format", "json",
            "-show_format",
            "-show_streams",
            video_path
        ]
        
        result = subprocess.run(cmd, capture_output=True, text=True)
        data = json.loads(result.stdout)
        
        video_stream = next(s for s in data["streams"] 
                          if s["codec_type"] == "video")
        audio_stream = next((s for s in data["streams"] 
                           if s["codec_type"] == "audio"), None)
        
        return VideoInfo(
            width=int(video_stream["width"]),
            height=int(video_stream["height"]),
            fps=eval(video_stream["r_frame_rate"]),
            duration=float(data["format"]["duration"]),
            codec=video_stream["codec_name"],
            bitrate=int(data["format"]["bit_rate"]),
            has_audio=audio_stream is not None
        )
    
    def convert_format(self, input_path: str, output_path: str,
                       codec: str = "libx264",
                       bitrate: str = "20M",
                       preset: str = "medium") -> bool:
        """格式转换"""
        cmd = [
            self.ffmpeg_path,
            "-i", input_path,
            "-c:v", codec,
            "-b:v", bitrate,
            "-preset", preset,
            "-y",
            output_path
        ]
        
        result = subprocess.run(cmd, capture_output=True)
        return result.returncode == 0
    
    def extract_frames(self, video_path: str, output_dir: str,
                       fps: float = 1.0,
                       format: str = "png") -> List[str]:
        """提取帧"""
        os.makedirs(output_dir, exist_ok=True)
        output_pattern = os.path.join(output_dir, f"frame_%06d.{format}")
        
        cmd = [
            self.ffmpeg_path,
            "-i", video_path,
            "-vf", f"fps={fps}",
            "-y",
            output_pattern
        ]
        
        subprocess.run(cmd, capture_output=True)
        
        frames = sorted(glob.glob(os.path.join(output_dir, f"*.{format}")))
        return frames
```

                                   webhook_url: str,
                                   payload: dict):

            async with aiohttp.ClientSession() as session:
                for attempt in range(self.retry_policy.max_retries):
                    try:
                        async with session.post(
                            webhook_url,
                            json=payload,

                        ) as resp:

                                return True
                    except Exception as e:
                        if attempt < self.retry_policy.max_retries - 1:
                            wait = self.retry_policy.backoff_factor ** attempt
                            await asyncio.sleep(wait)
                        else:
                            logger.error(f"Webhook failed: {task_id}")
                            return False
        except Exception as e:
            logger.error(f"Webhook error: {e}")

# 工作流引擎
class WorkflowEngine:
    """工作流编排引擎"""
    
    def __init__(self):
        self.workflows = {}
        self.state_store = StateStore()
        self.event_bus = EventBus()
    
    def register_workflow(self, name: str, workflow: Workflow):
        """注册工作流"""
        self.workflows[name] = workflow
    
    async def execute(self, workflow_name: str, 
                     input_data: dict,
                     context: ExecutionContext) -> WorkflowResult:
        """执行工作流"""
        workflow = self.workflows.get(workflow_name)
        if not workflow:
            raise ValueError(f"Workflow not found: {workflow_name}")
        
        execution_id = str(uuid.uuid4())
        context.execution_id = execution_id
        
        # 初始化状态
        state = WorkflowState(
            execution_id=execution_id,
            current_step=workflow.start_step,
            status="running",
            data=input_data,
            errors=[]
        )
        await self.state_store.save(state)
        
        try:
            # 执行步骤
            current_step = workflow.start_step
            while current_step:
                step = workflow.steps[current_step]
                state.current_step = current_step
                
                # 执行步骤
                try:
                    result = await step.execute(context, state.data)
                    state.data.update(result.data)
                    
                    if result.status == StepStatus.SUCCESS:
                        current_step = step.next_step
                    elif result.status == StepStatus.CONDITIONAL:
                        current_step = step.conditional_next(result.condition)
                    else:
                        state.status = "failed"
                        break
                        
                except StepError as e:
                    state.errors.append(e)
                    
                    # 错误处理策略
                    if step.retry_policy:
                        if await step.retry_policy.retry(e, state):
                            continue
                    
                    if step.fallback_step:
                        current_step = step.fallback_step
                    else:
                        state.status = "failed"
                        break
                
                # 保存中间状态
                await self.state_store.save(state)
                
                # 发布进度事件
                await self.event_bus.publish("workflow.progress", {
                    "execution_id": execution_id,
                    "step": current_step,
                    "progress": state.progress
                })
            
            if state.status != "failed":
                state.status = "completed"
            
            await self.state_store.save(state)
            
            return WorkflowResult(
                execution_id=execution_id,
                status=state.status,
                data=state.data,
                errors=state.errors
            )
            
        except Exception as e:
            state.status = "failed"
            state.errors.append(e)
            await self.state_store.save(state)
            raise

class ResourcePool:
    """资源池"""
    
    def __init__(self, max_instances: int):
        self.max_instances = max_instances
        self.instances = asyncio.Queue(maxsize=max_instances)
        self.active_count = 0
        self._lock = asyncio.Lock()
    
    async def acquire(self, timeout: int = 300) -> EngineInstance:
        """获取资源"""
        try:
            instance = await asyncio.wait_for(
                self.instances.get(), timeout=timeout
            )
            async with self._lock:
                self.active_count += 1
            return instance
        except asyncio.TimeoutError:
            raise ResourceExhaustedError("Engine pool exhausted")
    
    async def release(self, instance: EngineInstance):
        """释放资源"""
        instance.status = "idle"
        instance.current_task = None
        instance.last_used_at = datetime.utcnow()
        
        await self.instances.put(instance)
        async with self._lock:
            self.active_count -= 1

class CacheManager:
    """缓存管理器"""
    
    def __init__(self, redis_url: str):
        self.redis = redis.asyncio.from_url(redis_url)
    
    async def get_task(self, task_id: str) -> Optional[dict]:
        """获取任务缓存"""
        data = await self.redis.get(f"task:{task_id}")
        return json.loads(data) if data else None
    
    async def set_task(self, task_id: str, task_data: dict,
                      ttl: int = 3600):
        """设置任务缓存"""
        await self.redis.setex(
            f"task:{task_id}",
            ttl,
            json.dumps(task_data)
        )
    
    async def get_style_preset(self, style_name: str) -> Optional[dict]:
        """获取风格预设缓存"""
        data = await self.redis.get(f"style:{style_name}")
        return json.loads(data) if data else None
    
    async def set_style_preset(self, style_name: str, preset: dict):
        """设置风格预设缓存"""
        await self.redis.set(
            f"style:{style_name}",
            json.dumps(preset)
        )
    
    async def increment_counter(self, key: str, amount: int = 1) -> int:
        """递增计数器"""
        return await self.redis.incrby(key, amount)
    
    async def get_rate_limit(self, user_id: str, 
                            limit: int = 100,
                            window: int = 60) -> bool:
        """限流检查"""
        key = f"rate_limit:{user_id}"
        current = await self.redis.get(key)
        
        if current is None:
            await self.redis.setex(key, window, 1)
            return True
        
        if int(current) >= limit:
            return False
        
        await self.redis.incr(key)
        return True

# 消息队列
class MessageQueue:
    """消息队列"""
    
    def __init__(self, broker_url: str):
        self.broker_url = broker_url
        self.redis = redis.asyncio.from_url(broker_url)
    
    async def publish(self, queue: str, message: dict):
        """发布消息"""
        await self.redis.rpush(queue, json.dumps(message))
    
    async def publish_batch(self, queue: str, messages: List[dict]):
        """批量发布"""
        if messages:
            pipe = self.redis.pipeline()
            for msg in messages:
                pipe.rpush(queue, json.dumps(msg))
            await pipe.execute()
    
    async def consume(self, queue: str) -> Optional[dict]:
        """消费消息"""
        result = await self.redis.blpop(queue, timeout=5)
        if result:
            _, data = result
            return json.loads(data)
        return None
    
    async def queue_length(self, queue: str) -> int:
        """队列长度"""
        return await self.redis.llen(queue)

# 对象存储抽象层
class ObjectStorage:
    """对象存储抽象"""
    
    def __init__(self, provider: str = "s3", **kwargs):
        self.provider = provider
        self.client = self._create_client(provider, **kwargs)
    
    def _create_client(self, provider: str, **kwargs):
        """创建存储客户端"""
        if provider == "s3":
            import boto3
            return boto3.client(
                "s3",
                aws_access_key_id=kwargs.get("access_key"),
                aws_secret_access_key=kwargs.get("secret_key"),
                endpoint_url=kwargs.get("endpoint_url"),
                region_name=kwargs.get("region", "us-east-1")
            )
        elif provider == "aliyun":
            import oss2
            auth = oss2.Auth(
                kwargs.get("access_key"),
                kwargs.get("secret_key")
            )
            return oss2.Bucket(
                auth,
                kwargs.get("endpoint"),
                kwargs.get("bucket")
            )
        elif provider == "local":
            return LocalStorage(kwargs.get("base_dir"))
        else:
            raise ValueError(f"Unknown provider: {provider}")
    
    async def upload(self, local_path: str, bucket: str, 
                    key: str) -> bool:
        """上传文件"""
        if self.provider == "s3":
            self.client.upload_file(local_path, bucket, key)
        elif self.provider == "local":
            self.client.upload(local_path, key)
        return True
    
    async def download(self, bucket: str, key: str,
                      local_path: str) -> bool:
        """下载文件"""
        if self.provider == "s3":
            self.client.download_file(bucket, key, local_path)
        elif self.provider == "local":
            self.client.download(key, local_path)
        return True
    
    async def get_presigned_url(self, bucket: str, key: str,
                               expires_in: int = 3600) -> str:
        """获取预签名URL"""
        if self.provider == "s3":
            return self.client.generate_presigned_url(
                "get_object",
                Params={"Bucket": bucket, "Key": key},
                ExpiresIn=expires_in
            )
        return ""
    
    async def delete(self, bucket: str, key: str) -> bool:
        """删除文件"""
        if self.provider == "s3":
            self.client.delete_object(Bucket=bucket, Key=key)
        return True
    
    async def list_files(self, bucket: str, prefix: str = "") -> List[str]:
        """列出文件"""
        if self.provider == "s3":
            response = self.client.list_objects_v2(
                Bucket=bucket, Prefix=prefix
            )
            return [obj["Key"] for obj in response.get("Contents", [])]
        return []
