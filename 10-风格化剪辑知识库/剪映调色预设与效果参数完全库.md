## 🛠️ 剪映预设→AE批量转换脚本

### 十五、预设转换脚本

```javascript
// 剪映预设到AE效果批量转换脚本
// 使用方法：在AE中运行此脚本，选择剪映预设JSON文件

function convertJianYingPresetToAE(presetData) {
    var comp = app.project.activeItem;
    if (!(comp instanceof CompItem)) {
        alert("请选择一个合成");
        return;
    }
    
    var layer = comp.layer(1);
    if (!layer) {
        alert("合成中没有图层");
        return;
    }
    
    // 基础调色参数转换
    if (presetData.temperature) {
        var cb = layer.Effects.addProperty("ADBE Color Balance (HLS)");
        var temp = presetData.temperature * 0.4;
        cb.property("Hue").setValue(temp);
    }
    
    if (presetData.saturation) {
        var sat = layer.Effects.addProperty("ADBE Saturation");
        sat.property("Master Saturation").setValue(presetData.saturation + 100);
    }
    
    if (presetData.contrast) {
        var curves = layer.Effects.addProperty("ADBE Curves");
        // 简化的对比度调整
        var curveValue = curves.property("Curve").value;
        // 实际项目中需要更复杂的曲线调整
    }
    
    if (presetData.grain) {
        var fg = layer.Effects.addProperty("ADBE Film Grain");
        fg.property("Grain Size").setValue(presetData.grain * 0.5);
        fg.property("Grain Intensity").setValue(presetData.grain * 0.8);
    }
    
    if (presetData.vignette) {
        var vg = layer.Effects.addProperty("ADBE Vignette");
        vg.property("Amount").setValue(presetData.vignette);
        vg.property("Size").setValue(Math.max(50, 70 - presetData.vignette / 3));
    }
    
    alert("剪映预设转换完成！");
}
```
