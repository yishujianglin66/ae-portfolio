```python
from unified_tool_integrator import UnifiedToolIntegrator

# 初始化调度器

    default_mode="simulate",

    preset_file="workflow_presets_library.json",
    on_progress=lambda w, p, m: print(f"{p*100:.0f}% {m}")

# 列出所有预设
```

### 2.4 自动化脚本模板

**PowerShell批处理脚本（AE渲染后自动触发Topaz）**：

```powershell
# AE渲染完成后，自动触发Topaz增强+FFmpeg编码
param(
    [string]$InputFile,
    [string]$Model = "Proteus",
    [int]$Scale = 2,
    [string]$OutputDir = "D:\AE-Work\成品库"
)

# 1. Topaz Video AI CLI增强
$topazCli = "C:\Program Files\Topaz Labs LLC\Topaz Video AI\Topaz Video AI.exe"
$topazArgs = @(
    "-i", "`"$InputFile`"",
    "-o", "`"$OutputDir\_temp_topaz.mov`"",
    "-m", $Model,
    "-s", $Scale,
    "--output-format", "prores"
)
Start-Process -FilePath $topazCli -ArgumentList $topazArgs -Wait -NoNewWindow

# 2. FFmpeg H.265硬件编码
$ffmpeg = "ffmpeg.exe"
$ffmpegArgs = @(
    "-i", "`"$OutputDir\_temp_topaz.mov`"",
    "-c:v", "hevc_nvenc",
    "-preset", "slow",
    "-cq", "23",
    "-c:a", "aac",
    "-b:a", "320k",
    "`"$OutputDir\$([System.IO.Path]::GetFileNameWithoutExtension($InputFile))_final.mp4`""
)
Start-Process -FilePath $ffmpeg -ArgumentList $ffmpegArgs -Wait -NoNewWindow

# 3. 清理临时文件
Remove-Item "$OutputDir\_temp_topaz.mov" -Force
Write-Host "处理完成: $($InputFile) → 已增强+编码"
```

### 3.2 四种集成模式

### 4.4 Python自动化桥接脚本

**Blender→AE 摄像机动画导出脚本（Blender Python）**：

```python
import bpy
import json
import os

# 导出摄像机关键帧数据到JSON，供AE读取
def export_camera_to_ae(camera_obj, output_path):
    """
    将Blender摄像机动画导出为AE可读取的JSON格式
    """
    scene = bpy.context.scene
    frame_start = scene.frame_start
    frame_end = scene.frame_end
    fps = scene.render.fps / scene.render.fps_base
    
    camera_data = {
        "fps": fps,
        "duration": (frame_end - frame_start + 1) / fps,
        "frames": []
    }
    
    for frame in range(frame_start, frame_end + 1):
        scene.frame_set(frame)
        
        # 获取摄像机变换矩阵
        matrix = camera_obj.matrix_world
        location = matrix.to_translation()
        rotation = matrix.to_euler()
        
        # 获取焦距和传感器尺寸
        cam = camera_obj.data
        focal_length = cam.lens
        sensor_width = cam.sensor_width
        sensor_height = cam.sensor_height
        
        frame_data = {
            "frame": frame,
            "position": {
                "x": location.x,
                "y": location.y,
                "z": location.z
            },
            "rotation": {
                "x": rotation.x,
                "y": rotation.y,
                "z": rotation.z
            },
            "camera": {
                "focal_length": focal_length,
                "sensor_width": sensor_width,
                "sensor_height": sensor_height,
                "fov": 2 * math.atan(sensor_width / (2 * focal_length))
            }
        }
        camera_data["frames"].append(frame_data)
    
    # 写入JSON
    with open(output_path, 'w', encoding='utf-8') as f:
        json.dump(camera_data, f, indent=2, ensure_ascii=False)
    
    print(f"摄像机数据已导出: {output_path}")
    return camera_data

# 使用示例
if __name__ == "__main__":
    cam = bpy.data.objects.get("Camera")
    if cam:
        export_camera_to_ae(cam, r"D:\AE-Work\3D渲染\camera_data.json")
```

```javascript
// AE导入Blender摄像机动画
function importCameraFromJSON(jsonPath) {
    var file = File(jsonPath);
    file.open("r");
    var content = file.read();
    file.close();
    
    var data = JSON.parse(content);
    var comp = app.project.activeItem;
    
    if (!comp || !(comp instanceof CompItem)) {
        alert("请先打开一个合成");
        return;
    }
    
    // 创建摄像机
    var cam = comp.layers.addCamera("Blender_Camera", [comp.width/2, comp.height/2]);
    
    // 设置帧速率
    comp.frameRate = data.fps;
    
    // 逐帧设置摄像机动画
    for (var i = 0; i < data.frames.length; i++) {
        var frame = data.frames[i];
        var time = (frame.frame - 1) / data.fps;
        
        // 位置转换（Blender坐标 → AE坐标）
        var pos = frame.position;
        var aePos = [
            pos.x * 100,  // 缩放因子
            -pos.z * 100, // Blender Z → AE Y (翻转)
            pos.y * 100   // Blender Y → AE Z
        ];
        
        cam.property("Position").setValueAtTime(time, aePos);
    }
    
    alert("摄像机动画导入完成: " + data.frames.length + " 帧");
}
```

```javascript
// AE脚本调用FFmpeg进行格式转换
function callFFmpeg(inputFile, outputFile, options) {
    var ffmpegPath = "ffmpeg.exe"; // 确保在PATH中
    
    var cmd = ffmpegPath + " -y -i \"" + inputFile + "\" ";
    
    if (options.videoCodec) {
        cmd += "-c:v " + options.videoCodec + " ";
    }
    if (options.videoBitrate) {
        cmd += "-b:v " + options.videoBitrate + " ";
    }
    if (options.audioCodec) {
        cmd += "-c:a " + options.audioCodec + " ";
    }
    if (options.preset) {
        cmd += "-preset " + options.preset + " ";
    }
    if (options.filters) {
        cmd += "-filter_complex \"" + options.filters + "\" ";
    }
    
    cmd += "\"" + outputFile + "\"";
    
    // 执行命令
    var result = system.callSystem(cmd);
    return result;
}

// 使用示例
var comp = app.project.activeItem;
var outputPath = "D:\\AE-Work\\_temp\\render.mov";

// 先渲染
var renderQueue = app.project.renderQueue;
var item = renderQueue.items.add(comp);
item.outputModules[0].file = File(outputPath);
item.render();

// 渲染完成后FFmpeg转码
callFFmpeg(outputPath, outputPath.replace(".mov", ".mp4"), {
    videoCodec: "libx264",
    preset: "slow",
    videoBitrate: "8M",
    audioCodec: "aac",
    audioBitrate: "320k"
});
```

## 十、性能优化与瓶颈排查

### 9.1 流水线瓶颈检测方法

| 瓶颈类型 | 检测方法 | 典型症状 | 解决方案 |
|---------|---------|---------|---------|
| **I/O瓶颈** | 磁盘使用率监测 | 渲染速度忽快忽慢/磁盘100% | SSD阵列/网络存储优化 |
| **CPU瓶颈** | CPU使用率监测 | CPU100%但GPU空闲 | 减少CPU特效/升级CPU |
| **GPU瓶颈** | GPU使用率监测 | GPU满载/显存不足 | 降低分辨率/升级GPU |
| **内存瓶颈** | 内存使用率监测 | 频繁卡顿/磁盘交换 | 增加内存/减少图层 |
| **网络瓶颈** | 网络速率监测 | 素材加载慢/协作卡顿 | 本地缓存/升级网络 |

### 9.2 十大优化策略

1. **代理工作流**：4K素材→HD代理编辑，输出时换回原素材
2. **预渲染复杂合成**：嵌套合成预渲染为视频
3. **GPU加速**：优先使用GPU效果，开启硬件编码
4. **内存优化**：关闭不必要的图层，合理设置缓存
5. **磁盘缓存**：设置独立SSD作为缓存盘
6. **分布式渲染**：多机并行渲染（aerender -mp）
7. **格式优化**：使用ProRes/DNxHR等编辑友好格式
8. **效果栈优化**：调整效果顺序，减少重复计算
9. **表达式优化**：使用valueAtTime缓存，减少重复计算
10. **版本管理**：及时清理旧版本，释放磁盘空间

### 9.3 性能基准测试表

| 操作 | 中低端配置 | 主流配置 | 高端配置 | 工作站 |
|------|----------|---------|---------|-------|
| AE 1080p预览 | 1-5 fps | 10-20 fps | 25-30 fps | 30+ fps |
| AE 4K预览 | <1 fps | 2-5 fps | 8-15 fps | 20+ fps |
| Topaz 1080p超分 | 0.5x实时 | 1-2x实时 | 3-5x实时 | 8x+实时 |

| FFmpeg H.264 1080p | 1-2x实时 | 2-5x实时 | 5-10x实时 | 15x+实时 |
